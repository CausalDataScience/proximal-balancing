\subsection{Proofs for Section~\ref{sec:identification}}

\paragraph{Exact identification.}
\begin{proof}[Proof of Theorem~\ref{thm:exact-proxy-identification}]
Because $Z_j$ is constructed from $V_j$ and the independent variable $\epsilon_j$, the two clauses of Assumption~\ref{ass:joint-latent-exchangeability} remain valid after additionally conditioning on $Z_j$. Thus, for each $a\in\{0,1\}$,
\begin{align}
\mathbb E\{Y(a)\mid A,V_j,U,Z_j\}
&=
\mathbb E\{Y(a)\mid V_j,U,Z_j\},
\label{eq:conditioning-safe-outcome-app}\\
\mathcal L(W_j\mid A,V_j,U,Z_j)
&=
K_j(\cdot\mid V_j,U).
\label{eq:common-channel-after-z-app}
\end{align}
Integrating the second identity over $Q_{b,j,z}$ and using the definition of $\mathcal K_j$ gives
\begin{align}
\mathcal L(X,W_j\mid A=b,Z_j=z)
=
\mathcal K_jQ_{b,j,z}.
\label{eq:held-out-proxy-factorization-app}
\end{align}
Exact balance in equation~\eqref{eq:exact-proxy-balance} therefore implies
\begin{align}
\mathcal K_jQ_{1,j,z}=\mathcal K_jQ_{0,j,z}
\label{eq:balanced-channel-images-app}
\end{align}
for $P_{Z_j}$-almost every $z$. Representation overlap gives $Q_{b,j,z}\ll\mathcal L(V_j,U\mid Z_j=z)$ for $b\in\{0,1\}$. Assumption~\ref{ass:consistency}, equation~\eqref{eq:conditioning-safe-outcome-app}, and Jensen's inequality give the required integrability, so $Q_{b,j,z}\in\mathcal Q_{j,z}$ almost everywhere. Assumption~\ref{ass:outcome-relevant-completeness} therefore gives, for each $a\in\{0,1\}$,
\begin{align}
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(V_j,U)\}
=
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:orc-equal-latent-means}
\end{align}
By equation~\eqref{eq:conditioning-safe-outcome-app} and iterated expectation, for each $b\in\{0,1\}$,
\begin{align}
\mathbb E\{Y(a)\mid A=b,Z_j=z\}
=
\mathbb E_{Q_{b,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\end{align}
Equation~\eqref{eq:orc-equal-latent-means} shows that the conditional mean of $Y(a)$ given $(A,Z_j)$ does not depend on $A$. Consistency then gives
\begin{align}
\mathbb E(Y\mid A=a,Z_j=z)
&=
\mathbb E\{Y(a)\mid A=a,Z_j=z\}
\nonumber\\
&=
\mathbb E\{Y(a)\mid Z_j=z\}.
\end{align}
Integrating over $P_{Z_j}$ yields equation~\eqref{eq:exact-identification}.\qedhere
\end{proof}

\paragraph{Non-nesting of canonical proximal and PROBE sufficient conditions.}
\begin{proof}[Proof of Proposition~\ref{prop:non-nesting-proximal-probe}]
We give one finite-state law for each direction. In both constructions, $X$ is constant, $Y=Y(A)$, and every treatment draw is generated by an exogenous randomizer independent of all proxy noises and outcome noises.

First, we construct a law satisfying the canonical proximal conditions but not the PROBE conditions for the designated orientation. Let
\begin{align*}
U&\sim\operatorname{Bernoulli}(1/2),
&P(A=1\mid U=u)&=\frac14+\frac12u,
\end{align*}
and let
\begin{align*}
Z^{\mathrm{tr}}=U\mathbin\oplus E_Z,
\qquad
Z^{\mathrm{out}}=W_j=U\mathbin\oplus E_W,
\end{align*}
where $E_Z$ and $E_W$ are mutually independent $\operatorname{Bernoulli}(1/4)$ variables, independent of the treatment randomizer. Set $Y(a)=a+U$ and $V_j=Z^{\mathrm{tr}}$. Consistency, integrability, and strict latent overlap hold. Independence of the exogenous noises gives the three canonical proxy-validity conditions.

The function
\begin{align*}
h(z^{\mathrm{out}},a)=a+2z^{\mathrm{out}}-\frac12
\end{align*}
is an outcome bridge because $\mathbb E(Z^{\mathrm{out}}\mid U=u)=1/4+u/2$, so $\mathbb E\{h(Z^{\mathrm{out}},a)\mid U=u\}=a+u$. Iterated expectation gives the required observed bridge equation. The binary channel from $U$ to $Z^{\mathrm{tr}}$ has matrix
\begin{align*}
\begin{pmatrix}
3/4&1/4\\
1/4&3/4
\end{pmatrix}
\end{align*}
and determinant $1/2$. Conditional on either treatment arm, both latent states have positive probability, and the corresponding unnormalized conditional-expectation matrix has determinant $3/32$. It therefore has full column rank, which proves the canonical treatment-proxy completeness condition.

Now consider any measurable randomized representation $Z_j=\phi_j(V_j,\epsilon_j)$, where $\epsilon_j$ is independent of all preceding variables. The Markov relation $U\to V_j\to Z_j$ holds. Conditional on $(U,Z_j)$, treatment and $W_j$ are independent, and
\begin{align*}
\mathbb E(A\mid U,Z_j)
=
\mathbb E(W_j\mid U,Z_j)
=
\frac14+\frac12U.
\end{align*}
The conditional covariance decomposition gives
\begin{align*}
\operatorname{Cov}(A,W_j\mid Z_j)
=
\frac14\operatorname{Var}(U\mid Z_j).
\end{align*}
Because $P(U=1\mid V_j=0)=1/4$ and $P(U=1\mid V_j=1)=3/4$,
\begin{align*}
P(U=1\mid Z_j)
=
\mathbb E\{P(U=1\mid V_j)\mid Z_j\}
\in[1/4,3/4]
\end{align*}
almost surely. Hence $\operatorname{Var}(U\mid Z_j)\ge3/16$ and
\begin{align*}
\operatorname{Cov}(A,W_j\mid Z_j)\ge\frac{3}{64}>0.
\end{align*}
Exact balance therefore fails for every such representation. This proves the first direction.

Second, we construct a law satisfying the PROBE conditions for the designated orientation but not the canonical proximal conditions. Let $U=(U_1,U_2)$, where $U_1$ and $U_2$ are independent $\operatorname{Bernoulli}(1/2)$ variables. Let $Z^{\mathrm{tr}}$ and $Z^{\mathrm{out}}$ be separate observed coordinates that agree almost surely with $U_1$, and set
\begin{align*}
V_j=Z^{\mathrm{tr}}=U_1,
\qquad
W_j=Z^{\mathrm{out}}=U_1,
\qquad
Z_j=U_1.
\end{align*}
Define
\begin{align*}
P(A=1\mid U_1,U_2)
=
\frac3{20}+\frac15U_1+\frac25U_2,
\end{align*}
and let $\xi$ be an independent mean-zero Rademacher variable, also independent of the treatment randomizer. Set
\begin{align*}
Y(a)=a+U_1+(1+U_2)\xi.
\end{align*}
The four latent treatment probabilities are $3/20$, $11/20$, $7/20$, and $15/20$, so strict latent overlap holds. Latent exchangeability follows from the independent outcome noise, and the common held-out channel holds because $W_j$ is deterministic given $U$. Moreover,
\begin{align*}
P(A=1\mid Z_j=0)=\frac7{20},
\qquad
P(A=1\mid Z_j=1)=\frac{11}{20},
\end{align*}
so representation overlap holds. Because $W_j=Z_j$ and $X$ is constant, exact balance also holds.

It remains to verify the all-$Q$ outcome-relevant completeness condition. Within a stratum $Z_j=z$, every admissible law $Q\in\mathcal Q_{j,z}$ is supported on $V_j=U_1=z$, while $U_2$ may vary. On this support,
\begin{align*}
\mu_{a,j,z}(v,u)=a+z.
\end{align*}
Thus, for any admissible $Q,Q'$ and either potential treatment level,
\begin{align*}
\mathcal K_jQ=\mathcal K_jQ'
\quad\Longrightarrow\quad
\mathbb E_Q\{\mu_{a,j,z}(V_j,U)\}
=a+z
=\mathbb E_{Q'}\{\mu_{a,j,z}(V_j,U)\}.
\end{align*}
This proves all PROBE conditions for the designated orientation.

The outcome bridge $h(z^{\mathrm{out}},a)=a+z^{\mathrm{out}}$ exists because
\begin{align*}
\mathbb E(Y\mid Z^{\mathrm{tr}}=z,A=a)
=a+z
=\mathbb E\{h(Z^{\mathrm{out}},a)\mid Z^{\mathrm{tr}}=z,A=a\}.
\end{align*}
Canonical treatment-proxy completeness nevertheless fails. In the treated arm,
\begin{align*}
P(U_2=1\mid U_1=0,A=1)=\frac{11}{14},
\qquad
P(U_2=1\mid U_1=1,A=1)=\frac{15}{22}.
\end{align*}
Define the square-integrable, nonzero function
\begin{align*}
g_1(U)
=
U_2-
\begin{cases}
11/14,&U_1=0,\\
15/22,&U_1=1.
\end{cases}
\end{align*}
Then
\begin{align*}
\mathbb E\{g_1(U)\mid Z^{\mathrm{tr}},A=1\}
=
\mathbb E\{g_1(U)\mid U_1,A=1\}
=0
\end{align*}
almost surely, although $g_1(U)$ is not almost surely zero. Hence the canonical completeness condition fails, proving the second direction.
\end{proof}

\paragraph{Exhaustive plurality identification.}
\begin{proof}[Proof of Theorem~\ref{thm:unknown-valid-block-identification}]
Under condition~\eqref{eq:unique-target-plurality}, the number of proxy splits in $\mathcal B$ whose candidate value equals $\tau$ is strictly larger than the count for every non-target value. Therefore, $\tau$ is the unique value maximizing the population count:
\begin{align*}
\{\tau\}
=
\underset{c\in\mathbb R}{\operatorname{arg\,max}}
\bigl|\{S\in\mathcal B:\theta_S=c\}\bigr|.
\end{align*}
If no non-target value occurs, the convention in Theorem~\ref{thm:unknown-valid-block-identification} compares the nonempty target cluster with zero and gives the same conclusion.
\end{proof}

\paragraph{Target-valued majority.}
\begin{proof}[Proof of Corollary~\ref{cor:target-majority-median}]
Define
\begin{align*}
N_\tau
&\triangleq
\bigl|\{S\in\mathcal B:\theta_S=\tau\}\bigr|,\\
N_c
&\triangleq
\bigl|\{S\in\mathcal B:\theta_S=c\}\bigr|,
\qquad c\ne\tau.
\end{align*}
Condition~\eqref{eq:target-valued-majority} gives $N_\tau>|\mathcal B|/2$. For every $c\ne\tau$,
\begin{align*}
N_c
\le
|\mathcal B|-N_\tau
<
N_\tau.
\end{align*}
Therefore, $\tau$ is the unique candidate value of greatest multiplicity.

It remains to establish the median statement. Let $n=|\mathcal B|$. If $n=2k+1$ is odd, then $N_\tau\ge k+1$. At most $k$ candidate values are strictly below $\tau$, and at most $k$ are strictly above $\tau$, so the $(k+1)$st order statistic equals $\tau$. If $n=2k$ is even, then $N_\tau\ge k+1$ and $n-N_\tau\le k-1$. Hence both the $k$th and the $(k+1)$st order statistics equal $\tau$. Under the conventional definition of the even-sample median as the average of these two middle order statistics, the median equals $\tau$. Thus,
\begin{align*}
\operatorname{median}\bigl((\theta_S)_{S\in\mathcal B}\bigr)=\tau.
\end{align*}
The stronger condition that more than half of the proxy splits in $\mathcal B$ satisfy the grouped assumptions is sufficient because each such proxy split has candidate value $\tau$, but it is not necessary.
\end{proof}

\paragraph{Zero discrepancy and exact balance.}
\begin{proof}[Proof of Proposition~\ref{prop:zero-discrepancy}]
The nonnegative random variable $\Delta_j(Z_j;\phi_j)$ has expectation zero exactly when it is zero almost surely. Characteristicness converts zero MMD into equality of the two conditional laws, and overlap makes this equality equivalent to conditional independence.
\end{proof}

\paragraph{Supporting bias decomposition.}
For this supporting calculation, define $\pi_j(z)\triangleq P(A=1\mid Z_j=z)$.
For each $a\in\{0,1\}$ and $P_{Z_j}$-almost every $z$, define
\begin{align}
\delta_{a,j}(z)
\triangleq
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(V_j,U)\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:stratum-mass-propensity}
\end{align}
Under Assumptions~\ref{ass:consistency} and \ref{ass:joint-latent-exchangeability}, and under $0<P(A=1\mid Z_j)<1$ almost surely, consistency and equation~\eqref{eq:conditioning-safe-outcome-app} give, for $P_{Z_j}$-almost every $z$,
\begin{align}
\mathbb E(Y\mid A=a,Z_j=z)
=
\mathbb E_{Q_{a,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:observed-arm-mean}
\end{align}
Conditional expectation over $A$ gives
\begin{align}
\mathbb E\{Y(a)\mid Z_j=z\}
={}&
\pi_j(z)\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}\}
\nonumber\\
&+\{1-\pi_j(z)\}\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}\}.
\label{eq:potential-arm-mixture}
\end{align}
Subtracting the two conditional expressions and collecting terms gives, for $P_{Z_j}$-almost every $z$,
\begin{align}
&\mathbb E(Y\mid A=1,Z_j=z)
-
\mathbb E(Y\mid A=0,Z_j=z)
\nonumber\\
&\quad-
\left[
\mathbb E\{Y(1)\mid Z_j=z\}
-
\mathbb E\{Y(0)\mid Z_j=z\}
\right]
\nonumber\\
&\qquad=
\{1-\pi_j(z)\}\delta_{1,j}(z)
+
\pi_j(z)\delta_{0,j}(z).
\label{eq:pointwise-bias-decomposition}
\end{align}

\paragraph{Approximate PROBE bias bound.}
\begin{proof}[Proof of Theorem~\ref{thm:approximate-proxy-bias-bound}]
Representation overlap gives $Q_{b,j,z}\ll\mathcal L(V_j,U\mid Z_j=z)$ for $b\in\{0,1\}$. Assumption~\ref{ass:consistency}, equation~\eqref{eq:conditioning-safe-outcome-app}, and Jensen's inequality give the required integrability, so $Q_{b,j,z}\in\mathcal Q_{j,z}$ almost everywhere.
The common held-out channel and the definition of $\Delta_j(z;\phi_j)$ give, for $P_{Z_j}$-almost every $z$,
\begin{align}
\Delta_j(z;\phi_j)
=
\operatorname{MMD}_{\kappa_j}(\mathcal K_jQ_{1,j,z},\mathcal K_jQ_{0,j,z}).
\label{eq:observed-latent-discrepancy-link}
\end{align}
Assumption~\ref{ass:uniform-stability} and equation~\eqref{eq:observed-latent-discrepancy-link} therefore give, for each $a\in\{0,1\}$ and almost every $z$,
\begin{align}
|\delta_{a,j}(z)|
\le
\Gamma_j\Delta_j(z;\phi_j).
\label{eq:stratum-bias-stability-bound}
\end{align}
The bounded kernel and finite $\Gamma_j$ imply $D_j(\phi_j)<\infty$, so the right-hand side of equation~\eqref{eq:pointwise-bias-decomposition} is integrable. Assumption~\ref{ass:consistency} makes the conditional potential-outcome contrast integrable. Equation~\eqref{eq:pointwise-bias-decomposition} therefore also makes the observed conditional contrast integrable. We may now integrate that identity and use the triangle inequality and equation~\eqref{eq:stratum-bias-stability-bound} to obtain
\begin{align}
&\left|
\mathbb E\!\left[
\mathbb E(Y\mid A=1,Z_j)
-
\mathbb E(Y\mid A=0,Z_j)
\right]
-\tau
\right|
\nonumber\\
&\le\mathbb E[\{1-\pi_j(Z_j)\}|\delta_{1,j}(Z_j)|+\pi_j(Z_j)|\delta_{0,j}(Z_j)|]\\
&\le\Gamma_j\mathbb E\{\Delta_j(Z_j;\phi_j)\}.
\end{align}
\end{proof}

\subsection{Proofs for Section~\ref{sec:estimation}}

\paragraph{Uniform discrepancy deviation.}
\begin{proof}[Proof of Proposition~\ref{prop:uniform-discrepancy-deviation}]
Equation~\eqref{eq:learning-sample-threshold} and a multiplicative Chernoff bound imply that every candidate treatment-by-stratum cell contains at least $n_{\mathrm D}p_{\min}\eta/2$ observations except on an event of probability at most $\delta_{\mathrm D}/4$. Conditional on the cell assignments, bounded Hilbert-space mean concentration and a union bound over at most $2M_jN_j$ cells give
\begin{align}
|\widehat\Delta_{j,z}(\phi)-\Delta_j(z;\phi)|
\le
2e_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\end{align}
simultaneously, with failure probability at most $\delta_{\mathrm D}/2$. A multinomial $L_1$ concentration bound gives
\begin{align}
\sum_z|\widehat p_{j,z}(\phi)-p_{j,z}(\phi)|
\le
\sqrt{
\frac{
2\left[
M_j\log2+
\log\left(4N_j/\delta_{\mathrm D}\right)
\right]
}{n_{\mathrm D}}
}
\end{align}
simultaneously over candidates, with failure probability at most $\delta_{\mathrm D}/4$. On the intersection of the three events, use $0\le\Delta_j(z;\phi)\le2\rho_j$ and the triangle inequality to obtain equation~\eqref{eq:uniform-discrepancy-deviation}. The three failure probabilities sum to $\delta_{\mathrm D}$.
\end{proof}

\paragraph{Representation-learning oracle inequality.}
\begin{proof}[Proof of Theorem~\ref{thm:representation-oracle}]
On the uniform-deviation event,
\begin{align}
D_j(\widehat\phi_j)
&\le
\widehat D_j(\widehat\phi_j)+r_{j,n_{\mathrm D}}(\delta_{\mathrm D})\\
&\le
\inf_{\phi\in\mathcal F_j}\widehat D_j(\phi)
+\varepsilon_{\mathrm{opt},j}
+r_{j,n_{\mathrm D}}(\delta_{\mathrm D})\\
&\le
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+\varepsilon_{\mathrm{opt},j}.
\end{align}
This proves equation~\eqref{eq:representation-oracle-discrepancy}. Conditional on $\mathcal I_{\mathrm D}$, the selected map is fixed. Theorem~\ref{thm:approximate-proxy-bias-bound} and the common stability budget give $|\tau_{\widehat Z_j}-\tau|\le\overline\Gamma_jD_j(\widehat\phi_j)$. Substitution proves equation~\eqref{eq:representation-oracle-causal}.
\end{proof}

\paragraph{Exact signed nuisance remainder.}
\begin{proof}[Proof of Proposition~\ref{prop:aipw-remainder}]
Fix $\widehat Z_j=z$. Taking the conditional expectation of the score and using equation~\eqref{eq:aipw-nuisances} gives
\begin{align}
\mathbb E\{S_j(O,\epsilon_j)\mid\widehat Z_j=z\}
={}&
\widehat m_{1,j}(z)-\widehat m_{0,j}(z)
\nonumber\\
&+
\frac{e_j(z)\{m_{1,j}(z)-\widehat m_{1,j}(z)\}}
{\widehat e_j(z)}
\nonumber\\
&-
\frac{\{1-e_j(z)\}\{m_{0,j}(z)-\widehat m_{0,j}(z)\}}
{1-\widehat e_j(z)}.
\end{align}
Subtract $m_{1,j}(z)-m_{0,j}(z)$ and collect the two nuisance errors. Averaging the resulting identity over $\widehat Z_j$ proves equation~\eqref{eq:aipw-remainder}.
\end{proof}

\paragraph{Conditional finite-sample AIPW error.}
\begin{proof}[Proof of Theorem~\ref{thm:conditional-aipw-error}]
The boundedness and clipping conditions imply
\begin{align}
|S_j(O,\epsilon_j)|
\le
2B\left(1+\frac{1}{\eta}\right).
\end{align}
Conditional on the two learning samples, Hoeffding's inequality bounds the difference between the evaluation average and its conditional expectation by the first term in equation~\eqref{eq:aipw-error-radius}. Proposition~\ref{prop:aipw-remainder}, Cauchy--Schwarz, and the propensity clipping bound control the absolute conditional remainder by
\begin{align}
\frac{q_{e,j}(q_{0,j}+q_{1,j})}{\eta}.
\end{align}
The triangle inequality proves equation~\eqref{eq:conditional-aipw-error}.
\end{proof}

\paragraph{End-to-end sensitivity bound.}
\begin{proof}[Proof of Theorem~\ref{thm:end-to-end-sensitivity}]
The triangle inequality gives
\begin{align}
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
\le
|\widehat\tau_j^{\mathrm{AIPW}}-\tau_{\widehat Z_j}|
+
|\tau_{\widehat Z_j}-\tau|.
\label{eq:end-to-end-triangle}
\end{align}
Theorem~\ref{thm:conditional-aipw-error} bounds the first term. Theorem~\ref{thm:approximate-proxy-bias-bound} and uniform candidate stability bound the second term by $\overline\Gamma_jD_j(\widehat\phi_j)$. Proposition~\ref{prop:uniform-discrepancy-deviation} gives
\begin{align}
D_j(\widehat\phi_j)
\le
\widehat D_j(\widehat\phi_j)
+
r_{j,n_{\mathrm D}}(\delta_{\mathrm D}),
\end{align}
which proves equation~\eqref{eq:end-to-end-empirical}. Equation~\eqref{eq:representation-oracle-discrepancy} proves the oracle form. A union bound over the discrepancy-learning and evaluation events gives the stated joint probability.
\end{proof}

\paragraph{Consistency.}
\begin{proof}[Proof of Corollary~\ref{cor:learned-aipw-consistency}]
The first three conditions make the evaluation-sample and nuisance-product components of $\varepsilon_{\mathrm{AIPW},j}$ converge to zero. Equation~\eqref{eq:consistency-representation-condition} makes the stability-weighted representation term vanish. The failure probability of Theorem~\ref{thm:end-to-end-sensitivity} also converges to zero, which proves convergence in probability.
\end{proof}

\paragraph{Random orientation search.}
\begin{proof}[Proof of Theorem~\ref{thm:random-orientation-plurality}]
Fix $m$ with $P(M=m)>0$. Conditional on $(M=m,R=r)$, the retained orientations form a uniformly sampled $r$-element subset of $\mathcal B$. For a non-target value $c$, define the finite-population variable
\begin{align*}
H_c(S)
\triangleq
\mathbf 1\{\theta_S=\tau\}
-
\mathbf 1\{\theta_S=c\},
\qquad S\in\mathcal B.
\end{align*}
It takes values in $[-1,1]$, and its population mean satisfies
\begin{align*}
\frac{1}{N}\sum_{S\in\mathcal B}H_c(S)
=
p_\tau-\frac{n(c)}{N}
\ge
p_\tau-p_\star
=
\Delta.
\end{align*}
Let $\overline H_{c,r}$ be its average over the retained orientations. The event that value $c$ ties or defeats the target value implies $\overline H_{c,r}\le0$. Hoeffding's inequality for sampling without replacement therefore gives
\begin{align*}
P\!\left(
\overline H_{c,r}\le0
\mid M=m,R=r
\right)
&\le
\exp\left\{
-\frac{2r\Delta^2}{(1-(-1))^2}
\right\}\\
&=
\exp\left\{-\frac{r\Delta^2}{2}\right\}.
\end{align*}
When $R=r\ge r_0$, a union bound over the $L$ non-target values yields
\begin{align*}
P(\text{failure}\mid M=m,R=r)
\le
L\exp\left\{-\frac{r_0\Delta^2}{2}\right\}.
\end{align*}
Combining this bound with
\begin{align*}
P(R<r_0\mid M=m)\le\eta_R(m)
\end{align*}
gives
\begin{align*}
P(\text{success}\mid M=m)
\ge
\left[
1-\eta_R(m)
-
L\exp\left\{-\frac{r_0\Delta^2}{2}\right\}
\right]_+.
\end{align*}
Here a tie and the event $R=0$ are counted as failures. If $L=0$, every orientation in $\mathcal B$ has candidate value $\tau$, so any retained candidate returns $\tau$ and the procedure succeeds whenever $R\ge1$.
\end{proof}

If $M$ is random and independent of the observed data and learning outcomes, then conditional on every realized $(M,R)=(m,r)$ the retained set remains a uniform $r$-element subset of $\mathcal B$. Therefore, if $P(R\ge r_0)\ge1-\eta_R$, the same argument after averaging over $M$ gives the unconditional bound in the main text with $\eta_R(m)$ replaced by $\eta_R$. When $L=0$, success occurs whenever $R\ge1$.
