\subsection{Proofs for Section~\ref{sec:identification}}

\paragraph{Sufficient conditions for outcome-relevant completeness.}
\begin{proof}[Proof of Proposition~\ref{prop:orc-sufficient}]
Fix $z$ and $Q,Q'\in\mathcal Q_{j,z}$ with $\mathcal K_jQ=\mathcal K_jQ'$. Write $\pi_Q$ for the law of $(X,U)$ under $Q$, $\pi_{Q,X}$ for its $X$-marginal, and $\pi_Q(\cdot\mid x)$ for its conditional law of $U$ given $X=x$. By condition (i), $K_j(\cdot\mid v,u)=K_j(\cdot\mid x,u)$, so the law $\mathcal K_jQ$ of $(X,W_j)$ has $X$-marginal $\pi_{Q,X}$ and conditional law $\int K_j(\cdot\mid x,u)\,\pi_Q(du\mid x)$ given $X=x$. The equality $\mathcal K_jQ=\mathcal K_jQ'$ therefore gives $\pi_{Q,X}=\pi_{Q',X}$ and, for $\pi_{Q,X}$-almost every $x$, $\int K_j(\cdot\mid x,u)\,\pi_Q(du\mid x)=\int K_j(\cdot\mid x,u)\,\pi_{Q'}(du\mid x)$. Condition (ii) yields $\pi_Q(\cdot\mid x)=\pi_{Q'}(\cdot\mid x)$ for almost every $x$, hence $\pi_Q=\pi_{Q'}$. By condition (i), $\mu_{a,j,z}(v,u)=\bar\mu_{a,z}(x,u)$ for a measurable function $\bar\mu_{a,z}$, so $\mathbb E_Q\{\mu_{a,j,z}(V_j,U)\}=\int\bar\mu_{a,z}\,d\pi_Q=\int\bar\mu_{a,z}\,d\pi_{Q'}=\mathbb E_{Q'}\{\mu_{a,j,z}(V_j,U)\}$, which is Assumption~\ref{ass:outcome-relevant-completeness}.

For case (a), the law of $W_j$ given $X=x$ is the convolution of the pushforward of $\pi$ under $g(x,\cdot)$ with the law of $\epsilon_j$. Its characteristic function is the product of the characteristic function of the pushforward and that of $\epsilon_j$, and the latter vanishes nowhere by assumption. Two laws $\pi$ and $\pi'$ with equal convolutions therefore have pushforwards with equal characteristic functions, hence equal pushforwards by the uniqueness theorem for characteristic functions, and injectivity of $g(x,\cdot)$ gives $\pi=\pi'$. Thus (ii) holds.

For case (b), the probability generating function of $W_j$ given $X=x$ under a law $\pi$ of $U$ is $\mathbb E(\prod_\ell s_\ell^{W_{j,\ell}})=\int\exp\{-\sum_\ell\lambda_\ell(x,u)(1-s_\ell)\}\,\pi(du)$ for $s\in[0,1]^{d}$, which is the Laplace transform of the pushforward of $\pi$ under $\lambda(x,\cdot)$ evaluated on $[0,1]^{d}$. A finite measure on $[0,\infty)^{d}$ is determined by its Laplace transform on a set with nonempty interior, so two laws with equal mixtures have equal pushforwards, and injectivity of $\lambda(x,\cdot)$ gives $\pi=\pi'$. In the scalar case this is the identifiability of Poisson mixtures of \citet{teicher1961identifiability}. Thus (ii) holds.

\end{proof}

\paragraph{Exact identification.}
\begin{proof}[Proof of Theorem~\ref{thm:exact-proxy-identification}]
Because $Z_j=\phi_j(V_j)$ is a measurable function of $V_j$, conditioning on $(U,V_j,Z_j)$ is equivalent to conditioning on $(U,V_j)$. The two clauses of Assumption~\ref{ass:joint-latent-exchangeability} therefore remain valid after additionally conditioning on $Z_j$. Thus, for each $a\in\{0,1\}$,
\begin{align}
\mathbb E\{Y(a)\mid A,V_j,U,Z_j\}
&=
\mathbb E\{Y(a)\mid V_j,U,Z_j\},
\label{eq:conditioning-safe-outcome-app}\\
\mathcal L(W_j\mid A,V_j,U,Z_j)
&=
K_j(\cdot\mid V_j,U).
\label{eq:common-channel-after-z-app}
\end{align}
Integrating the second identity over $Q_{b,j,z}$ and using the definition of $\mathcal K_j$ gives
\begin{align}
\mathcal L(X,W_j\mid A=b,Z_j=z)
=
\mathcal K_jQ_{b,j,z}.
\label{eq:held-out-proxy-factorization-app}
\end{align}
Exact balance in equation~\eqref{eq:exact-proxy-balance} therefore implies
\begin{align}
\mathcal K_jQ_{1,j,z}=\mathcal K_jQ_{0,j,z}
\label{eq:balanced-channel-images-app}
\end{align}
for $P_{Z_j}$-almost every $z$. Representation overlap gives $Q_{b,j,z}\ll\mathcal L(V_j,U\mid Z_j=z)$ for $b\in\{0,1\}$. Assumption~\ref{ass:consistency}, equation~\eqref{eq:conditioning-safe-outcome-app}, and Jensen's inequality give the required integrability, so $Q_{b,j,z}\in\mathcal Q_{j,z}$ almost everywhere. Assumption~\ref{ass:outcome-relevant-completeness} therefore gives, for each $a\in\{0,1\}$,
\begin{align}
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(V_j,U)\}
=
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:orc-equal-latent-means}
\end{align}
By equation~\eqref{eq:conditioning-safe-outcome-app} and iterated expectation, for each $b\in\{0,1\}$,
\begin{align}
\mathbb E\{Y(a)\mid A=b,Z_j=z\}
=
\mathbb E_{Q_{b,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\end{align}
Equation~\eqref{eq:orc-equal-latent-means} shows that the conditional mean of $Y(a)$ given $(A,Z_j)$ does not depend on $A$. Consistency then gives
\begin{align}
\mathbb E(Y\mid A=a,Z_j=z)
&=
\mathbb E\{Y(a)\mid A=a,Z_j=z\}
\nonumber\\
&=
\mathbb E\{Y(a)\mid Z_j=z\}.
\end{align}
Integrating over $P_{Z_j}$ yields equation~\eqref{eq:exact-identification}.\qedhere
\end{proof}

\paragraph{Non-nesting of canonical proximal and PROBE sufficient conditions.}
\begin{proof}[Proof of Proposition~\ref{prop:non-nesting-proximal-probe}]
We give one finite-state law for each direction. In both constructions, $X$ is constant, $Y=Y(A)$, and every treatment draw is generated by an exogenous randomizer independent of all proxy noises and outcome noises.

First, we construct a law satisfying the canonical proximal conditions but not the PROBE conditions for the designated proxy split. Let
\begin{align*}
U&\sim\operatorname{Bernoulli}(1/2),
&P(A=1\mid U=u)&=\frac14+\frac12u,
\end{align*}
and let
\begin{align*}
Z^{\mathrm{tr}}=U\mathbin\oplus E_Z,
\qquad
Z^{\mathrm{out}}=W_j=U\mathbin\oplus E_W,
\end{align*}
where $E_Z$ and $E_W$ are mutually independent $\operatorname{Bernoulli}(1/4)$ variables, independent of the treatment randomizer. Set $Y(a)=a+U$ and $V_j=Z^{\mathrm{tr}}$. Consistency, integrability, and strict latent overlap hold. Independence of the exogenous noises gives the three canonical proxy-validity conditions.

The function
\begin{align*}
h(z^{\mathrm{out}},a)=a+2z^{\mathrm{out}}-\frac12
\end{align*}
is an outcome bridge because $\mathbb E(Z^{\mathrm{out}}\mid U=u)=1/4+u/2$, so $\mathbb E\{h(Z^{\mathrm{out}},a)\mid U=u\}=a+u$. Iterated expectation gives the required observed bridge equation. The binary channel from $U$ to $Z^{\mathrm{tr}}$ has matrix
\begin{align*}
\begin{pmatrix}
3/4&1/4\\
1/4&3/4
\end{pmatrix}
\end{align*}
and determinant $1/2$. Conditional on either treatment arm, both latent states have positive probability, and the corresponding unnormalized conditional-expectation matrix has determinant $3/32$. It therefore has full column rank, which proves the canonical treatment-proxy completeness condition.

Now consider any measurable representation $Z_j=\phi_j(V_j)$. The Markov relation $U\to V_j\to Z_j$ holds. Conditional on $(U,Z_j)$, treatment and $W_j$ are independent, and
\begin{align*}
\mathbb E(A\mid U,Z_j)
=
\mathbb E(W_j\mid U,Z_j)
=
\frac14+\frac12U.
\end{align*}
The conditional covariance decomposition gives
\begin{align*}
\operatorname{Cov}(A,W_j\mid Z_j)
=
\frac14\operatorname{Var}(U\mid Z_j).
\end{align*}
Because $P(U=1\mid V_j=0)=1/4$ and $P(U=1\mid V_j=1)=3/4$,
\begin{align*}
P(U=1\mid Z_j)
=
\mathbb E\{P(U=1\mid V_j)\mid Z_j\}
\in[1/4,3/4]
\end{align*}
almost surely. Hence $\operatorname{Var}(U\mid Z_j)\ge3/16$ and
\begin{align*}
\operatorname{Cov}(A,W_j\mid Z_j)\ge\frac{3}{64}>0.
\end{align*}
Exact balance therefore fails for every such representation. This proves the first direction.

Second, we construct a law satisfying the PROBE conditions for the designated proxy split but not the canonical proximal conditions. Let $U=(U_1,U_2)$, where $U_1$ and $U_2$ are independent $\operatorname{Bernoulli}(1/2)$ variables. Let $Z^{\mathrm{tr}}$ and $Z^{\mathrm{out}}$ be separate observed coordinates that agree almost surely with $U_1$, and set
\begin{align*}
V_j=Z^{\mathrm{tr}}=U_1,
\qquad
W_j=Z^{\mathrm{out}}=U_1,
\qquad
Z_j=U_1.
\end{align*}
Define
\begin{align*}
P(A=1\mid U_1,U_2)
=
\frac3{20}+\frac15U_1+\frac25U_2,
\end{align*}
and let $\xi$ be an independent mean-zero Rademacher variable, also independent of the treatment randomizer. Set
\begin{align*}
Y(a)=a+U_1+(1+U_2)\xi.
\end{align*}
The four latent treatment probabilities are $3/20$, $11/20$, $7/20$, and $15/20$, so strict latent overlap holds. Latent exchangeability follows from the independent outcome noise, and the common held-out channel holds because $W_j$ is deterministic given $U$. Moreover,
\begin{align*}
P(A=1\mid Z_j=0)=\frac7{20},
\qquad
P(A=1\mid Z_j=1)=\frac{11}{20},
\end{align*}
so representation overlap holds. Because $W_j=Z_j$ and $X$ is constant, exact balance also holds.

It remains to verify the all-$Q$ outcome-relevant completeness condition. Within a stratum $Z_j=z$, every admissible law $Q\in\mathcal Q_{j,z}$ is supported on $V_j=U_1=z$, while $U_2$ may vary. On this support,
\begin{align*}
\mu_{a,j,z}(v,u)=a+z.
\end{align*}
Thus, for any admissible $Q,Q'$ and either potential treatment level,
\begin{align*}
\mathcal K_jQ=\mathcal K_jQ'
\quad\Longrightarrow\quad
\mathbb E_Q\{\mu_{a,j,z}(V_j,U)\}
=a+z
=\mathbb E_{Q'}\{\mu_{a,j,z}(V_j,U)\}.
\end{align*}
This proves all PROBE conditions for the designated proxy split.

The outcome bridge $h(z^{\mathrm{out}},a)=a+z^{\mathrm{out}}$ exists because
\begin{align*}
\mathbb E(Y\mid Z^{\mathrm{tr}}=z,A=a)
=a+z
=\mathbb E\{h(Z^{\mathrm{out}},a)\mid Z^{\mathrm{tr}}=z,A=a\}.
\end{align*}
Canonical treatment-proxy completeness nevertheless fails. In the treated arm,
\begin{align*}
P(U_2=1\mid U_1=0,A=1)=\frac{11}{14},
\qquad
P(U_2=1\mid U_1=1,A=1)=\frac{15}{22}.
\end{align*}
Define the square-integrable, nonzero function
\begin{align*}
g_1(U)
=
U_2-
\begin{cases}
11/14,&U_1=0,\\
15/22,&U_1=1.
\end{cases}
\end{align*}
Then
\begin{align*}
\mathbb E\{g_1(U)\mid Z^{\mathrm{tr}},A=1\}
=
\mathbb E\{g_1(U)\mid U_1,A=1\}
=0
\end{align*}
almost surely, although $g_1(U)$ is not almost surely zero. Hence the canonical completeness condition fails, proving the second direction.
\end{proof}

\paragraph{Exhaustive plurality identification.}
\begin{proof}[Proof of Theorem~\ref{thm:unknown-valid-block-identification}]
Under condition~\eqref{eq:unique-target-plurality}, the number of proxy splits in $\mathcal B$ whose candidate value equals $\tau$ is strictly larger than the count for every non-target value. Therefore, $\tau$ is the unique value maximizing the population count:
\begin{align*}
\{\tau\}
=
\underset{c\in\mathbb R}{\operatorname{arg\,max}}
\bigl|\{S\in\mathcal B:\theta_S=c\}\bigr|.
\end{align*}
If no non-target value occurs, the convention in Theorem~\ref{thm:unknown-valid-block-identification} compares the nonempty target cluster with zero and gives the same conclusion.
\end{proof}

\paragraph{Target-valued majority.}
\begin{proof}[Proof of Corollary~\ref{cor:target-majority-median}]
Define
\begin{align*}
N_\tau
&\triangleq
\bigl|\{S\in\mathcal B:\theta_S=\tau\}\bigr|,\\
N_c
&\triangleq
\bigl|\{S\in\mathcal B:\theta_S=c\}\bigr|,
\qquad c\ne\tau.
\end{align*}
Condition~\eqref{eq:target-valued-majority} gives $N_\tau>|\mathcal B|/2$. For every $c\ne\tau$,
\begin{align*}
N_c
\le
|\mathcal B|-N_\tau
<
N_\tau.
\end{align*}
Therefore, $\tau$ is the unique candidate value of greatest multiplicity.

It remains to establish the median statement. Let $n=|\mathcal B|$. If $n=2k+1$ is odd, then $N_\tau\ge k+1$. At most $k$ candidate values are strictly below $\tau$, and at most $k$ are strictly above $\tau$, so the $(k+1)$st order statistic equals $\tau$. If $n=2k$ is even, then $N_\tau\ge k+1$ and $n-N_\tau\le k-1$. Hence both the $k$th and the $(k+1)$st order statistics equal $\tau$. Under the conventional definition of the even-sample median as the average of these two middle order statistics, the median equals $\tau$. Thus,
\begin{align*}
\operatorname{median}\bigl((\theta_S)_{S\in\mathcal B}\bigr)=\tau.
\end{align*}
The stronger condition that more than half of the proxy splits in $\mathcal B$ satisfy the grouped assumptions is sufficient because each such proxy split has candidate value $\tau$, but it is not necessary.
\end{proof}

\paragraph{Zero discrepancy and exact balance.}
\begin{proof}[Proof of Proposition~\ref{prop:zero-discrepancy}]
By Definition~\ref{def:held-out-proxy-discrepancy}, $D_{\mathrm{res},j}(\phi)=0$ if and only if
\begin{align}
q_{j,\phi}(T_j,Z_j^\phi)=e_{j,\phi}(Z_j^\phi)
\quad\text{almost surely}.
\end{align}
Because $A$ is binary, this equality is equivalent to $T_j\perp\!\!\!\perp A\mid Z_j^\phi$.
\end{proof}

\paragraph{Supporting bias decomposition.}
Fix a representation $\phi$ and abbreviate $e(z)=e_{j,\phi}(z)$. For each $a\in\{0,1\}$ and $P_{Z_j^\phi}$-almost every $z$, define
\begin{align}
\delta_{a,j}^{\phi}(z)
\triangleq
\mathbb E_{Q_{1,j,z}^{\phi}}\{\mu_{a,j,z}^{\phi}(V_j,U)\}
-
\mathbb E_{Q_{0,j,z}^{\phi}}\{\mu_{a,j,z}^{\phi}(V_j,U)\}.
\label{eq:stratum-mass-propensity}
\end{align}
Under Assumptions~\ref{ass:consistency} and \ref{ass:joint-latent-exchangeability}, and under $0<e(Z_j^\phi)<1$ almost surely, the conditioning argument used in equation~\eqref{eq:conditioning-safe-outcome-app} gives
\begin{align}
\mathbb E(Y\mid A=a,Z_j^\phi=z)
=
\mathbb E_{Q_{a,j,z}^{\phi}}\{\mu_{a,j,z}^{\phi}(V_j,U)\}.
\label{eq:observed-arm-mean}
\end{align}
Conditional expectation over $A$ gives
\begin{align}
\mathbb E\{Y(a)\mid Z_j^\phi=z\}
={}&
e(z)\mathbb E_{Q_{1,j,z}^{\phi}}\{\mu_{a,j,z}^{\phi}\}
\nonumber\\
&+\{1-e(z)\}\mathbb E_{Q_{0,j,z}^{\phi}}\{\mu_{a,j,z}^{\phi}\}.
\label{eq:potential-arm-mixture}
\end{align}
Subtracting the two conditional expressions and collecting terms gives
\begin{align}
&\mathbb E(Y\mid A=1,Z_j^\phi=z)
-\mathbb E(Y\mid A=0,Z_j^\phi=z)
\nonumber\\
&\quad-
\left[
\mathbb E\{Y(1)\mid Z_j^\phi=z\}
-
\mathbb E\{Y(0)\mid Z_j^\phi=z\}
\right]
\nonumber\\
&\qquad=
\{1-e(z)\}\delta_{1,j}^{\phi}(z)
+e(z)\delta_{0,j}^{\phi}(z).
\label{eq:pointwise-bias-decomposition}
\end{align}

\paragraph{Approximate PROBE bias bound.}
\begin{proof}[Proof of Theorem~\ref{thm:approximate-proxy-bias-bound}]
Fix $z$ in the almost-sure set on which overlap and the structural conditions hold. Bayes' rule gives
\begin{align}
\frac{d\mathcal L(T_j\mid A=1,Z_j^\phi=z)}{dP_{j,\phi,z}}(t)
&=\frac{q_{j,\phi}(t,z)}{e(z)},\\
\frac{d\mathcal L(T_j\mid A=0,Z_j^\phi=z)}{dP_{j,\phi,z}}(t)
&=\frac{1-q_{j,\phi}(t,z)}{1-e(z)}.
\end{align}
Subtracting gives the density identity
\begin{align}
\frac{d\{\mathcal L(T_j\mid A=1,Z_j^\phi=z)
-\mathcal L(T_j\mid A=0,Z_j^\phi=z)\}}
{dP_{j,\phi,z}}(t)
&=
\frac{q_{j,\phi}(t,z)-e(z)}
{e(z)\{1-e(z)\}}.
\label{eq:residual-density-identity}
\end{align}
The common held-out channel gives $\mathcal K_jQ_{b,j,z}^{\phi}=\mathcal L(T_j\mid A=b,Z_j^\phi=z)$ for $b\in\{0,1\}$. \textcolor{blue}{As in the proof of Thm.~\ref{thm:exact-proxy-identification}, representation overlap and Assumption~\ref{ass:consistency} give $Q_{b,j,z}^{\phi}\in\mathcal Q_{j,z}^{\phi}$ for $b\in\{0,1\}$.} Assumption~\ref{ass:uniform-stability} therefore implies
\begin{align}
|\delta_{a,j}^{\phi}(z)|
&\le
\frac{\Gamma_j(\phi)}{e(z)\{1-e(z)\}}
\left[
\mathbb E\!\left(
\left.
\{q_{j,\phi}(T_j,z)-e(z)\}^2
\right|Z_j^\phi=z
\right)
\right]^{1/2}.
\label{eq:stratum-bias-stability-bound}
\end{align}
Apply the triangle inequality to equation~\eqref{eq:pointwise-bias-decomposition}, use equation~\eqref{eq:stratum-bias-stability-bound} for both potential outcomes, and integrate over $Z_j^\phi$. Cauchy--Schwarz gives
\begin{align}
&\left|
\mathbb E\!\left[
\mathbb E(Y\mid A=1,Z_j^\phi)
-
\mathbb E(Y\mid A=0,Z_j^\phi)
\right]
-\tau
\right|
\nonumber\\
&\le
\mathbb E\!\left[
\frac{\Gamma_j(\phi)}
{e_{j,\phi}(Z_j^\phi)\{1-e_{j,\phi}(Z_j^\phi)\}}
\left\{
\mathbb E\!\left[
\left.
\{q_{j,\phi}(T_j,Z_j^\phi)-e_{j,\phi}(Z_j^\phi)\}^2
\right|Z_j^\phi
\right]
\right\}^{1/2}
\right]\\
&\le
\frac{\Gamma_j(\phi)}{\eta(1-\eta)}D_{\mathrm{res},j}(\phi).
\end{align}
Because $\Gamma_j(\phi)$ does not depend on $z$, the overlap bound $\eta\le e_{j,\phi}(z)\le1-\eta$ gives the last inequality and proves equation~\eqref{eq:uniform-approximate-proxy-bias-bound}.
\end{proof}

\paragraph{Population sensitivity region.}
\begin{proof}[Proof of Proposition~\ref{prop:population-sensitivity-region}]
Theorem~\ref{thm:approximate-proxy-bias-bound} gives $|\tau_\phi-\tau|\le\Gamma_j(\phi)D_{\mathrm{res},j}(\phi)/\{\eta(1-\eta)\}$, and the displayed interval is the set of values of $\tau$ compatible with this inequality.
\end{proof}

\paragraph{Uniform discrepancy deviation.}
\begin{proof}[Proof of Proposition~\ref{prop:uniform-discrepancy-deviation}]
For clarity, the joint encoder--critic loss classes used in the proposition are
\begin{align}
\mathcal L_{0,j,n_{\mathrm D}}
&\triangleq
\left\{
o\mapsto\{a-h(\phi(v_j))\}^2:
\phi\in\mathcal F_{j,n_{\mathrm D}},\ h\in\mathcal H_{0,n_{\mathrm D}}
\right\},\\
\mathcal L_{1,j,n_{\mathrm D}}
&\triangleq
\left\{
o\mapsto\{a-h(t_j,\phi(v_j))\}^2:
\phi\in\mathcal F_{j,n_{\mathrm D}},\ h\in\mathcal H_{1,n_{\mathrm D}}
\right\},
\label{eq:joint-encoder-critic-loss-classes-app}
\end{align}
where $o=(v_j,t_j,a,y)$. For a loss class $\mathcal L$, its expected Rademacher complexity is
\begin{align}
\mathfrak R_{n_{\mathrm D}}(\mathcal L)
\triangleq
\mathbb E\!\left[
\sup_{\ell\in\mathcal L}
\frac1{n_{\mathrm D}}\sum_{i=1}^{n_{\mathrm D}}\sigma_i\ell(O_i)
\right],
\label{eq:expected-rademacher-complexity-app}
\end{align}
where the $\sigma_i$ are independent Rademacher signs independent of the observations. The loss classes are pointwise measurable and separable, so these suprema are measurable.
The same argument may instead be stated with outer probability and outer expectation.

Every loss in either class lies in $[0,1]$. Standard symmetrization and bounded-difference concentration therefore imply, simultaneously for $b\in\{0,1\}$, that
\begin{align}
\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}},\,h\in\mathcal H_{b,n_{\mathrm D}}}
|\widehat R_{b,\phi}(h)-R_{b,\phi}(h)|
\le
2\mathfrak R_{n_{\mathrm D}}(\mathcal L_{b,j,n_{\mathrm D}})
+\sqrt{\frac{\log(4/\delta_{\mathrm D})}{2n_{\mathrm D}}}
\label{eq:brier-risk-uniform-deviation-app}
\end{align}
with probability at least $1-\delta_{\mathrm D}$. Let $R_{b,\phi}^{\star}=\inf_{h\in\mathcal H_{b,n_{\mathrm D}}}R_{b,\phi}(h)$ and define $\widehat R_{b,\phi}^{\star}$ analogously. Equation~\eqref{eq:brier-risk-uniform-deviation-app} controls $|\widehat R_{b,\phi}^{\star}-R_{b,\phi}^{\star}|$ by the same right-hand side. Equation~\eqref{eq:approximate-critic-fits} also gives
\begin{align}
0
\le
\widehat R_{b,\phi}(\widehat h_{b,\phi})
-\widehat R_{b,\phi}^{\star}
\le
\varepsilon_{b,\phi}.
\label{eq:critic-optimization-gap-app}
\end{align}

The conditional-mean property of squared loss yields the Brier identity
\begin{align}
&\mathbb E[\{A-e_{j,\phi}(Z_j^\phi)\}^2]
-\mathbb E[\{A-q_{j,\phi}(T_j,Z_j^\phi)\}^2]
\nonumber\\
&\qquad=
\mathbb E[\{q_{j,\phi}(T_j,Z_j^\phi)-e_{j,\phi}(Z_j^\phi)\}^2]
=D_{\mathrm{res},j}^2(\phi).
\label{eq:brier-residual-identity-app}
\end{align}
It follows from equation~\eqref{eq:critic-approximation-error} that
\begin{align}
\left|
R_{0,\phi}^{\star}-R_{1,\phi}^{\star}
-D_{\mathrm{res},j}^2(\phi)
\right|
\le
a_{0,n_{\mathrm D}}(\phi)+a_{1,n_{\mathrm D}}(\phi)
\le a_{j,n_{\mathrm D}}.
\label{eq:population-critic-gap-app}
\end{align}
Combine equations~\eqref{eq:brier-risk-uniform-deviation-app}--\eqref{eq:population-critic-gap-app}. The positive-part map is one-Lipschitz and $D_{\mathrm{res},j}^2(\phi)\ge0$, so taking the supremum over $\phi$ gives equation~\eqref{eq:uniform-discrepancy-deviation}.
\end{proof}

\paragraph{Representation-learning oracle inequality.}
\begin{proof}[Proof of Proposition~\ref{thm:representation-oracle}]
On the event in equation~\eqref{eq:uniform-discrepancy-deviation}, set
\begin{align}
\xi_{j,n_{\mathrm D}}
\triangleq
r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}.
\end{align}
The deviation inequality at $\widehat\phi_j$, Equation~\eqref{eq:approximate-erm}, and the deviation inequality at a near-minimizer of $D_{\mathrm{res},j}^2$ over $\mathcal F_{j,n_{\mathrm D}}$ give
\begin{align}
D_{\mathrm{res},j}^2(\widehat\phi_j)
&\le
\widetilde D_{j,n_{\mathrm D}}^2(\widehat\phi_j)+\xi_{j,n_{\mathrm D}}\\
&\le
\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}
\widetilde D_{j,n_{\mathrm D}}^2(\phi)
+\varepsilon_{\mathrm{enc},j}+\xi_{j,n_{\mathrm D}}\\
&\le
\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}
D_{\mathrm{res},j}^2(\phi)
+2\xi_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{enc},j},
\end{align}
which proves equation~\eqref{eq:representation-oracle-discrepancy}.
\end{proof}

\paragraph{Causal error of the learned representation.}
\begin{proof}[Proof of Theorem~\ref{cor:representation-causal}]
Conditional on the training observations $\{O_i:i\in\mathcal I_{\mathrm D}\}$ and any algorithmic seed used in training, the selected encoder is fixed; the population expectations below use a fresh observation independent of these training inputs. Equation~\eqref{eq:uniform-approximate-proxy-bias-bound} gives
\begin{align}
|\tau_{\widehat Z_j}-\tau|
\le
\frac{\Gamma_j(\widehat\phi_j)}{\eta(1-\eta)}
D_{\mathrm{res},j}(\widehat\phi_j).
\end{align}
Taking the square root of equation~\eqref{eq:representation-oracle-discrepancy} and substituting it into this display proves equation~\eqref{eq:representation-oracle-causal}.
\end{proof}

\paragraph{Exact signed nuisance remainder.}
\begin{proof}[Proof of Proposition~\ref{prop:aipw-remainder}]
Fix $\widehat Z_j=z$. Taking the conditional expectation of the score and using equation~\eqref{eq:aipw-nuisances} gives
\begin{align}
\mathbb E\{\mathtt{UIF}_j(O)\mid\widehat Z_j=z\}
={}&
\widehat m_{1,j}(z)-\widehat m_{0,j}(z)
\nonumber\\
&+
\frac{e_j(z)\{m_{1,j}(z)-\widehat m_{1,j}(z)\}}
{\widehat e_j(z)}
\nonumber\\
&-
\frac{\{1-e_j(z)\}\{m_{0,j}(z)-\widehat m_{0,j}(z)\}}
{1-\widehat e_j(z)}.
\end{align}
Subtract $m_{1,j}(z)-m_{0,j}(z)$ and collect the two nuisance errors. Averaging the resulting identity over $\widehat Z_j$ proves equation~\eqref{eq:aipw-remainder}.
\end{proof}

\paragraph{Conditional finite-sample AIPW error.}
\begin{proof}[Proof of Theorem~\ref{thm:conditional-aipw-error}]
 Conditional on the two learning samples, the evaluation observations are independent and the score $\mathtt{UIF}_j(O)$ has variance $\sigma_j^2$, so Chebyshev's inequality bounds the difference between the evaluation average and its conditional expectation by $\sigma_j/\sqrt{n_{\mathrm E}\delta_{\mathrm E}}$, the first term in equation~\eqref{eq:aipw-error-radius}, with probability at least $1-\delta_{\mathrm E}$. Proposition~\ref{prop:aipw-remainder}, Cauchy--Schwarz, and the propensity clipping bound control the absolute conditional remainder by
\begin{align}
\frac{q_{e,j}(q_{0,j}+q_{1,j})}{\eta}.
\end{align}
The triangle inequality proves equation~\eqref{eq:conditional-aipw-error}.
For equation~\eqref{eq:end-to-end-sensitivity}, the triangle inequality gives $|\widehat\tau_j^{\mathrm{AIPW}}-\tau|\le|\widehat\tau_j^{\mathrm{AIPW}}-\tau_{\widehat Z_j}|+|\tau_{\widehat Z_j}-\tau|$. Equation~\eqref{eq:conditional-aipw-error} bounds the first term on the same event, and equation~\eqref{eq:uniform-approximate-proxy-bias-bound} bounds the second term by $\Gamma_j(\widehat\phi_j)D_{\mathrm{res},j}(\widehat\phi_j)/\{\eta(1-\eta)\}$.
\end{proof}

\paragraph{Consistency.}
\begin{proof}[Proof of Corollary~\ref{cor:learned-aipw-consistency}]
 Condition (C1) makes the evaluation term $\sigma_{j,n}/\sqrt{n_{\mathrm E}\delta_{\mathrm E,n}}$ of $\varepsilon_{\mathrm{AIPW},j}$ vanish, condition (C2) makes its nuisance-product term vanish, and condition (C3) makes the stability-weighted residual-discrepancy term in equation~\eqref{eq:end-to-end-sensitivity} vanish. The failure probability $\delta_{\mathrm E,n}$ of Theorem~\ref{thm:conditional-aipw-error} also converges to zero, which proves convergence in probability.
\end{proof}

\paragraph[Plurality under random subsampling.]{ Plurality under random subsampling.}
\begin{lemma}[Plurality under random subsampling]
\label{lem:random-subsampling-plurality}
\normalfont

Let $\widehat{\mathcal B}\subseteq\mathfrak S$ have $N\ge1$ elements, let each $S\in\widehat{\mathcal B}$ carry a value $\theta_S\in\mathbb R$, and define $n(c)$, $L$, $p_\tau$, $p_\star$, and $\Delta$ from these values by Equations~\eqref{eq:population-orientation-count} and~\eqref{eq:population-plurality-gap}. Suppose $\Delta>0$. Draw $m$ elements of $\mathfrak S$ uniformly without replacement, let $\mathcal R_m$ be the drawn elements that belong to $\widehat{\mathcal B}$, and let $R=|\mathcal R_m|$, so that $R\sim\operatorname{Hypergeometric}(T,N,m)$. Then the value $\tau$ is attained by strictly more elements of $\mathcal R_m$ than any other value with probability at least $1-P(R=0)-L\,\mathbb E\{e^{-R\Delta^2/2}\}$.

\end{lemma}
\begin{proof}[ Proof of Lemma~\ref{lem:random-subsampling-plurality}]
 Conditional on $R=r\ge1$, the retained proxy splits form a uniformly sampled $r$-element subset of $\widehat{\mathcal B}$, because the $m$ draws are uniform without replacement from $\mathfrak S$ and $\widehat{\mathcal B}$ is a fixed subset of $\mathfrak S$. For a non-target value $c$, define the finite-population variable
\begin{align*}
H_c(S)
\triangleq
\mathbf 1\{\theta_S=\tau\}
-
\mathbf 1\{\theta_S=c\},
\qquad S\in\widehat{\mathcal B}.
\end{align*}
It takes values in $[-1,1]$, and its population mean satisfies
\begin{align*}
\frac{1}{N}\sum_{S\in\widehat{\mathcal B}}H_c(S)
=
p_\tau-\frac{n(c)}{N}
\ge
p_\tau-p_\star
=
\Delta.
\end{align*}
Let $\overline H_{c,r}$ be its average over the retained proxy splits. The event that value $c$ ties or defeats the target value implies $\overline H_{c,r}\le0$. Hoeffding's inequality for sampling without replacement \citep{hoeffding1963probability} therefore gives
\begin{align*}
P\!\left(
\overline H_{c,r}\le0
\mid R=r
\right)
&\le
\exp\left\{
-\frac{2r\Delta^2}{(1-(-1))^2}
\right\}\\
&=
\exp\left\{-\frac{r\Delta^2}{2}\right\}.
\end{align*}
 A union bound over the $L$ non-target values yields
\begin{align*}
P(\text{failure}\mid R=r)
\le
L\exp\left\{-\frac{r\Delta^2}{2}\right\},\qquad r\ge1,
\end{align*}
where failure denotes the event that some non-target value ties or defeats $\tau$.

Averaging over the distribution of $R$ in Equation~\eqref{eq:retained-orientation-count} and counting $R=0$ as a failure gives

\begin{align*}
P(\text{failure})
\le
P(R=0)+\sum_{r\ge1}P(R=r)\,L\,e^{-r\Delta^2/2}
\le
P(R=0)+L\,\mathbb E\bigl\{e^{-R\Delta^2/2}\bigr\}.
\end{align*}

 If $L=0$, every proxy split in $\widehat{\mathcal B}$ has candidate value $\tau$, so any retained candidate returns $\tau$ and $\tau$ is the unique retained value whenever $R\ge1$.
\end{proof}

\paragraph[Error of Algorithm 1.]{ Error of Algorithm~\ref{alg:probe-search}.}
\begin{proof}[ Proof of Theorem~\ref{thm:probe-search-error}]

Work conditionally on the learning samples. Let $E_1$ be the event of condition (b), and let $E_2$ be the event that $R\ge1$ and that $\tau$ is attained by strictly more splits in $\mathcal R_m$ than any other value. Lemma~\ref{lem:random-subsampling-plurality}, applied to $\widehat{\mathcal B}$ with the values $\theta_S$, gives $P(E_2^{\mathrm c})\le P(R=0)+L\,\mathbb E\{e^{-R\Delta^2/2}\}$, and condition (b) gives $P(E_1^{\mathrm c})\le\delta$. It remains to show that $|\widehat\tau-\tau|\le\rho$ on $E_1\cap E_2$. On $E_1$, take $S,S'\in\mathcal R_m$. If $\theta_S=\theta_{S'}$, then $|\widehat\theta_S-\widehat\theta_{S'}|\le|\widehat\theta_S-\theta_S|+|\theta_{S'}-\widehat\theta_{S'}|\le2\rho$, so $S$ and $S'$ are linked in line~\ref{ln:link} of Algorithm~\ref{alg:probe-search}. If $\theta_S\ne\theta_{S'}$, condition (c) gives $|\widehat\theta_S-\widehat\theta_{S'}|\ge|\theta_S-\theta_{S'}|-2\rho>2\rho$, so they are not linked. Hence every connected component is exactly a set $\{S\in\mathcal R_m:\theta_S=c\}$ for one value $c$, and its size is the number of retained splits with value $c$. On $E_2$, the component of the value $\tau$ is the unique largest one, so line~\ref{ln:return} returns the median of $\{\widehat\theta_S:S\in\mathcal R_m,\theta_S=\tau\}$. On $E_1$ each of these numbers lies in $[\tau-\rho,\tau+\rho]$, and the median of finitely many numbers in an interval, whether one of them or the midpoint of two of them, lies in the same interval. Therefore $|\widehat\tau-\tau|\le\rho$ on $E_1\cap E_2$, and a union bound over $E_1^{\mathrm c}$ and $E_2^{\mathrm c}$ yields Equation~\eqref{eq:probe-search-error-probability}.
\end{proof}

\paragraph{Budget for a target failure level.}
\begin{proof}[Proof of Corollary~\ref{cor:budget-rule}]
Write $c\triangleq\Delta^2/2$. First, $P(R=0)=\prod_{i=0}^{m-1}(T-N-i)/(T-i)\le(1-p)^m\le e^{-mp}$. Second, $R$ is the sum of the indicators $\mathbf 1\{S\in\widehat{\mathcal B}\}$ over the $m$ splits drawn without replacement from $\mathfrak S$, and $x\mapsto e^{-cx}$ is continuous and convex, so Theorem~4 of \citet{hoeffding1963probability} bounds $\mathbb E\{e^{-cR}\}$ by the same expectation under sampling with replacement, where the sum has the $\operatorname{Binomial}(m,p)$ distribution. Hence $\mathbb E\{e^{-cR}\}\le(1-p+pe^{-c})^m=\{1-p(1-e^{-c})\}^m\le\exp\{-mp(1-e^{-c})\}$. Since $1-e^{-c}\le1$, also $e^{-mp}\le\exp\{-mp(1-e^{-c})\}$, and adding the two bounds gives Equation~\eqref{eq:sampling-failure-closed-form}. Its right side is at most $\epsilon$ exactly when $m\ge\log\{(L+1)/\epsilon\}/\{p(1-e^{-c})\}$, and Theorem~\ref{thm:probe-search-error} then gives $P(|\widehat\tau-\tau|\le\rho)\ge1-\delta-\epsilon$. For the simpler budget, $1-e^{-c}\ge c/(1+c)=\Delta^2/(2+\Delta^2)\ge\Delta^2/3$ because $\Delta\le1$. If $L=0$, only the first bound is needed, and $e^{-mp}\le\epsilon$ when $m\ge\log(1/\epsilon)/p$.
\end{proof}

\paragraph{Median under majority.}
\begin{proof}[Proof of Corollary~\ref{cor:median-under-majority}]

 On the event of condition (b), every retained split with $\theta_S=\tau$ satisfies $|\widehat\theta_S-\tau|\le\rho$, so more than half of the values $\{\widehat\theta_S:S\in\mathcal R_m\}$ lie in $[\tau-\rho,\tau+\rho]$. The median equals one of the two middle order statistics or their midpoint, and both middle order statistics belong to the interval when more than half of the values do. Hence the median lies in $[\tau-\rho,\tau+\rho]$.

\end{proof}

\subsection{Finite-Strata MMD Specialization}
\label{app:finite-strata-mmd-specialization}

This subsection records the earlier finite-strata analysis as an independent specialization. It is not used by the continuous-representation results in the main text.

Fix a finite library $\mathcal F_j^{\mathrm{fs}}\subseteq\Phi_j$ with $N_j^{\mathrm{fs}}\triangleq|\mathcal F_j^{\mathrm{fs}}|$. Every $\phi\in\mathcal F_j^{\mathrm{fs}}$ is deterministic and takes values in a common finite set $\mathcal S_j$ of size $M_j$. Write $Z_j^\phi=\phi(V_j)$, let $\mathcal R_j(\phi)$ be its positive-mass strata, and set $p_{j,z}(\phi)=P(Z_j^\phi=z)$. Assume, uniformly over the library and $z\in\mathcal R_j(\phi)$,
\begin{align}
P(Z_j^\phi=z)&\ge p_{\min}>0,
&
\eta\le P(A=1\mid Z_j^\phi=z)&\le1-\eta
\label{eq:finite-mmd-mass-overlap}
\end{align}
for some $\eta\in(0,1/2]$. Let $\kappa_j$ be a characteristic kernel on $(X,W_j)$ with reproducing kernel Hilbert space $\mathcal H_j$ and
\begin{align}
\sup_v\sqrt{\kappa_j(v,v)}\le\rho_j<\infty.
\label{eq:finite-mmd-kernel-bound}
\end{align}
For $z\in\mathcal R_j(\phi)$, define
\begin{align}
\Delta_{j,z}^{\mathrm{MMD}}(\phi)
&\triangleq
\operatorname{MMD}_{\kappa_j}
\left\{
\mathcal L(X,W_j\mid A=1,Z_j^\phi=z),
\mathcal L(X,W_j\mid A=0,Z_j^\phi=z)
\right\},\\
D_j^{\mathrm{MMD}}(\phi)
&\triangleq
\sum_{z\in\mathcal R_j(\phi)}
p_{j,z}(\phi)\Delta_{j,z}^{\mathrm{MMD}}(\phi).
\label{eq:finite-mmd-population-discrepancy}
\end{align}

For $i\in\mathcal I_{\mathrm D}$, let $Z_{j,i}^\phi=\phi(V_{j,i})$ and define
\begin{align}
N_{a,z}^\phi
&\triangleq
\sum_{i\in\mathcal I_{\mathrm D}}
\mathbf 1\{A_i=a,Z_{j,i}^\phi=z\},
&
N_z^\phi&\triangleq N_{0,z}^\phi+N_{1,z}^\phi,\\
\widehat p_{j,z}(\phi)&\triangleq\frac{N_z^\phi}{n_{\mathrm D}},
&
\widehat m_{a,j,z}(\phi)
&\triangleq
\frac{1}{N_{a,z}^\phi}
\sum_{i\in\mathcal I_{\mathrm D}}
\mathbf 1\{A_i=a,Z_{j,i}^\phi=z\}
\kappa_j\{(X_i,W_{j,i}),\cdot\}.
\end{align}
When both treatment-arm cells are nonempty, set
\begin{align}
\widehat\Delta_{j,z}^{\mathrm{MMD}}(\phi)
&\triangleq
\|\widehat m_{1,j,z}(\phi)-\widehat m_{0,j,z}(\phi)\|_{\mathcal H_j},\\
\widehat D_j^{\mathrm{MMD}}(\phi)
&\triangleq
\sum_{z:N_z^\phi>0}
\widehat p_{j,z}(\phi)\widehat\Delta_{j,z}^{\mathrm{MMD}}(\phi).
\label{eq:finite-mmd-empirical-discrepancy}
\end{align}
If an occupied stratum contains only one treatment arm, set $\widehat D_j^{\mathrm{MMD}}(\phi)=+\infty$.

The MMD specialization uses its own structural stability condition. For every $\phi\in\mathcal F_j^{\mathrm{fs}}$, every positive-mass stratum $z$, every $a\in\{0,1\}$, and all $Q,Q'\in\mathcal Q_{j,z}$, assume
\begin{align}
\left|\mathbb E_Q\{\mu_{a,j,z}\}-\mathbb E_{Q'}\{\mu_{a,j,z}\}\right|
\le
\Gamma_j^{\mathrm{MMD}}(\phi)
\operatorname{MMD}_{\kappa_j}(\mathcal K_jQ,\mathcal K_jQ'),
\qquad
\sup_{\phi\in\mathcal F_j^{\mathrm{fs}}}
\Gamma_j^{\mathrm{MMD}}(\phi)
\le\overline\Gamma_j^{\mathrm{MMD}}<\infty.
\label{eq:finite-mmd-stability}
\end{align}

\begin{proposition}[Finite-strata MMD learning bound]
\label{prop:finite-strata-mmd-learning-bound}
Let $\delta_{\mathrm D}\in(0,1)$ and suppose
\begin{align}
n_{\mathrm D}
\ge
\frac{8}{p_{\min}\eta}
\log\left(\frac{8M_jN_j^{\mathrm{fs}}}{\delta_{\mathrm D}}\right).
\label{eq:finite-mmd-sample-threshold}
\end{align}
Define
\begin{align}
e_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
&\triangleq
\rho_j
\left[
2+\sqrt{2\log\left(\frac{4M_jN_j^{\mathrm{fs}}}{\delta_{\mathrm D}}\right)}
\right]
\sqrt{\frac{2}{n_{\mathrm D}p_{\min}\eta}},\\
r_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
&\triangleq
2e_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
+2\rho_j
\sqrt{
\frac{
2\left[M_j\log2+\log\{4N_j^{\mathrm{fs}}/\delta_{\mathrm D}\}\right]
}{n_{\mathrm D}}
}.
\label{eq:finite-mmd-radius}
\end{align}
Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
\sup_{\phi\in\mathcal F_j^{\mathrm{fs}}}
\left|
\widehat D_j^{\mathrm{MMD}}(\phi)-D_j^{\mathrm{MMD}}(\phi)
\right|
\le
r_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D}).
\label{eq:finite-mmd-uniform-deviation}
\end{align}
If $\widehat\phi_j^{\mathrm{fs}}$ obeys
\begin{align}
\widehat D_j^{\mathrm{MMD}}(\widehat\phi_j^{\mathrm{fs}})
\le
\inf_{\phi\in\mathcal F_j^{\mathrm{fs}}}
\widehat D_j^{\mathrm{MMD}}(\phi)
+\varepsilon_{\mathrm{opt},j}^{\mathrm{fs}},
\label{eq:finite-mmd-approximate-erm}
\end{align}
then, on the same event,
\begin{align}
D_j^{\mathrm{MMD}}(\widehat\phi_j^{\mathrm{fs}})
&\le
\inf_{\phi\in\mathcal F_j^{\mathrm{fs}}}
D_j^{\mathrm{MMD}}(\phi)
+2r_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
+\varepsilon_{\mathrm{opt},j}^{\mathrm{fs}},
\label{eq:finite-mmd-oracle}\\
|\tau_{\widehat Z_j^{\mathrm{fs}}}-\tau|
&\le
\overline\Gamma_j^{\mathrm{MMD}}
\left[
\inf_{\phi\in\mathcal F_j^{\mathrm{fs}}}
D_j^{\mathrm{MMD}}(\phi)
+2r_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
+\varepsilon_{\mathrm{opt},j}^{\mathrm{fs}}
\right],
\label{eq:finite-mmd-causal-oracle}
\end{align}
where $\widehat Z_j^{\mathrm{fs}}=\widehat\phi_j^{\mathrm{fs}}(V_j)$, provided the primitive structural conditions and overlap used in the exact bias decomposition hold for every candidate.
\end{proposition}

\begin{proof}
Equation~\eqref{eq:finite-mmd-sample-threshold} and a multiplicative Chernoff bound imply that every candidate treatment-by-stratum cell contains at least $n_{\mathrm D}p_{\min}\eta/2$ observations except on an event of probability at most $\delta_{\mathrm D}/4$. Conditional on the cell assignments, bounded Hilbert-space mean concentration and a union bound over at most $2M_jN_j^{\mathrm{fs}}$ cells give
\begin{align}
\left|
\widehat\Delta_{j,z}^{\mathrm{MMD}}(\phi)
-\Delta_{j,z}^{\mathrm{MMD}}(\phi)
\right|
\le
2e_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
\end{align}
simultaneously, with failure probability at most $\delta_{\mathrm D}/2$. A multinomial $L_1$ concentration bound gives
\begin{align}
\sum_z|\widehat p_{j,z}(\phi)-p_{j,z}(\phi)|
\le
\sqrt{
\frac{
2\left[M_j\log2+\log\{4N_j^{\mathrm{fs}}/\delta_{\mathrm D}\}\right]
}{n_{\mathrm D}}
}
\end{align}
simultaneously over candidates, with failure probability at most $\delta_{\mathrm D}/4$. On the intersection of these events, $0\le\Delta_{j,z}^{\mathrm{MMD}}(\phi)\le2\rho_j$ and the triangle inequality give equation~\eqref{eq:finite-mmd-uniform-deviation}.

The empirical-risk comparison yields
\begin{align}
D_j^{\mathrm{MMD}}(\widehat\phi_j^{\mathrm{fs}})
&\le
\widehat D_j^{\mathrm{MMD}}(\widehat\phi_j^{\mathrm{fs}})
+r_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})\\
&\le
\inf_{\phi\in\mathcal F_j^{\mathrm{fs}}}
D_j^{\mathrm{MMD}}(\phi)
+2r_{j,n_{\mathrm D}}^{\mathrm{MMD}}(\delta_{\mathrm D})
+\varepsilon_{\mathrm{opt},j}^{\mathrm{fs}},
\end{align}
which proves equation~\eqref{eq:finite-mmd-oracle}. The exact bias decomposition and the MMD stability condition in equation~\eqref{eq:finite-mmd-stability} give
\begin{align}
|\tau_{Z_j^\phi}-\tau|
\le
\Gamma_j^{\mathrm{MMD}}(\phi)D_j^{\mathrm{MMD}}(\phi).
\end{align}
Apply this inequality to $\widehat\phi_j^{\mathrm{fs}}$ and substitute equation~\eqref{eq:finite-mmd-oracle} to obtain equation~\eqref{eq:finite-mmd-causal-oracle}.
\end{proof}
