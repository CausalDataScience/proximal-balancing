\section{Identification}
\label{sec:identification}

We now construct the adjustment representation. Let $\mathcal X$, $\mathcal W_j$, and $\mathcal V_j$ denote the value spaces of $X$, $W_j$, and $V_j$, respectively. Fix a positive integer $M_j$, write $[M_j]\triangleq\{1,\ldots,M_j\}$, and prespecify a class $\Phi_j$ of measurable maps from $\mathcal V_j\times[0,1]$ to $[M_j]$. Let
\begin{align}
\epsilon_j&\sim\operatorname{Unif}(0,1),
\nonumber\\
\epsilon_j&\perp\!\!\!\perp
\{U,V_j,W_j,A,Y(0),Y(1)\}
\end{align}
and, for a fixed $\phi_j\in\Phi_j$, define
\begin{align}
Z_j\triangleq\phi_j(V_j,\epsilon_j).
\label{eq:fixed-view-representation}
\end{align}
The auxiliary variable $\epsilon_j$ permits a stochastic representation. A deterministic representation is the special case in which $\phi_j$ ignores its second argument. The finite range makes balance, overlap, and adjustment explicit within strata, but does not require the hidden variable $U$ to be discrete. A value $z\in[M_j]$ is retained when $P(Z_j=z)>0$.

\subsection{Assumptions}
\label{sec:identification-assumptions}

\begin{assumption}[Consistency and integrability]
\label{ass:consistency}
For each $a\in\{0,1\}$,
\begin{align}
Y=Y(A),
\qquad
\mathbb E|Y(a)|<\infty.
\label{eq:consistency}
\end{align}
\end{assumption}

\begin{assumption}[Latent exchangeability and a common held-out channel]
\label{ass:joint-latent-exchangeability}
\label{ass:held-out-separation}
For each $a\in\{0,1\}$, the following two clauses hold separately:
\begin{align}
Y(a)&\perp\!\!\!\perp A\mid(U,V_j),
\label{eq:joint-latent-exchangeability}\\
W_j&\perp\!\!\!\perp A\mid(U,V_j).
\label{eq:held-out-separation}
\end{align}
The first clause says that the hidden state $U$ and the full representation input $V_j=(X,W_{-j})$ suffice to remove confounding of treatment and each potential outcome. The second says that, after the same variables are fixed, the conditional measurement law of the held-out block is common to both treatment arms. These clauses allow $W_{-j}$ to affect treatment and outcome. They do not assert the stronger joint independence $A\perp\!\!\!\perp\{Y(a),W_j\}\mid(U,V_j)$, which is not needed below.
\end{assumption}

\begin{assumption}[Representation overlap]
\label{ass:representation-overlap}
For every retained representation value $z$,
\begin{align}
0<P(A=1\mid Z_j=z)<1.
\label{eq:representation-overlap}
\end{align}
Both treatment arms therefore occur with positive probability in every retained stratum.
\end{assumption}

We next define the objects required to state the last identification assumption. For $a\in\{0,1\}$ and any retained $z$, let
\begin{align}
Q_{a,j,z}
\triangleq
\mathcal L(V_j,U\mid A=a,Z_j=z).
\label{eq:latent-arm-law}
\end{align}
This law records both the remaining variation in the representation input and the hidden state within one treatment arm and one representation stratum.

For every measurable $C\subseteq\mathcal W_j$, define the held-out conditional law
\begin{align}
K_j(C\mid v,u)
\triangleq
P(W_j\in C\mid V_j=v,U=u).
\label{eq:held-out-channel}
\end{align}
For a probability distribution $Q$ on $(V_j,U)$ and a measurable $B\subseteq\mathcal X\times\mathcal W_j$, define
\begin{align}
(\mathcal K_jQ)(B)
\triangleq
\int
\left[
\int
\mathbf 1\{(x,w_j)\in B\}
K_j\{dw_j\mid(x,w_{-j}),u\}
\right]
Q(dx,dw_{-j},du).
\label{eq:held-out-operator}
\end{align}
Thus, $\mathcal K_jQ$ is the induced distribution of $(X,W_j)$: draw $(V_j,U)=(X,W_{-j},U)$ from $Q$, generate $W_j$ through $K_j$, and retain only $(X,W_j)$.

For each treatment level and retained stratum, define
\begin{align}
\mu_{a,j,z}(v,u)
\triangleq
\mathbb E\{Y(a)\mid V_j=v,U=u,Z_j=z\}.
\label{eq:conditional-potential-mean}
\end{align}
Let
\begin{align}
S_{j,z}
\triangleq
\operatorname{supp}\{\mathcal L(V_j,U\mid Z_j=z)\}
\end{align}
and let $\mathcal Q_{j,z}$ contain all probability distributions $Q$ supported on $S_{j,z}$ for which the two means below are finite. For $Q\in\mathcal Q_{j,z}$, write
\begin{align}
\mathbb E_Q\{\mu_{a,j,z}(V_j,U)\}
\triangleq
\int\mu_{a,j,z}(v,u)\,Q(dv,du).
\label{eq:latent-mean-shorthand}
\end{align}
The realized arm laws $Q_{0,j,z}$ and $Q_{1,j,z}$ are required to belong to $\mathcal Q_{j,z}$.

\begin{assumption}[Exact outcome-relevant completeness]
\label{ass:outcome-relevant-completeness}
For every retained $z$, every $Q,Q'\in\mathcal Q_{j,z}$, and every $a\in\{0,1\}$,
\begin{align}
\mathcal K_jQ=\mathcal K_jQ'
\quad\Longrightarrow\quad
\mathbb E_Q\{\mu_{a,j,z}(V_j,U)\}
=
\mathbb E_{Q'}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:outcome-relevant-completeness}
\end{align}
Two admissible laws that are indistinguishable through the induced distribution of $(X,W_j)$ must agree on each outcome-relevant mean. This condition is imposed over the entire class $\mathcal Q_{j,z}$ defined above, not only on the two realized arm laws. It does not require $Q=Q'$ and therefore does not imply latent balance. On the same admissible class it is weaker than full mixture injectivity, which would require $\mathcal K_jQ=\mathcal K_jQ'$ to imply $Q=Q'$. It is not uniformly comparable to standard proximal completeness because the two conditions act on different operators and domains.
\end{assumption}

\subsection{Exact Cross-View Identification}
\label{sec:exact-cross-view-identification}

\begin{lemma}[Conditioning safety]
\label{lem:conditioning-safety}
Under Assumption~\ref{ass:joint-latent-exchangeability}, for each $a\in\{0,1\}$,
\begin{align}
\mathbb E\{Y(a)\mid A,V_j,U,Z_j\}
=
\mathbb E\{Y(a)\mid V_j,U,Z_j\}.
\label{eq:conditioning-safe-outcome}
\end{align}
Under the common-channel clause of the same assumption, for every measurable $C$,
\begin{align}
P(W_j\in C\mid A,V_j,U,Z_j)
=K_j(C\mid V_j,U).
\label{eq:common-channel-after-z}
\end{align}
\end{lemma}

\begin{proof}
Conditional on $V_j$, the representation $Z_j$ depends only on the auxiliary random variable $\epsilon_j$. By the explicit independence condition imposed on $\epsilon_j$ above, adding $Z_j$ to the conditioning set does not introduce information about $A$, $Y(a)$, or $W_j$ beyond $(V_j,U)$. Applying the two separate clauses of Assumption~\ref{ass:joint-latent-exchangeability} therefore gives the two displayed conditional-law equalities. Equation~\eqref{eq:held-out-channel} identifies the common conditional law in the second equality.
\end{proof}

\begin{definition}[Adjustment functional]
\label{def:adjustment-functional}
For the fixed representation $Z_j$, define
\begin{align}
\tau_{Z_j}
\triangleq
\sum_{z:P(Z_j=z)>0}
P(Z_j=z)
\left[
\mathbb E(Y\mid A=1,Z_j=z)
-
\mathbb E(Y\mid A=0,Z_j=z)
\right].
\label{eq:adjustment-functional}
\end{align}
This is an observed-data functional. Its equality with $\tau$ has not yet been asserted.
\end{definition}

For each retained stratum, write
\begin{align}
p_{j,z}\triangleq P(Z_j=z),
\qquad
\pi_{j,z}\triangleq P(A=1\mid Z_j=z).
\label{eq:stratum-mass-propensity}
\end{align}

\begin{theorem}[Exact bias decomposition]
\label{thm:exact-bias-decomposition}
Under Assumptions~\ref{ass:consistency}--\ref{ass:representation-overlap},
\begin{align}
\tau_{Z_j}-\tau
=
\sum_zp_{j,z}
\Big[&
(1-\pi_{j,z})
\left[
\mathbb E_{Q_{1,j,z}}\{\mu_{1,j,z}\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{1,j,z}\}
\right]
\nonumber\\
&+
\pi_{j,z}
\left[
\mathbb E_{Q_{1,j,z}}\{\mu_{0,j,z}\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{0,j,z}\}
\right]
\Big].
\label{eq:exact-bias-decomposition}
\end{align}
\end{theorem}

\begin{proof}
Consistency and Lemma~\ref{lem:conditioning-safety} give
\begin{align}
\mathbb E(Y\mid A=a,Z_j=z)
=
\mathbb E_{Q_{a,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:observed-arm-mean}
\end{align}
The law of total expectation over treatment gives
\begin{align}
\mathbb E\{Y(a)\mid Z_j=z\}
={}&
\pi_{j,z}\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(V_j,U)\}
\nonumber\\
&+(1-\pi_{j,z})\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\label{eq:potential-arm-mixture}
\end{align}
Substitute these identities into $\tau_{Z_j}$ and $\tau$, subtract, and collect the coefficients of the two arm laws. This yields equation~\eqref{eq:exact-bias-decomposition}.
\end{proof}

\begin{lemma}[Cross-view factorization]
\label{lem:cross-view-factorization}
Under the common-channel clause of Assumption~\ref{ass:joint-latent-exchangeability},
\begin{align}
\mathcal L(X,W_j\mid A=a,Z_j=z)
=
\mathcal K_jQ_{a,j,z}
\label{eq:cross-view-factorization}
\end{align}
for every treatment arm and retained stratum.
\end{lemma}

\begin{proof}
For any measurable $B\subseteq\mathcal X\times\mathcal W_j$, iterated expectation over $(V_j,U)$ gives
\begin{align}
&P\{(X,W_j)\in B\mid A=a,Z_j=z\}
\nonumber\\
&\quad=
\int
\left[
\int
\mathbf 1\{(x,w_j)\in B\}
P\{dw_j\mid A=a,V_j=(x,w_{-j}),U=u,Z_j=z\}
\right]
Q_{a,j,z}(dx,dw_{-j},du).
\end{align}
Lemma~\ref{lem:conditioning-safety} replaces the inner conditional law by $K_j\{dw_j\mid(x,w_{-j}),u\}$. Equation~\eqref{eq:held-out-operator} then gives the stated factorization.
\end{proof}

\begin{definition}[Exact PROBE balance]
\label{def:exact-cross-view-balance}
The fixed representation satisfies exact PROBE balance when
\begin{align}
(X,W_j)\perp\!\!\!\perp A\mid Z_j.
\label{eq:exact-cross-view-balance}
\end{align}
Under representation overlap, this condition is equivalent to equality of the two conditional laws of $(X,W_j)$ in every retained stratum.
\end{definition}

\begin{remark}[Why copying the representation input is not an automatic solution]
Outside the finite-range representation class, consider the non-compressive benchmark $Z_j=V_j$. Then balance of $X$ is automatic because $X$ is contained in $V_j$. Balance of the held-out block is not automatic. Exact PROBE balance reduces to
\begin{align}
W_j\perp\!\!\!\perp A\mid V_j.
\end{align}
Thus, self-copying is permitted by the identification theory but does not automatically pass the observable balance endpoint. Representation learning instead seeks a lower-dimensional tradeoff between held-out balance and representation overlap.
\end{remark}

Exact PROBE balance and Lemma~\ref{lem:cross-view-factorization} imply
\begin{align}
\mathcal K_jQ_{1,j,z}
=
\mathcal K_jQ_{0,j,z}
\label{eq:balanced-channel-images}
\end{align}
for every retained stratum.

\begin{theorem}[Exact identification under PROBE balance]
\label{thm:exact-cross-view-identification}
Suppose Assumptions~\ref{ass:consistency}--\ref{ass:outcome-relevant-completeness} hold for the fixed block and representation. If $Z_j$ satisfies exact PROBE balance, then, for each $a\in\{0,1\}$ and retained $z$,
\begin{align}
\mathbb E\{Y(a)\mid A=1,Z_j=z\}
=
\mathbb E\{Y(a)\mid A=0,Z_j=z\}.
\label{eq:conditional-mean-exchangeability}
\end{align}
Consequently,
\begin{align}
\boxed{\tau_{Z_j}=\tau,}
\label{eq:exact-identification}
\end{align}
and the population ATE is point identified by equation~\eqref{eq:adjustment-functional}.
\end{theorem}

\begin{proof}
Fix a retained stratum. Exact PROBE balance and Lemma~\ref{lem:cross-view-factorization} give equation~\eqref{eq:balanced-channel-images}. Both arm laws belong to $\mathcal Q_{j,z}$, so outcome-relevant completeness gives
\begin{align}
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(V_j,U)\}
=
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(V_j,U)\}
\end{align}
for each treatment level $a$. Lemma~\ref{lem:conditioning-safety} identifies these two quantities as the corresponding arm-specific conditional means of $Y(a)$, proving equation~\eqref{eq:conditional-mean-exchangeability}. Both differences in the exact bias decomposition are therefore zero, which proves $\tau_{Z_j}=\tau$.
\end{proof}

\begin{remark}[What is, and is not, identified]
\label{rem:no-full-latent-balance}
The theorem establishes conditional mean exchangeability, which is sufficient for ATE adjustment. It does not assert the stronger conditional independence $Y(a)\perp\!\!\!\perp A\mid Z_j$, equality of the arm laws $Q_{1,j,z}=Q_{0,j,z}$, or latent balance.
\end{remark}

\begin{remark}[Attainability]
\label{rem:exact-attainability}
The theorem is conditional on a representation attaining exact PROBE balance. It does not guarantee that such a representation exists in $\Phi_j$. A restricted class may leave positive residual imbalance, which motivates the next subsection.
\end{remark}

\subsection{Approximate Cross-View Bias Bound}
\label{sec:approximate-cross-view-bound}

Let $\kappa_j$ be a bounded, measurable, positive-semidefinite kernel on the support of $(X,W_j)$, and let $\mathcal H_j$ be its reproducing kernel Hilbert space. For a distribution $P$ of $(X,W_j)$, define its kernel mean embedding
\begin{align}
m_j(P)
\triangleq
\int\kappa_j(r,\cdot)\,P(dr)
\in\mathcal H_j.
\end{align}
For two such distributions $P$ and $P'$, define
\begin{align}
\operatorname{MMD}_{\kappa_j}(P,P')
\triangleq
\|m_j(P)-m_j(P')\|_{\mathcal H_j}.
\end{align}
Assume that $\kappa_j$ is characteristic, meaning that this discrepancy is zero exactly when $P=P'$.

\begin{definition}[Cross-view discrepancy]
\label{def:cross-view-discrepancy}
For each retained stratum, define
\begin{align}
\Delta_{j,z}(\phi_j)
\triangleq
\operatorname{MMD}_{\kappa_j}
\left\{
\mathcal L(X,W_j\mid A=1,Z_j=z),
\mathcal L(X,W_j\mid A=0,Z_j=z)
\right\}.
\label{eq:stratum-cross-view-discrepancy}
\end{align}
The aggregate discrepancy is
\begin{align}
D_j(\phi_j)
\triangleq
\sum_zp_{j,z}\Delta_{j,z}(\phi_j),
\label{eq:aggregate-cross-view-discrepancy}
\end{align}
where the sum is over retained strata.
\end{definition}

\begin{proposition}[Zero discrepancy and exact balance]
\label{prop:zero-discrepancy}
Under representation overlap and the characteristic-kernel condition,
\begin{align}
\boxed{
D_j(\phi_j)=0
\quad\Longleftrightarrow\quad
(X,W_j)\perp\!\!\!\perp A\mid Z_j.
}
\label{eq:zero-discrepancy-equivalence}
\end{align}
\end{proposition}

\begin{proof}
Every summand is nonnegative and every retained stratum has positive mass. Thus, $D_j(\phi_j)=0$ exactly when every stratum-specific MMD is zero. Characteristicness converts each zero MMD into equality of the two conditional laws, and overlap makes those equalities equivalent to conditional independence.
\end{proof}

The factorization lemma also gives
\begin{align}
\Delta_{j,z}(\phi_j)
=
\operatorname{MMD}_{\kappa_j}
\left(
\mathcal K_jQ_{1,j,z},
\mathcal K_jQ_{0,j,z}
\right).
\label{eq:observed-latent-discrepancy-link}
\end{align}

\begin{assumption}[Uniform outcome-relevant stability]
\label{ass:uniform-stability}
There is a finite constant $\Gamma_j$ such that, for every retained $z$, every $a\in\{0,1\}$, and every $Q,Q'\in\mathcal Q_{j,z}$,
\begin{align}
&\left|
\mathbb E_Q\{\mu_{a,j,z}(V_j,U)\}
-
\mathbb E_{Q'}\{\mu_{a,j,z}(V_j,U)\}
\right|
\nonumber\\
&\qquad\le
\Gamma_j
\operatorname{MMD}_{\kappa_j}
\left(
\mathcal K_jQ,
\mathcal K_jQ'
\right).
\label{eq:uniform-stability}
\end{align}
\end{assumption}

The constant $\Gamma_j$ converts an observed discrepancy into a bound on an outcome-relevant latent difference. It is a structural sensitivity parameter and is not generally identified from the observed distribution.

\begin{theorem}[Approximate cross-view bias bound]
\label{thm:approximate-cross-view-bound}
Suppose Assumptions~\ref{ass:consistency}--\ref{ass:representation-overlap} and Assumption~\ref{ass:uniform-stability} hold for the fixed representation. Then
\begin{align}
\boxed{
|\tau_{Z_j}-\tau|
\le
\Gamma_jD_j(\phi_j).
}
\label{eq:approximate-cross-view-bound}
\end{align}
\end{theorem}

\begin{proof}
For each treatment level and retained stratum, define
\begin{align}
\delta_{a,j,z}
\triangleq
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(V_j,U)\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(V_j,U)\}.
\end{align}
The exact bias decomposition and the triangle inequality give
\begin{align}
|\tau_{Z_j}-\tau|
\le
\sum_zp_{j,z}
\left[
(1-\pi_{j,z})|\delta_{1,j,z}|
+
\pi_{j,z}|\delta_{0,j,z}|
\right].
\end{align}
Uniform stability and equation~\eqref{eq:observed-latent-discrepancy-link} imply
\begin{align}
|\delta_{a,j,z}|
\le
\Gamma_j\Delta_{j,z}(\phi_j).
\end{align}
Substitution and $(1-\pi_{j,z})+\pi_{j,z}=1$ yield equation~\eqref{eq:approximate-cross-view-bound}.
\end{proof}

\begin{remark}[Exact completeness and quantitative stability]
\label{rem:exact-vs-stability}
Outcome-relevant completeness preserves equality at zero discrepancy. Uniform stability controls nonzero discrepancies and therefore implies outcome-relevant completeness. The converse need not hold because equality at zero can be preserved even when no finite Lipschitz constant $\Gamma_j$ exists.
\end{remark}

\begin{remark}[Sensitivity interpretation]
\label{rem:sensitivity-interpretation}
For a scientifically specified $\Gamma_j$, the approximate bound gives
\begin{align}
\tau
\in
\left[
\tau_{Z_j}-\Gamma_jD_j(\phi_j),
\tau_{Z_j}+\Gamma_jD_j(\phi_j)
\right].
\end{align}
A small discrepancy alone does not guarantee small causal bias. The causal quantity is the product $\Gamma_jD_j(\phi_j)$.
\end{remark}

\begin{remark}[Uniformity across candidate representations]
\label{rem:uniform-across-representations}
For representation learning, let $\Gamma_j(\phi)$ be a stability budget for candidate $\phi$. Comparing candidates by discrepancy requires
\begin{align}
\sup_{\phi\in\Phi_j}\Gamma_j(\phi)
\le
\overline\Gamma_j
<\infty.
\label{eq:uniform-candidate-stability}
\end{align}
Without this uniform condition, a representation can have a small observed discrepancy and an arbitrarily large stability constant.
\end{remark}
