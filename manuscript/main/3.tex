\section{Identification}
\label{sec:identification}

We now introduce the cross-view representation. Fix one block index $j\in\{1,\ldots,J\}$ before analyzing the data. Let
\begin{align}
W_{-j}
\triangleq
(W_1,\ldots,W_{j-1},W_{j+1},\ldots,W_J)
\end{align}
contain every proxy block except $W_j$. The excluded block $W_j$ is the held-out proxy block.

Fix a positive integer $M_j$, write $[M_j]\triangleq\{1,\ldots,M_j\}$, and prespecify a class
\begin{align}
\Phi_j
\triangleq
\left\{
\phi:\mathcal X\times\mathcal W_{-j}\longrightarrow[M_j]
\right\}.
\end{align}
For a fixed map $\phi_j\in\Phi_j$, define
\begin{align}
Z_j\triangleq\phi_j(X,W_{-j}).
\label{eq:fixed-view-representation}
\end{align}
The representation is finite-valued so that balance, overlap, and adjustment can be stated within explicit strata. This choice does not require the hidden variable $U$ to be discrete. A value $z\in[M_j]$ is retained when $P(Z_j=z)>0$.

Figure~\ref{fig:compatible-proxy-dags} displays three causal graphs compatible with the broad pretreatment ordering. In every panel, $W_{-j}$ is the representation-side proxy view, $W_j$ is the held-out proxy view, and $Z_j$ is the deterministic function defined in equation~\eqref{eq:fixed-view-representation}. Solid arrows denote causal relations. Dashed arrows into the rectangular $Z_j$ node denote deterministic construction and are not causal arrows. The graphs alone do not imply the cross-view conditional independences imposed below.

The identification argument compares the hidden states of treated and untreated units within one retained stratum. For $a\in\{0,1\}$ and any retained $z$ satisfying $P(A=a,Z_j=z)>0$, define
\begin{align}
Q_{a,j,z}
\triangleq
\mathcal L(X,U\mid A=a,Z_j=z).
\label{eq:latent-arm-law}
\end{align}
For every measurable set $C\subseteq\mathcal W_j$, define the conditional measurement law
\begin{align}
K_j(C\mid x,u)
\triangleq
P(W_j\in C\mid X=x,U=u).
\label{eq:held-out-channel}
\end{align}
The law $K_j$ describes how the hidden state $(x,u)$ generates the held-out proxy block.

For any probability distribution $Q$ on $(X,U)$ and measurable set $B\subseteq\mathcal X\times\mathcal W_j$, define
\begin{align}
(\mathcal K_jQ)(B)
\triangleq
\int
\left[
\int
\mathbf 1\{(x,w_j)\in B\}
K_j(dw_j\mid x,u)
\right]
Q(dx,du).
\label{eq:held-out-operator}
\end{align}
Thus, $\mathcal K_jQ$ is the distribution of $(X,W_j)$ obtained by first drawing $(X,U)$ from $Q$ and then drawing $W_j$ from its conditional measurement law.

For each treatment level and retained stratum, define
\begin{align}
\mu_{a,j,z}(x,u)
\triangleq
\mathbb E\{Y(a)\mid X=x,U=u,Z_j=z\}.
\label{eq:conditional-potential-mean}
\end{align}
Let
\begin{align}
S_{j,z}
\triangleq
\operatorname{supp}\{\mathcal L(X,U\mid Z_j=z)\}
\end{align}
be the support of $(X,U)$ within stratum $z$. Define $\mathcal Q_{j,z}$ as the set of all probability distributions $Q$ supported on $S_{j,z}$ for which
\begin{align}
\int|\mu_{a,j,z}(x,u)|\,Q(dx,du)<\infty,
\qquad a\in\{0,1\}.
\end{align}
The two realized arm laws satisfy $Q_{0,j,z},Q_{1,j,z}\in\mathcal Q_{j,z}$. For $Q\in\mathcal Q_{j,z}$, write
\begin{align}
\mathbb E_Q\{\mu_{a,j,z}(X,U)\}
\triangleq
\int\mu_{a,j,z}(x,u)\,Q(dx,du).
\label{eq:latent-mean-shorthand}
\end{align}

\subsection{Assumptions}
\label{sec:identification-assumptions}

\begin{assumption}[Consistency and integrability]
\label{ass:consistency}
For each $a\in\{0,1\}$,
\begin{align}
Y=Y(A),
\qquad
\mathbb E|Y(a)|<\infty.
\label{eq:consistency}
\end{align}
\end{assumption}

\begin{assumption}[Joint latent exchangeability for representation inputs]
\label{ass:joint-latent-exchangeability}
For each $a\in\{0,1\}$,
\begin{align}
A\perp\!\!\!\perp\bigl(Y(a),W_{-j}\bigr)\mid(X,U).
\label{eq:joint-latent-exchangeability}
\end{align}
Once $(X,U)$ is fixed, treatment contains no additional information about either the potential outcome or the variables used to construct $Z_j$.
\end{assumption}

\begin{assumption}[Separation of the held-out proxy block]
\label{ass:held-out-separation}
\begin{align}
W_j\perp\!\!\!\perp(A,W_{-j})\mid(X,U).
\label{eq:held-out-separation}
\end{align}
Once $(X,U)$ is fixed, the measurement law of $W_j$ is the same across treatment arms and does not depend on the representation-side proxy blocks.
\end{assumption}

\begin{assumption}[Representation overlap]
\label{ass:representation-overlap}
For every retained representation value $z$,
\begin{align}
0<P(A=1\mid Z_j=z)<1.
\label{eq:representation-overlap}
\end{align}
Both treatment arms therefore occur with positive probability in every retained stratum.
\end{assumption}

\begin{assumption}[Exact outcome-relevant completeness]
\label{ass:outcome-relevant-completeness}
For every retained $z$, every $Q,Q'\in\mathcal Q_{j,z}$, and every $a\in\{0,1\}$,
\begin{align}
\mathcal K_jQ=\mathcal K_jQ'
\quad\Longrightarrow\quad
\mathbb E_Q\{\mu_{a,j,z}(X,U)\}
=
\mathbb E_{Q'}\{\mu_{a,j,z}(X,U)\}.
\label{eq:outcome-relevant-completeness}
\end{align}
Two latent distributions that are indistinguishable through $(X,W_j)$ must agree on the mean of each potential outcome. The assumption does not require the two latent distributions themselves to be equal.
\end{assumption}

\subsection{Exact Cross-View Identification}
\label{sec:exact-cross-view-identification}

\begin{lemma}[Conditioning safety]
\label{lem:conditioning-safety}
Under Assumption~\ref{ass:joint-latent-exchangeability}, for each $a\in\{0,1\}$,
\begin{align}
Y(a)\perp\!\!\!\perp A\mid(X,U,Z_j).
\label{eq:conditioning-safe-outcome}
\end{align}
Under Assumption~\ref{ass:held-out-separation},
\begin{align}
W_j\perp\!\!\!\perp(A,Z_j)\mid(X,U),
\label{eq:conditioning-safe-proxy}
\end{align}
and hence
\begin{align}
P(W_j\in C\mid X,U,A,Z_j)
=K_j(C\mid X,U)
\label{eq:common-channel-after-z}
\end{align}
for every measurable set $C$.
\end{lemma}

\begin{proof}
The representation $Z_j$ is a measurable function of $(X,W_{-j})$. Assumption~\ref{ass:joint-latent-exchangeability} therefore implies
\begin{align}
A\perp\!\!\!\perp\bigl(Y(a),Z_j\bigr)\mid(X,U).
\end{align}
Weak union gives equation~\eqref{eq:conditioning-safe-outcome}. Similarly, Assumption~\ref{ass:held-out-separation} and measurability of $Z_j$ imply equation~\eqref{eq:conditioning-safe-proxy}. The conditional law in equation~\eqref{eq:common-channel-after-z} then equals the law in equation~\eqref{eq:held-out-channel}.
\end{proof}

\begin{definition}[Adjustment functional]
\label{def:adjustment-functional}
For the fixed representation $Z_j$, define
\begin{align}
\tau_{Z_j}
\triangleq
\sum_{z:P(Z_j=z)>0}
P(Z_j=z)
\left[
\mathbb E(Y\mid A=1,Z_j=z)
-
\mathbb E(Y\mid A=0,Z_j=z)
\right].
\label{eq:adjustment-functional}
\end{align}
This is an observed-data functional. Its equality with $\tau$ has not yet been asserted.
\end{definition}

For each retained stratum, write
\begin{align}
p_{j,z}\triangleq P(Z_j=z),
\qquad
\pi_{j,z}\triangleq P(A=1\mid Z_j=z).
\label{eq:stratum-mass-propensity}
\end{align}

\begin{theorem}[Exact bias decomposition]
\label{thm:exact-bias-decomposition}
Under Assumptions~\ref{ass:consistency}, \ref{ass:joint-latent-exchangeability}, and \ref{ass:representation-overlap},
\begin{align}
\tau_{Z_j}-\tau
=
\sum_zp_{j,z}
\Big[&
(1-\pi_{j,z})
\left[
\mathbb E_{Q_{1,j,z}}\{\mu_{1,j,z}\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{1,j,z}\}
\right]
\nonumber\\
&+
\pi_{j,z}
\left[
\mathbb E_{Q_{1,j,z}}\{\mu_{0,j,z}\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{0,j,z}\}
\right]
\Big].
\label{eq:exact-bias-decomposition}
\end{align}
\end{theorem}

\begin{proof}
Consistency and Lemma~\ref{lem:conditioning-safety} give, for $a\in\{0,1\}$,
\begin{align}
\mathbb E(Y\mid A=a,Z_j=z)
=
\mathbb E_{Q_{a,j,z}}\{\mu_{a,j,z}(X,U)\}.
\label{eq:observed-arm-mean}
\end{align}
The law of total expectation over treatment gives
\begin{align}
\mathbb E\{Y(a)\mid Z_j=z\}
={}&
\pi_{j,z}
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(X,U)\}
\nonumber\\
&+
(1-\pi_{j,z})
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(X,U)\}.
\label{eq:potential-arm-mixture}
\end{align}
Insert equation~\eqref{eq:observed-arm-mean} into Definition~\ref{def:adjustment-functional}. Insert equation~\eqref{eq:potential-arm-mixture} into the stratum decomposition of $\tau$. Subtracting the two expressions and collecting the coefficients of each latent arm law gives equation~\eqref{eq:exact-bias-decomposition}.
\end{proof}

Theorem~\ref{thm:exact-bias-decomposition} isolates the obstacle to adjustment by $Z_j$. Unbiased adjustment requires equality of two outcome-relevant latent averages within each stratum. Equality of the full latent distributions is sufficient but is not necessary.

\begin{lemma}[Cross-view factorization]
\label{lem:cross-view-factorization}
Under Assumption~\ref{ass:held-out-separation}, for every treatment arm and retained stratum,
\begin{align}
\mathcal L(X,W_j\mid A=a,Z_j=z)
=
\mathcal K_jQ_{a,j,z}.
\label{eq:cross-view-factorization}
\end{align}
\end{lemma}

\begin{proof}
Let $B$ be a measurable subset of the joint space of $(X,W_j)$. Iterated expectation gives
\begin{align}
&P\{(X,W_j)\in B\mid A=a,Z_j=z\}
\nonumber\\
&\quad=
\int
P\{(X,W_j)\in B\mid X=x,U=u,A=a,Z_j=z\}
Q_{a,j,z}(dx,du).
\end{align}
Lemma~\ref{lem:conditioning-safety} replaces the inner conditional law of $W_j$ by $K_j(\cdot\mid x,u)$. The resulting integral is $(\mathcal K_jQ_{a,j,z})(B)$ by equation~\eqref{eq:held-out-operator}. Equality for every measurable $B$ proves the result.
\end{proof}

\begin{definition}[Exact cross-view balance]
\label{def:exact-cross-view-balance}
The fixed representation satisfies exact cross-view balance when
\begin{align}
(X,W_j)\perp\!\!\!\perp A\mid Z_j.
\label{eq:exact-cross-view-balance}
\end{align}
Under Assumption~\ref{ass:representation-overlap}, this condition is equivalent to equality of the two conditional laws of $(X,W_j)$ in every retained stratum.
\end{definition}

Exact cross-view balance is a population condition on observed variables for a fixed representation. Lemma~\ref{lem:cross-view-factorization} converts it into
\begin{align}
\mathcal K_jQ_{1,j,z}
=
\mathcal K_jQ_{0,j,z}
\label{eq:balanced-channel-images}
\end{align}
for every retained stratum.

\begin{theorem}[Exact identification under cross-view balance]
\label{thm:exact-cross-view-identification}
Suppose Assumptions~\ref{ass:consistency} through \ref{ass:outcome-relevant-completeness} hold for the fixed block $j$ and fixed representation $Z_j=\phi_j(X,W_{-j})$. If $Z_j$ satisfies exact cross-view balance, then
\begin{align}
\boxed{\tau_{Z_j}=\tau.}
\label{eq:exact-identification}
\end{align}
Consequently, the population ATE is point identified by the observed functional in equation~\eqref{eq:adjustment-functional}.
\end{theorem}

\begin{proof}
Fix a retained stratum. Exact cross-view balance and Lemma~\ref{lem:cross-view-factorization} give equation~\eqref{eq:balanced-channel-images}. Both latent arm laws belong to $\mathcal Q_{j,z}$. Assumption~\ref{ass:outcome-relevant-completeness} therefore gives, for each $a\in\{0,1\}$,
\begin{align}
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(X,U)\}
=
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(X,U)\}.
\end{align}
Both treatment-arm differences in Theorem~\ref{thm:exact-bias-decomposition} are zero. Hence $\tau_{Z_j}-\tau=0$.
\end{proof}

\begin{remark}[Identification without full latent balance]
\label{rem:no-full-latent-balance}
Theorem~\ref{thm:exact-cross-view-identification} equates the outcome-relevant latent averages. It does not require $Q_{1,j,z}=Q_{0,j,z}$ and does not assert $(X,U)\perp\!\!\!\perp A\mid Z_j$. This distinction is the reason Assumption~\ref{ass:outcome-relevant-completeness} is weaker than full mixture injectivity.
\end{remark}

\begin{remark}[Attainability]
\label{rem:exact-attainability}
Theorem~\ref{thm:exact-cross-view-identification} applies when the chosen representation attains exact cross-view balance. It does not guarantee that such a map exists in $\Phi_j$. A restricted class may leave positive residual imbalance, which motivates the next subsection.
\end{remark}

\subsection{Approximate Cross-View Bias Bound}
\label{sec:approximate-cross-view-bound}

Define the held-out observed vector
\begin{align}
V_j\triangleq(X,W_j).
\end{align}
Let $\kappa_j$ be a bounded, measurable, positive-semidefinite kernel on the support of $V_j$, and let $\mathcal H_j$ be its reproducing kernel Hilbert space. For a distribution $P$ of $V_j$, define its kernel mean embedding
\begin{align}
m_j(P)
\triangleq
\int\kappa_j(v,\cdot)\,P(dv)
\in\mathcal H_j.
\end{align}
For two distributions $P$ and $P'$, define
\begin{align}
\operatorname{MMD}_{\kappa_j}(P,P')
\triangleq
\|m_j(P)-m_j(P')\|_{\mathcal H_j}.
\end{align}
Assume that $\kappa_j$ is characteristic, meaning that this discrepancy is zero exactly when $P=P'$.

\begin{definition}[Cross-view discrepancy]
\label{def:cross-view-discrepancy}
For each retained stratum, define
\begin{align}
\Delta_{j,z}(\phi_j)
\triangleq
\operatorname{MMD}_{\kappa_j}
\left\{
\mathcal L(V_j\mid A=1,Z_j=z),
\mathcal L(V_j\mid A=0,Z_j=z)
\right\}.
\label{eq:stratum-cross-view-discrepancy}
\end{align}
The aggregate discrepancy is
\begin{align}
D_j(\phi_j)
\triangleq
\sum_zp_{j,z}\Delta_{j,z}(\phi_j),
\label{eq:aggregate-cross-view-discrepancy}
\end{align}
where the sum is over retained strata. Both quantities are population functionals of observed variables.
\end{definition}

\begin{proposition}[Zero discrepancy and exact balance]
\label{prop:zero-discrepancy}
Under Assumption~\ref{ass:representation-overlap} and the characteristic-kernel condition,
\begin{align}
\boxed{
D_j(\phi_j)=0
\quad\Longleftrightarrow\quad
(X,W_j)\perp\!\!\!\perp A\mid Z_j.
}
\label{eq:zero-discrepancy-equivalence}
\end{align}
\end{proposition}

\begin{proof}
Every summand in equation~\eqref{eq:aggregate-cross-view-discrepancy} is nonnegative and every retained stratum has positive mass. Thus, the aggregate is zero exactly when every stratum-specific MMD is zero. Characteristicness converts each zero MMD into equality of the two conditional laws of $(X,W_j)$. Under overlap, these equalities over all retained strata are equivalent to conditional independence.
\end{proof}

Lemma~\ref{lem:cross-view-factorization} also gives
\begin{align}
\Delta_{j,z}(\phi_j)
=
\operatorname{MMD}_{\kappa_j}
\left(
\mathcal K_jQ_{1,j,z},
\mathcal K_jQ_{0,j,z}
\right).
\label{eq:observed-latent-discrepancy-link}
\end{align}

\begin{assumption}[Uniform outcome-relevant stability]
\label{ass:uniform-stability}
There is a finite constant $\Gamma_j$ such that, for every retained $z$, every $a\in\{0,1\}$, and every $Q,Q'\in\mathcal Q_{j,z}$,
\begin{align}
&\left|
\mathbb E_Q\{\mu_{a,j,z}(X,U)\}
-
\mathbb E_{Q'}\{\mu_{a,j,z}(X,U)\}
\right|
\nonumber\\
&\qquad\le
\Gamma_j
\operatorname{MMD}_{\kappa_j}
\left(
\mathcal K_jQ,
\mathcal K_jQ'
\right).
\label{eq:uniform-stability}
\end{align}
\end{assumption}

The constant $\Gamma_j$ converts an observed cross-view discrepancy into an upper bound on an outcome-relevant latent difference. It is a structural sensitivity parameter and is not generally identified from the observed distribution.

\begin{theorem}[Approximate cross-view bias bound]
\label{thm:approximate-cross-view-bound}
Suppose Assumptions~\ref{ass:consistency} through \ref{ass:representation-overlap} and Assumption~\ref{ass:uniform-stability} hold for the fixed representation. Then
\begin{align}
\boxed{
|\tau_{Z_j}-\tau|
\le
\Gamma_jD_j(\phi_j).
}
\label{eq:approximate-cross-view-bound}
\end{align}
\end{theorem}

\begin{proof}
For each treatment level and retained stratum, define
\begin{align}
\delta_{a,j,z}
\triangleq
\mathbb E_{Q_{1,j,z}}\{\mu_{a,j,z}(X,U)\}
-
\mathbb E_{Q_{0,j,z}}\{\mu_{a,j,z}(X,U)\}.
\end{align}
Theorem~\ref{thm:exact-bias-decomposition} and the triangle inequality give
\begin{align}
|\tau_{Z_j}-\tau|
\le
\sum_zp_{j,z}
\left[
(1-\pi_{j,z})|\delta_{1,j,z}|
+
\pi_{j,z}|\delta_{0,j,z}|
\right].
\end{align}
Assumption~\ref{ass:uniform-stability} and equation~\eqref{eq:observed-latent-discrepancy-link} imply
\begin{align}
|\delta_{a,j,z}|
\le
\Gamma_j\Delta_{j,z}(\phi_j).
\end{align}
Substitution, the identity $(1-\pi_{j,z})+\pi_{j,z}=1$, and Definition~\ref{def:cross-view-discrepancy} give equation~\eqref{eq:approximate-cross-view-bound}.
\end{proof}

\begin{remark}[Exact completeness and quantitative stability]
\label{rem:exact-vs-stability}
Assumption~\ref{ass:outcome-relevant-completeness} preserves equality at zero discrepancy. Assumption~\ref{ass:uniform-stability} controls nonzero discrepancies and therefore implies exact outcome-relevant completeness. The converse need not hold because exact equality can be preserved even when no finite Lipschitz constant $\Gamma_j$ exists.
\end{remark}

\begin{remark}[Sensitivity interpretation]
\label{rem:sensitivity-interpretation}
For a scientifically specified $\Gamma_j$, Theorem~\ref{thm:approximate-cross-view-bound} gives the population sensitivity interval
\begin{align}
\tau
\in
\left[
\tau_{Z_j}-\Gamma_jD_j(\phi_j),
\tau_{Z_j}+\Gamma_jD_j(\phi_j)
\right].
\end{align}
A small $D_j(\phi_j)$ alone does not guarantee small causal bias. The causal quantity is the product $\Gamma_jD_j(\phi_j)$.
\end{remark}

\begin{remark}[Uniformity across candidate representations]
\label{rem:uniform-across-representations}
For representation learning, let $\Gamma_j(\phi)$ be a stability budget for candidate $\phi$. A common finite budget
\begin{align}
\sup_{\phi\in\Phi_j}\Gamma_j(\phi)
\le
\overline\Gamma_j
<\infty
\label{eq:uniform-candidate-stability}
\end{align}
is required to compare candidates by discrepancy alone. Without this condition, a representation can have a small observed discrepancy and an arbitrarily large stability constant.
\end{remark}
