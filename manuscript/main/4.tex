\section{Representation Learning and AIPW Estimation}
\label{sec:estimation}

The identification results concern one fixed representation. We now select a representation from data and estimate the adjustment functional that it induces. The representation-learning and AIPW analysis below uses one prespecified held-out block $j$ throughout.

\subsection{Representation Learning}
\label{sec:representation-learning}

Recall that one observed record is $O_i=(X_i,W_i,A_i,Y_i)$. Partition the observed sample into three mutually disjoint subsets,
\begin{align}
	\mathcal I_{\mathrm D},
	\qquad
	\mathcal I_{\mathrm N},
	\qquad
	\mathcal I_{\mathrm E},
	\label{eq:honest-three-way-split}
\end{align}
with respective sizes $n_{\mathrm D}$, $n_{\mathrm N}$, and $n_{\mathrm E}$. The discrepancy-learning sample $\mathcal I_{\mathrm D}$ selects a representation. The nuisance-learning sample $\mathcal I_{\mathrm N}$ estimates the outcome regressions and propensity used by AIPW. The evaluation sample $\mathcal I_{\mathrm E}$ calculates the final estimate. Conditional on the preceding samples, the objects evaluated on the next sample are fixed.

 Let $\mathcal F_{j,n_{\mathrm D}}\subseteq\Phi_j$ be a prespecified, possibly infinite class of deterministic encoders. For $\phi\in\mathcal F_{j,n_{\mathrm D}}$, write $Z_j^\phi=\phi(V_j)$ and retain $T_j=(X,W_j)$ from Section~\ref{sec:approximate-proxy-bound}.

\begin{assumption}[Common representation overlap]
\label{ass:library-overlap}
\normalfont
There is a constant $\eta\in(0,1/2]$ such that, for every $\phi\in\mathcal F_{j,n_{\mathrm D}}$,

\begin{align}
\eta\le e_{j,\phi}(Z_j^\phi)\le1-\eta\quad\text{almost surely}.
\label{eq:library-overlap}
\end{align}

\end{assumption}
For any prespecified nonempty proper subset $S\subset\{1,\ldots,J\}$, the results below also apply with $(V_j,W_j,Z_j^\phi)$ replaced by $(V_S,W_S,\phi(V_S))$ and with the same assumptions imposed on this grouped split.

The Brier-risk definition below estimates residual discrepancy by comparing squared-loss prediction of $A$ from $Z_j^\phi$ alone with prediction from $(T_j,Z_j^\phi)$ \citep{brier1950verification}.

\begin{definition}[Brier-risk residual discrepancy]
\label{def:empirical-held-out-proxy-discrepancy}
\normalfont
 Let $\mathcal H_{0,n_{\mathrm D}}$ and $\mathcal H_{1,n_{\mathrm D}}$ be prespecified classes of measurable $[0,1]$-valued predictors with inputs $Z_j^\phi$ and $(T_j,Z_j^\phi)$, respectively, fixed independently of $\mathcal I_{\mathrm D}$. Assume $\mathcal H_{1,n_{\mathrm D}}$ contains each member of $\mathcal H_{0,n_{\mathrm D}}$ lifted to ignore $T_j$. For $b\in\{0,1\}$, define
\begin{align}
	R_{0,\phi}(h)&\triangleq\mathbb E[\{A-h(Z_j^\phi)\}^2],
	&
	\widehat R_{0,\phi}(h)&\triangleq\frac1{n_{\mathrm D}}\sum_{i\in\mathcal I_{\mathrm D}}\{A_i-h(Z_{j,i}^\phi)\}^2,\\
	R_{1,\phi}(h)&\triangleq\mathbb E[\{A-h(T_j,Z_j^\phi)\}^2],
	&
	\widehat R_{1,\phi}(h)&\triangleq\frac1{n_{\mathrm D}}\sum_{i\in\mathcal I_{\mathrm D}}\{A_i-h(T_{j,i},Z_{j,i}^\phi)\}^2,
	\label{eq:brier-risks}
\end{align}
where $Z_{j,i}^\phi=\phi(V_{j,i})$. For each $\phi$, let $\widehat h_{b,\phi}\in\mathcal H_{b,n_{\mathrm D}}$ be an approximate empirical minimizer satisfying, for some $\varepsilon_{b,\phi}\ge0$,
\begin{align}
	\widehat R_{b,\phi}(\widehat h_{b,\phi})
	\le
	\inf_{h\in\mathcal H_{b,n_{\mathrm D}}}\widehat R_{b,\phi}(h)
	+
	\varepsilon_{b,\phi},
	\qquad b\in\{0,1\}.
	\label{eq:approximate-critic-fits}
\end{align}
Define $\varepsilon_{\mathrm{crit},j}\triangleq\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}}}(\varepsilon_{0,\phi}+\varepsilon_{1,\phi})$ and
\begin{align}
	\widetilde D_{j,n_{\mathrm D}}^2(\phi)
	\triangleq
	\max\left\{(
	\widehat R_{0,\phi}(\widehat h_{0,\phi})
	-
	\widehat R_{1,\phi}(\widehat h_{1,\phi})), 0
	\right\},
	\label{eq:empirical-discrepancy}
\end{align}
\end{definition}

Each encoder $\phi$ and critic $h$ give a squared-loss function of one observation: $\{A-h(\phi(V_j))\}^2$ or $\{A-h(T_j,\phi(V_j))\}^2$. Varying both $\phi$ and $h$ over their prespecified classes gives the loss classes $\mathcal L_{0,j,n_{\mathrm D}}$ and $\mathcal L_{1,j,n_{\mathrm D}}$, respectively. Let $\mathfrak R_{n_{\mathrm D}}(\mathcal L)$ denote the expected Rademacher complexity of a loss class $\mathcal L$. For $\delta_{\mathrm D}\in(0,1)$, define

\begin{align}
	r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
	\triangleq
	2\{\mathfrak R_{n_{\mathrm D}}(\mathcal L_{0,j,n_{\mathrm D}})+\mathfrak R_{n_{\mathrm D}}(\mathcal L_{1,j,n_{\mathrm D}})\}
	+
	2\sqrt{\frac{\log(4/\delta_{\mathrm D})}{2n_{\mathrm D}}}.
	\label{eq:discrepancy-radius}
\end{align}

This radius controls sampling fluctuation uniformly over the encoder--critic choices. The complexity terms account for the flexibility of the two loss classes, and the last term sets the failure-probability level $\delta_{\mathrm D}$.

\begin{proposition}[Uniform residual-discrepancy deviation]
\label{prop:uniform-discrepancy-deviation}
\normalfont
For each $\phi\in\mathcal F_{j,n_{\mathrm D}}$, define
\begin{align}
	a_{0,n_{\mathrm D}}(\phi)
	&\triangleq
	\inf_{h\in\mathcal H_{0,n_{\mathrm D}}}R_{0,\phi}(h)
	-
	\mathbb E[\{A-e_{j,\phi}(Z_j^\phi)\}^2],\\
	a_{1,n_{\mathrm D}}(\phi)
	&\triangleq
	\inf_{h\in\mathcal H_{1,n_{\mathrm D}}}R_{1,\phi}(h)
	-
	\mathbb E[\{A-q_{j,\phi}(T_j,Z_j^\phi)\}^2],\\
	a_{j,n_{\mathrm D}}
	&\triangleq
	\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}}}
	\{a_{0,n_{\mathrm D}}(\phi)+a_{1,n_{\mathrm D}}(\phi)\}.
	\label{eq:critic-approximation-error}
\end{align}

Under the measurability conditions stated in Assumption~\ref{ass:learning-measurability}, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
	\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}}}
	|\widetilde D_{j,n_{\mathrm D}}^2(\phi)-D_{\mathrm{res},j}^2(\phi)|
	\le
	r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}.
	\label{eq:uniform-discrepancy-deviation}
\end{align}
\end{proposition}
At any fixed confidence level $1-\delta_{\mathrm D}$, this error bound is $O(n_{\mathrm D}^{-1/2})$ if both Rademacher complexities, $a_{j,n_{\mathrm D}}$, and $\varepsilon_{\mathrm{crit},j}$ are each $O(n_{\mathrm D}^{-1/2})$.

\begin{definition}[Approximate empirical minimizer]
\label{def:approximate-empirical-minimizer}
\normalfont
 For a prescribed tolerance $\varepsilon_{\mathrm{enc},j}\ge0$, an approximate empirical minimizer is an encoder $\widehat\phi_j\in\mathcal F_{j,n_{\mathrm D}}$ selected measurably from the training observations and algorithmic seed such that
\begin{align}
\widetilde D_{j,n_{\mathrm D}}^2(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}\widetilde D_{j,n_{\mathrm D}}^2(\phi)
+
\varepsilon_{\mathrm{enc},j}.
\label{eq:approximate-erm}
\end{align}
An exact empirical minimizer has $\varepsilon_{\mathrm{enc},j}=0$.

\end{definition}

\begin{proposition}[Residual-discrepancy oracle inequality]
\label{thm:representation-oracle}
\normalfont
 Let $\delta_{\mathrm D}\in(0,1)$. Suppose the prespecified encoder and $[0,1]$-valued critic classes, approximate critic fits in Proposition~\ref{prop:uniform-discrepancy-deviation} and Assumption~\ref{ass:learning-measurability} hold, and $\widehat\phi_j$ satisfies Equation~\eqref{eq:approximate-erm}. Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
 D_{\mathrm{res},j}^2(\widehat\phi_j)
 &\le
 \inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)
 +2\{r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}\}
 +\varepsilon_{\mathrm{enc},j}.
\label{eq:representation-oracle-discrepancy}
\end{align}
\end{proposition}

Condition on the training observations $\{O_i:i\in\mathcal I_{\mathrm D}\}$ and any algorithmic seed used in training, which fixes $\widehat\phi_j$. Here and below, population expectations involving this learned encoder use a fresh observation $O=(X,W,A,Y)$ independent of the training observations and seed, with this conditioning understood.
The learned representation and its population adjustment functional are

\begin{align}
Z_j^{\widehat\phi_j}
&\triangleq\widehat\phi_j(V_j),\qquad\widehat Z_j\triangleq Z_j^{\widehat\phi_j},\nonumber\\
\tau_{\widehat Z_j}
&\triangleq
\mathbb E\!\left[
\mathbb E(Y\mid A=1,\widehat Z_j)
-
\mathbb E(Y\mid A=0,\widehat Z_j)
\right].
\label{eq:learned-adjustment-functional}
\end{align}

Thus $\tau_{\widehat Z_j}=\tau_\phi\big|_{\phi=\widehat\phi_j}$, where $\tau_\phi$ is the population adjustment functional defined in Proposition~\ref{prop:population-sensitivity-region}.
The adjustment functional is used whenever its two conditional means are absolutely integrable under the law of $\widehat Z_j$.

\begin{theorem}[Causal error of the learned representation]
\label{cor:representation-causal}
\normalfont

Let $\delta_{\mathrm D}\in(0,1)$. Suppose the prespecified encoder and $[0,1]$-valued critic classes, approximate critic fits in Proposition~\ref{prop:uniform-discrepancy-deviation} and Assumption~\ref{ass:learning-measurability} hold, and $\widehat\phi_j$ satisfies Equation~\eqref{eq:approximate-erm}. Suppose Assumptions~\ref{ass:consistency}, \ref{ass:joint-latent-exchangeability}, and \ref{ass:uniform-stability} hold for every $Z_j^\phi$ with $\phi\in\mathcal F_{j,n_{\mathrm D}}$, and there is a constant $\eta\in(0,1/2]$ such that $\eta\le e_{j,\phi}(Z_j^\phi)\le1-\eta$ almost surely for every such $\phi$. Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
	|\tau_{\widehat Z_j}-\tau|
	\le
	\frac{\Gamma_j(\widehat\phi_j)}{\eta(1-\eta)}
	\left[
	\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)
	+2\{r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}\}
	+\varepsilon_{\mathrm{enc},j}
	\right]^{1/2}.
	\label{eq:representation-oracle-causal}
\end{align}
\end{theorem}
At fixed confidence and fixed overlap constant $\eta$, the right-hand side is $O_p(n_{\mathrm D}^{-1/4})$ if $\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)=0$, each of $r_{j,n_{\mathrm D}}(\delta_{\mathrm D})$, $a_{j,n_{\mathrm D}}$, $\varepsilon_{\mathrm{crit},j}$, and $\varepsilon_{\mathrm{enc},j}$ is $O(n_{\mathrm D}^{-1/2})$, and $\Gamma_j(\widehat\phi_j)=O_p(1)$. This is a conditional upper-bound rate; it is not a universal or minimax rate.

\subsection{AIPW Estimation}
\label{sec:aipw-estimation}
This section estimates the adjustment functional $\tau_{\widehat Z_j}$ of the learned representation with an augmented inverse probability weighted (AIPW) estimator \citep{robins1994estimation,tsiatis2006semiparametric} computed on the evaluation sample.
 Define the conditional outcome means and representation propensity
\begin{align}\label{eq:aipw-nuisances}
	m_{a,j}(z) &\triangleq \mathbb E(Y\mid A=a,\widehat Z_j=z),\\
	e_j(z) &\triangleq P(A=1\mid\widehat Z_j=z).
\end{align}
For the observed adjustment functional $\tau_{\widehat Z_j}$, the following is the AIPW score:
\begin{align}\label{eq:fixed-representation-eif}
	\mathtt{IF}_j(O) \triangleq & m_{1,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j) +
	\frac{A\{Y-m_{1,j}(\widehat Z_j)\}}{e_j(\widehat Z_j)}
	-
	\frac{(1-A)\{Y-m_{0,j}(\widehat Z_j)\}}{1-e_j(\widehat Z_j)}
	-
	\tau_{\widehat Z_j}.
\end{align}
When this score is square-integrable, it is the \textit{efficient influence function} in the nonparametric model for $(\widehat Z_j,A,Y)$. Bounded outcomes together with uniform representation overlap provide one sufficient condition for square integrability. Use the nuisance-learning sample, with $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i})$, to estimate the nuisance functions. Denote the estimates by $\widehat m_{0,j}$, $\widehat m_{1,j}$, and $\widehat e_j$.

\begin{definition}[Honest AIPW estimator]
\label{def:honest-aipw}
\normalfont
For $i\in\mathcal I_{\mathrm E}$, define $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i})$ and
\begin{align}\label{eq:aipw-score}
	\mathtt{UIF}_j(O_i) \triangleq{}&
	\widehat m_{1,j}(\widehat Z_{j,i})
	-
	\widehat m_{0,j}(\widehat Z_{j,i})
	+
	\frac{A_i\{Y_i-\widehat m_{1,j}(\widehat Z_{j,i})\}}
	{\widehat e_j(\widehat Z_{j,i})}
	-
	\frac{(1-A_i)\{Y_i-\widehat m_{0,j}(\widehat Z_{j,i})\}}
	{1-\widehat e_j(\widehat Z_{j,i})}.
\end{align}
The score $\mathtt{UIF}_j$ is the uncentered plug-in version of $\mathtt{IF}_j$: it replaces the nuisance functions by their estimates and omits the centering term $\tau_{\widehat Z_j}$.
The honest AIPW estimator is
\begin{align}\label{eq:honest-aipw}
	\widehat\tau_j^{\mathrm{AIPW}} \triangleq \frac{1}{n_{\mathrm E}} \sum_{i\in\mathcal I_{\mathrm E}}\mathtt{UIF}_j(O_i).
\end{align}
\end{definition}

The plug-in score has the doubly robust remainder structure of augmented inverse probability weighting \citep{chernozhukov2017double,kennedy2022semiparametric}: its conditional bias is a product of the propensity error and the outcome-regression errors, as the next proposition states exactly.
\begin{proposition}[Exact signed nuisance remainder]
\label{prop:aipw-remainder}
\normalfont
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$,
\begin{align}\label{eq:aipw-remainder}
	\mathbb E\{\mathtt{UIF}_j(O)\}-\tau_{\widehat Z_j} &=
	\mathbb E
	\Bigg[
	\{\widehat e_j(\widehat Z_j)-e_j(\widehat Z_j)\}
	\Bigg\{
	\frac{\widehat m_{1,j}(\widehat Z_j)-m_{1,j}(\widehat Z_j)}
	{\widehat e_j(\widehat Z_j)} +
	\frac{\widehat m_{0,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)}
	{1-\widehat e_j(\widehat Z_j)}
	\Bigg\}
	\Bigg].
\end{align}
\end{proposition}

The remainder is zero if the propensity estimator is correct. It is also zero if both conditional outcome estimators are correct.

Clip the propensity estimate so that $\eta\le\widehat e_j(z)\le1-\eta$. Conditional on the two learning samples, that is, on their observations and the seeds used to fit the encoder and the nuisance functions, define for a function $g$ of $\widehat Z_j$ the norm $\|g\|_{2,\widehat Z_j}\triangleq[\mathbb E\{g(\widehat Z_j)^2\mid\mathcal I_{\mathrm D},\mathcal I_{\mathrm N}\}]^{1/2}$ and the realized nuisance errors
\begin{align}
q_{e,j}\triangleq\|\widehat e_j-e_j\|_{2,\widehat Z_j},
\qquad
q_{a,j}\triangleq\|\widehat m_{a,j}-m_{a,j}\|_{2,\widehat Z_j},
\qquad a\in\{0,1\}.
\label{eq:realized-nuisance-errors}
\end{align}
Let $\sigma_j^2\triangleq\mathrm{Var}\{\mathtt{UIF}_j(O)\mid\mathcal I_{\mathrm D},\mathcal I_{\mathrm N}\}$ denote the conditional variance of the plug-in score; it is finite whenever $\mathbb E(Y^2)<\infty$ and the outcome estimates are square-integrable. For $\delta_{\mathrm E}\in(0,1)$, define
\begin{align}
\varepsilon_{\mathrm{AIPW},j}
\triangleq
\frac{\sigma_j}{\sqrt{n_{\mathrm E}\delta_{\mathrm E}}}
+
\frac{q_{e,j}(q_{0,j}+q_{1,j})}{\eta}.
\label{eq:aipw-error-radius}
\end{align}
The next theorem is the final guarantee of this section: the causal error of the honest AIPW estimator is at most the estimation radius $\varepsilon_{\mathrm{AIPW},j}$ plus the stability-weighted residual discrepancy of the learned representation.

\begin{theorem}[Finite-sample causal error of honest AIPW]
\label{thm:conditional-aipw-error}
\normalfont

Suppose $\sigma_j<\infty$. Conditional on the two learning samples, with probability at least $1-\delta_{\mathrm E}$ over $\mathcal I_{\mathrm E}$,
\begin{align}
|\widehat\tau_j^{\mathrm{AIPW}}-\tau_{\widehat Z_j}|
\le
\varepsilon_{\mathrm{AIPW},j}.
\label{eq:conditional-aipw-error}
\end{align}
If, in addition, the assumptions of Theorem~\ref{cor:representation-causal} hold, then on the same event
\begin{align}
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
\le
\varepsilon_{\mathrm{AIPW},j}
+
\frac{\Gamma_j(\widehat\phi_j)}{\eta(1-\eta)}
D_{\mathrm{res},j}(\widehat\phi_j).
\label{eq:end-to-end-sensitivity}
\end{align}

\end{theorem}

\begin{corollary}[Consistency]
\label{cor:learned-aipw-consistency}
\normalfont

Consider a sequence of encoder and critic classes, sample splits, nuisance estimators, optimization tolerances, and selected encoders $\widehat\phi_{j,n}$ indexed by the total sample size $n$, with a common overlap constant $\eta\in(0,1/2]$, such that Theorem~\ref{thm:conditional-aipw-error} applies at every $n$. Suppose the following three conditions hold.
\begin{enumerate}[label=(C\arabic*),leftmargin=*]
\item \textit{Vanishing evaluation error.} $\delta_{\mathrm E,n}\to0$ and $\sigma_{j,n}^2/(n_{\mathrm E}\delta_{\mathrm E,n})\to0$. The evaluation sample grows faster than the confidence level shrinks.
\item \textit{Vanishing nuisance product.} $q_{e,j,n}(q_{0,j,n}+q_{1,j,n})\xrightarrow{p}0$. The propensity and outcome-regression errors need not vanish individually; their product vanishes, for example, when each is $o_p(n^{-1/4})$.
\item \textit{Vanishing learned imbalance.} $\Gamma_j(\widehat\phi_{j,n})D_{\mathrm{res},j}(\widehat\phi_{j,n})\xrightarrow{p}0$. By Proposition~\ref{thm:representation-oracle}, this holds when $\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)$, the two Rademacher complexities, $a_{j,n_{\mathrm D}}$, $\varepsilon_{\mathrm{crit},j,n}$, $\varepsilon_{\mathrm{enc},j,n}$, and $\delta_{\mathrm D,n}$ all converge to zero and $\Gamma_j(\widehat\phi_{j,n})=O_p(1)$.
\end{enumerate}
Then $\widehat\tau_{j,n}^{\mathrm{AIPW}}\xrightarrow{p}\tau$.

\end{corollary}

Equation~\eqref{eq:end-to-end-sensitivity} separates the final error into an AIPW estimation term and a hidden-confounding sensitivity term. The selected stability constant $\Gamma_j(\widehat\phi_j)$, critic approximation error $a_{j,n_{\mathrm D}}$, optimization tolerances, and nuisance errors are not generally observed. The result becomes an operational numerical certificate after these quantities are bounded by theory, computation, external information, or a declared sensitivity analysis.

\subsection[Searching Across Proxy Splits]{Searching Across Proxy Splits}
\label{sec:random-orientation-search}
Exhaustive evaluation of all proxy splits in $\mathfrak S$, defined in Equation~\eqref{eq:all-held-out-orientations}, may be computationally expensive. The number of splits, $|\mathfrak S|=2^J-2$, grows exponentially in the number of blocks $J$, and each split requires its own representation and nuisance fits.

 Algorithm~\ref{alg:probe-search} therefore samples $m$ proxy splits instead of evaluating all of them, and then works in four stages. The sampling stage draws the splits $S$ (line~\ref{ln:sample}). The screening stage keeps the splits that pass the empirical balance screen (lines~\ref{ln:screen-start}--\ref{ln:screen-end}). The estimation stage computes one honest AIPW estimate per retained split (lines~\ref{ln:estimate-start}--\ref{ln:estimate-end}). The aggregation stage links estimates that lie within a common radius $\rho$ of one another and returns the median of the largest group (lines~\ref{ln:link}--\ref{ln:return}), which is the plurality rule of Theorem~\ref{thm:unknown-valid-block-identification} applied to estimated candidates.

Three questions decide whether the output is the true effect. (1) Does the random draw keep enough splits in $\widehat{\mathcal B}$, the set of splits that would pass the screening stage in lines~\ref{ln:screen-start}--\ref{ln:screen-end}, which is the finite-sample counterpart of the population balance-screen set $\mathcal B$ in Equation~\eqref{eq:population-balance-screen}? (2) Among them, does the true effect remain the most frequent value, which is the plurality condition of Theorem~\ref{thm:unknown-valid-block-identification}? (3) Once the candidate values are estimated rather than exact, does the largest linked group still consist of the splits that carry the true effect?

\begin{algorithm}[t]
\caption{PROBE with proxy-split search}
\label{alg:probe-search}
	\KwIn{a block partition $W=(W_1,\ldots,W_J)$; observations split into $\mathcal I_{\mathrm D}$, $\mathcal I_{\mathrm N}$, $\mathcal I_{\mathrm E}$; a budget $m$; a threshold $t\ge0$; an overlap constant $\eta$; a linking radius $\rho>0$.}
	\KwOut{an estimate $\widehat\tau$ of $\tau$, or a candidate set.}
	Draw $m$ proxy splits uniformly without replacement from $\mathfrak S$\label{ln:sample}\;
	\ForEach{sampled split $S$\label{ln:screen-start}}{
		learn $\widehat\phi_S$ on $\mathcal I_{\mathrm D}$ by minimizing $\widetilde D_{S,n_{\mathrm D}}^2$\;
		retain $S$ if $\widetilde D_{S,n_{\mathrm D}}^2(\widehat\phi_S)\le t$ and $\eta\le\widehat e_S\le1-\eta$\;
	}
	Let $\mathcal R_m$ be the set of retained splits\label{ln:screen-end}\;
	\ForEach{$S\in\mathcal R_m$\label{ln:estimate-start}}{
		compute the honest AIPW estimate $\widehat\theta_S$ of Definition~\ref{def:honest-aipw} on $\mathcal I_{\mathrm E}$\label{ln:estimate-end}\;
	}
	Link $S$ and $S'$ whenever $|\widehat\theta_S-\widehat\theta_{S'}|\le2\rho$, and form the connected components\label{ln:link}\;
	\Return the median of $\{\widehat\theta_S\}$ over the largest component; if several components share the largest size, return their medians as a candidate set\label{ln:return}\;
\end{algorithm}

Throughout this subsection, condition on the learning samples $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$ and on any internal randomness of the learning procedure, so that for every $S\in\mathfrak S$ the learned encoder $\widehat\phi_S$, the nuisance estimates, the outcome of the screen in lines~\ref{ln:screen-start}--\ref{ln:screen-end}, and hence the set $\widehat{\mathcal B}$, are fixed. For $S\in\widehat{\mathcal B}$, let

\begin{align}\label{eq:learned-candidate-value}
	\theta_S\triangleq\tau_{\widehat Z_S}
	=\mathbb E\bigl[\mathbb E(Y\mid A=1,\widehat Z_S)-\mathbb E(Y\mid A=0,\widehat Z_S)\bigr],
	\qquad
	\widehat Z_S\triangleq\widehat\phi_S(V_S),
\end{align}
be the population adjustment value of Equation~\eqref{eq:learned-adjustment-functional} for the learned representation. By Theorem~\ref{thm:approximate-proxy-bias-bound}, $\theta_S=\tau$ whenever $\widehat Z_S$ is exactly balanced and the assumptions of that theorem hold for $S$.

Let $T\triangleq|\mathfrak S|=2^J-2$ and $N\triangleq|\widehat{\mathcal B}|$. Fix the budget $m\in\{1,\ldots,T\}$ of Algorithm~\ref{alg:probe-search}. Line~\ref{ln:sample} draws $m$ proxy splits uniformly without replacement from $\mathfrak S$, with randomness independent of the data. Let $\mathcal R_m$ be the sampled proxy splits that also belong to $\widehat{\mathcal B}$, which is the retained set of line~\ref{ln:screen-end}, and let $R\triangleq|\mathcal R_m|$. Then
\begin{align}
	R\sim\operatorname{Hypergeometric}(T,N,m),
\label{eq:retained-orientation-count}
\end{align}
since exactly $N$ of the $T$ candidates belong to $\widehat{\mathcal B}$ and the $m$ draws are made without replacement.

For $c\in\mathbb R$, define the count and the number of non-target values
\begin{align}\label{eq:population-orientation-count}
	n(c)\triangleq\bigl|\{S\in\widehat{\mathcal B}:\theta_S=c\}\bigr|,
	\qquad L\triangleq\bigl|\{c'\in\mathbb R:c'\ne\tau,\ n(c')>0\}\bigr|,
\end{align}
 so that $n(c)$ is the number of screened splits whose candidate value equals $c$ and $L$ is the number of distinct candidate values other than $\tau$.
 Adopt the convention that a maximum over an empty collection equals zero and define
\begin{align}\label{eq:population-plurality-gap}
	p_\tau
	\triangleq
	\frac{n(\tau)}{N},
	\qquad
	p_\star
	\triangleq
	\max_{c\ne\tau:\,n(c)>0}\frac{n(c)}{N},
	\qquad
	\Delta\triangleq p_\tau-p_\star.
\end{align}
Here $p_\tau$ is the fraction of screened proxy splits whose candidate value is the true effect, $p_\star$ is the largest fraction that share one incorrect value, and $\Delta$ is the margin by which the true effect wins the plurality vote. The condition $\Delta>0$ is the plurality condition of Eq.~\eqref{eq:unique-target-plurality} with $\mathcal B$ replaced by $\widehat{\mathcal B}$.

Two quantities of the distribution in Equation~\eqref{eq:retained-orientation-count} enter the guarantee below: the probability that no sampled split passes the screen, and the expected value of $e^{-R\Delta^2/2}$,
\begin{align}\label{eq:sampling-failure-terms}
	P(R=0)=\binom{T-N}{m}\Big/\binom{T}{m},
	\qquad
	\mathbb E\bigl\{e^{-R\Delta^2/2}\bigr\}
	=\sum_{r=0}^{\min(m,N)}\binom{N}{r}\binom{T-N}{m-r}\,e^{-r\Delta^2/2}\Big/\binom{T}{m}.
\end{align}
Both are finite sums, and both decrease as the budget $m$ grows, because $R$ is stochastically increasing in $m$.

The following theorem bounds the error of the output of Algorithm~\ref{alg:probe-search}. Its probability is over the evaluation sample and the random draw of splits, conditional on the learning samples.

\begin{theorem}[Error of Algorithm~\ref{alg:probe-search}]
\label{thm:probe-search-error}
\normalfont

Condition on the learning samples. Let $\widehat{\mathcal B}$ be the screened set of lines~\ref{ln:screen-start}--\ref{ln:screen-end}, let $\theta_S$ be as in Equation~\eqref{eq:learned-candidate-value}, let $n(c)$, $L$, and $\Delta$ be as in Equations~\eqref{eq:population-orientation-count} and~\eqref{eq:population-plurality-gap}, and let $\widehat\theta_S$ be the honest AIPW estimate of Definition~\ref{def:honest-aipw} for split $S$. Let $m\in\{1,\ldots,T\}$ be the budget, and suppose the following three conditions hold.
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item \textit{Plurality.} $\Delta>0$, that is, $n(\tau)>n(c)$ for every $c\ne\tau$.
\item \textit{Uniform radius.} On an event of probability at least $1-\delta$ over the evaluation sample, $|\widehat\theta_S-\theta_S|\le\rho$ for every $S\in\widehat{\mathcal B}$.
\item \textit{Separation.} Any two distinct values in $\{\theta_S:S\in\widehat{\mathcal B}\}$ differ by more than $4\rho$.
\end{enumerate}
Then the output $\widehat\tau$ of Algorithm~\ref{alg:probe-search} run with linking radius $\rho$ satisfies
\begin{align}\label{eq:probe-search-error-probability}
	P\bigl(|\widehat\tau-\tau|\le\rho\bigr)
	\ \ge\
	1-\delta-P(R=0)-L\,\mathbb E\bigl\{e^{-R\Delta^2/2}\bigr\},
\end{align}
where $R\sim\operatorname{Hypergeometric}(T,N,m)$ as in Equation~\eqref{eq:retained-orientation-count}, the two sampling terms are given in Equation~\eqref{eq:sampling-failure-terms}, and an output that is a candidate set rather than a single value is counted as a failure.

\end{theorem}

Theorem~\ref{thm:probe-search-error} answers the three questions. The two sampling terms come from Lemma~\ref{lem:random-subsampling-plurality} in Appendix~\ref{app:proofs}: a uniform draw of $m$ splits keeps the true effect as the strictly most frequent retained value except with probability at most $P(R=0)+L\,\mathbb E\{e^{-R\Delta^2/2}\}$, which is the plurality argument of Theorem~\ref{thm:unknown-valid-block-identification} on a random subsample. Conditions (b) and (c) make the aggregation exact: estimates of equal population values lie within $2\rho$ of each other and estimates of distinct values lie more than $2\rho$ apart, so the components of line~\ref{ln:link} are exactly the groups of equal population values. Condition (b) follows from Theorem~\ref{thm:conditional-aipw-error} applied to each $S\in\widehat{\mathcal B}$ with $\delta_{\mathrm E}/N$ in place of $\delta_{\mathrm E}$, which suggests $\rho=\max_S\varepsilon_{\mathrm{AIPW},S}$ with its population quantities replaced by estimates or declared bounds. When the true-effect splits are only approximately balanced, Equation~\eqref{eq:uniform-approximate-proxy-bias-bound} confines their values to a band $[\tau-b,\tau+b]$, where $b$ is the largest value of that bound over these splits, and the theorem holds with $\rho+b$ in place of $\rho$ in the linking radius, in condition (c), and in the conclusion.

The three failure terms of Equation~\eqref{eq:probe-search-error-probability} are paid with different resources. The estimation term $\delta$ is paid with the linking radius: Theorem~\ref{thm:conditional-aipw-error} delivers condition (b) at level $\delta$ with $\rho=\max_{S\in\widehat{\mathcal B}}\varepsilon_{\mathrm{AIPW},S}$ evaluated at $\delta_{\mathrm E}=\delta/N$, so a smaller $\delta$ requires a larger radius or a larger evaluation sample, and condition (c) caps the radius at $4\rho<\min\{|c-c'|:c\ne c',\ n(c)>0,\ n(c')>0\}$. The two sampling terms are paid with the budget $m$: the fraction $p\triangleq N/T$ of splits that pass the screen, the margin $\Delta$, and the number $L$ of non-target values are fixed by the learned candidates, and $m$ is the only remaining choice. The screen threshold $t$ moves these constants in opposite directions, since a stricter screen removes poorly balanced splits, which raises $\Delta$ and lowers $L$ but also lowers $p$. The next corollary converts a target level $\epsilon$ for the two sampling terms into a budget.

\begin{corollary}[Budget for a target failure level]
\label{cor:budget-rule}
\normalfont

Let $p\triangleq N/T$ and fix $\epsilon\in(0,1)$. Under the conditions of Theorem~\ref{thm:probe-search-error},
\begin{align}\label{eq:sampling-failure-closed-form}
	P(R=0)+L\,\mathbb E\bigl\{e^{-R\Delta^2/2}\bigr\}
	\ \le\
	(L+1)\exp\bigl\{-m\,p\,(1-e^{-\Delta^2/2})\bigr\}.
\end{align}
Consequently, $P(|\widehat\tau-\tau|\le\rho)\ge1-\delta-\epsilon$ whenever
\begin{align}\label{eq:budget-rule}
	m\ \ge\ \frac{\log\{(L+1)/\epsilon\}}{p\,(1-e^{-\Delta^2/2})},
\end{align}
and the simpler budget $m\ge3\log\{(L+1)/\epsilon\}/(p\Delta^2)$ is sufficient. If $L=0$, the budget $m\ge\log(1/\epsilon)/p$ suffices.

\end{corollary}

The budget grows like $\log\{(L+1)/\epsilon\}/(p\Delta^2)$: halving the margin $\Delta$ quadruples it, while tightening $\epsilon$ from $0.05$ to $0.01$ multiplies it by less than two. Neither $p$ nor $\Delta$ is observed, but Algorithm~\ref{alg:probe-search} estimates both after the fact: $R/m$ is an unbiased estimate of $p$, and on the event of Theorem~\ref{thm:probe-search-error} the sizes of the two largest components divided by $R$ estimate $p_\tau$ and $p_\star$.

\begin{corollary}[Median under majority]
\label{cor:median-under-majority}
\normalfont

Suppose condition (b) of Theorem~\ref{thm:probe-search-error} holds and more than half of the retained splits carry the true effect, that is, $|\{S\in\mathcal R_m:\theta_S=\tau\}|>R/2$. Then, on the same event, the median of $\{\widehat\theta_S:S\in\mathcal R_m\}$ lies within $\rho$ of $\tau$; conditions (a) and (c) and the linking step are not needed.

\end{corollary}

When the splits with $\theta_S=\tau$ form a strict majority, Corollary~\ref{cor:median-under-majority} dispenses with the linking step. It is the finite-sample counterpart of Corollary~\ref{cor:target-majority-median}.
