\section{Representation Learning and AIPW Estimation}
\label{sec:estimation}

The identification results concern one fixed representation. We now select a representation from data and estimate the adjustment functional that it induces. The analysis uses one prespecified held-out block $j$ throughout. The AIPW construction below applies to discrete or continuous $Z_j$. Explicit concentration constants are subsequently given for a finite-strata specialization.

\subsection{Representation Learning}
\label{sec:representation-learning}

Partition the observed sample into three mutually disjoint subsets,
\begin{align}
\mathcal I_{\mathrm D},
\qquad
\mathcal I_{\mathrm N},
\qquad
\mathcal I_{\mathrm E},
\label{eq:honest-three-way-split}
\end{align}
with respective sizes $n_{\mathrm D}$, $n_{\mathrm N}$, and $n_{\mathrm E}$. The discrepancy-learning sample $\mathcal I_{\mathrm D}$ selects a representation. The nuisance-learning sample $\mathcal I_{\mathrm N}$ estimates the outcome regressions and propensity used by AIPW. The evaluation sample $\mathcal I_{\mathrm E}$ calculates the final estimate. Conditional on the preceding samples, the objects evaluated on the next sample are fixed.

Augment each fold with fresh independent randomization variables $\epsilon_{j,i}^{\mathrm D}$, $\epsilon_{j,i}^{\mathrm N}$, and $\epsilon_{j,i}^{\mathrm E}$, each distributed as $\operatorname{Unif}(0,1)$ and independent of the observed data and of all other randomization variables. For a candidate $\phi$, its representation on a unit in fold $s\in\{\mathrm D,\mathrm N,\mathrm E\}$ is $\phi(V_{j,i},\epsilon_{j,i}^{s})$. Thus the empirical criterion below is computed from the augmented discrepancy-learning sample $\{(O_i,\epsilon_{j,i}^{\mathrm D}):i\in\mathcal I_{\mathrm D}\}$.

For a general candidate class $\mathcal F_j\subseteq\Phi_j$, let $\widehat D_j(\phi)$ be an empirical criterion estimating $D_j(\phi)$. Suppose that, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
\sup_{\phi\in\mathcal F_j}
|\widehat D_j(\phi)-D_j(\phi)|
\le r_{j,n_{\mathrm D}}.
\label{eq:generic-uniform-deviation}
\end{align}
If $\widehat\phi_j$ is an $\varepsilon_{\mathrm{opt},j}$-approximate minimizer of $\widehat D_j$, the standard empirical-risk argument gives
\begin{align}
D_j(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+2r_{j,n_{\mathrm D}}
+\varepsilon_{\mathrm{opt},j}.
\label{eq:generic-representation-oracle}
\end{align}
This oracle inequality is agnostic to whether the learned representation is discrete or continuous. Its operational content depends on constructing an empirical conditional-dependence criterion for which equation~\eqref{eq:generic-uniform-deviation} can be proved.
For a fresh $\epsilon_j\sim\operatorname{Unif}(0,1)$ independent of the causal variables, define
\begin{align}
\widehat Z_j\triangleq\widehat\phi_j(V_j,\epsilon_j),
\qquad
\tau_{\widehat Z_j}
\triangleq
\mathbb E\!\left[
\mathbb E(Y\mid A=1,\widehat Z_j)
-
\mathbb E(Y\mid A=0,\widehat Z_j)
\right].
\label{eq:learned-adjustment-functional}
\end{align}
This definition is used when the two conditional means are absolutely integrable under the law of $\widehat Z_j$.
If every candidate obeys the assumptions of Theorem~\ref{thm:approximate-cross-view-bound} with common stability budget $\overline\Gamma_j$, then the same event also gives
\begin{align}
|\tau_{\widehat Z_j}-\tau|
\le
\overline\Gamma_j
\left\{
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+2r_{j,n_{\mathrm D}}
+\varepsilon_{\mathrm{opt},j}
\right\}.
\label{eq:generic-representation-oracle-causal}
\end{align}

\paragraph{Finite-strata specialization.}
We now give one explicit realization of the preceding argument. Restrict to deterministic maps, identifying each such map with $v\mapsto\phi(v,0)$ and writing it as $\phi(v)$. Assume every candidate takes values in a common finite set $\mathcal S_j\subset\mathbb R^{d_{Z_j}}$, and define $M_j\triangleq|\mathcal S_j|$.

Fix a finite candidate library
\begin{align}
\mathcal F_j\subseteq\Phi_j,
\qquad
N_j\triangleq|\mathcal F_j|<\infty.
\end{align}
For $\phi\in\mathcal F_j$, define
\begin{align}
Z_j^\phi\triangleq\phi(X,W_{-j})
\end{align}
and let $\mathcal R_j(\phi)$ be its set of positive-mass strata. For $z\in\mathcal R_j(\phi)$, write $p_{j,z}(\phi)\triangleq P(Z_j^\phi=z)$. Assume there are constants $p_{\min}>0$ and $\eta\in(0,1/2]$ such that every candidate and every $z\in\mathcal R_j(\phi)$ satisfy
\begin{align}
P(Z_j^\phi=z)&\ge p_{\min},
\label{eq:library-minimum-mass}\\
\eta
\le P(A=1\mid Z_j^\phi=z)&\le1-\eta.
\label{eq:library-overlap}
\end{align}
Assume also that the bounded kernel satisfies
\begin{align}
\sup_v\sqrt{\kappa_j(v,v)}\le\rho_j<\infty.
\label{eq:kernel-bound}
\end{align}
The two primitive cross-view clauses of Assumption~\ref{ass:joint-latent-exchangeability} do not depend on the selected candidate. Representation overlap and outcome-relevant stability must hold for every candidate-specific representation. Their stability budgets are required to obey the common bound in equation~\eqref{eq:uniform-candidate-stability}.

\begin{definition}[Empirical cross-view discrepancy]
\label{def:empirical-cross-view-discrepancy}
For observation $i\in\mathcal I_{\mathrm D}$ and candidate $\phi\in\mathcal F_j$, write
\begin{align}
Z_{j,i}^{\phi}\triangleq\phi(X_i,W_{-j,i}).
\end{align}
For treatment arm $a$ and stratum $z$, define
\begin{align}
N_{a,z}^{\phi}
&\triangleq
\sum_{i\in\mathcal I_{\mathrm D}}
\mathbf 1\{A_i=a,Z_{j,i}^{\phi}=z\},\\
N_z^\phi
&\triangleq
N_{0,z}^{\phi}+N_{1,z}^{\phi},\\
\widehat p_{j,z}(\phi)
&\triangleq
\frac{N_z^\phi}{n_{\mathrm D}}.
\end{align}
When $N_{a,z}^{\phi}>0$, define the empirical kernel mean
\begin{align}
\widehat m_{a,j,z}(\phi)
\triangleq
\frac{1}{N_{a,z}^{\phi}}
\sum_{i\in\mathcal I_{\mathrm D}}
\mathbf 1\{A_i=a,Z_{j,i}^{\phi}=z\}
\kappa_j\{(X_i,W_{j,i}),\cdot\}.
\end{align}
When both treatment-arm cells are nonempty, define
\begin{align}
\widehat\Delta_{j,z}(\phi)
\triangleq
\|\widehat m_{1,j,z}(\phi)-\widehat m_{0,j,z}(\phi)\|_{\mathcal H_j}
\end{align}
and
\begin{align}
\widehat D_j(\phi)
\triangleq
\sum_{z:N_z^\phi>0}
\widehat p_{j,z}(\phi)\widehat\Delta_{j,z}(\phi).
\label{eq:empirical-discrepancy}
\end{align}
If an occupied empirical stratum contains only one treatment arm, set $\widehat D_j(\phi)=+\infty$.
\end{definition}

\begin{proposition}[Uniform deviation for the finite-strata specialization]
\label{prop:uniform-discrepancy-deviation}
Let $\delta_{\mathrm D}\in(0,1)$ and suppose
\begin{align}
n_{\mathrm D}
\ge
\frac{8}{p_{\min}\eta}
\log\left(\frac{8M_jN_j}{\delta_{\mathrm D}}\right).
\label{eq:learning-sample-threshold}
\end{align}
Define
\begin{align}
e_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\triangleq
\rho_j
\left[
2+
\sqrt{2\log\left(\frac{4M_jN_j}{\delta_{\mathrm D}}\right)}
\right]
\sqrt{\frac{2}{n_{\mathrm D}p_{\min}\eta}}
\end{align}
and
\begin{align}
r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\triangleq{}&
2e_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\nonumber\\
&+
2\rho_j
\sqrt{
\frac{
2\left[
M_j\log2+
\log\left(\frac{4N_j}{\delta_{\mathrm D}}\right)
\right]
}{n_{\mathrm D}}
}.
\label{eq:discrepancy-radius}
\end{align}
Under equations~\eqref{eq:library-minimum-mass} through \eqref{eq:kernel-bound}, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
\boxed{
\sup_{\phi\in\mathcal F_j}
|\widehat D_j(\phi)-D_j(\phi)|
\le
r_{j,n_{\mathrm D}}(\delta_{\mathrm D}).
}
\label{eq:uniform-discrepancy-deviation}
\end{align}
\end{proposition}

\begin{definition}[Approximate empirical minimizer]
\label{def:approximate-empirical-minimizer}
Let $\varepsilon_{\mathrm{opt},j}\ge0$. The representation learner returns $\widehat\phi_j\in\mathcal F_j$ satisfying
\begin{align}
\widehat D_j(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_j}\widehat D_j(\phi)
+
\varepsilon_{\mathrm{opt},j}.
\label{eq:approximate-erm}
\end{align}
In this deterministic specialization, the learned representation is
\begin{align}
\widehat Z_j\triangleq\widehat\phi_j(V_j).
\end{align}
The tolerance records optimization error and is zero for an exact empirical minimizer.
\end{definition}

\begin{theorem}[Finite-strata representation-learning oracle inequality]
\label{thm:representation-oracle}
Suppose the conditions of Proposition~\ref{prop:uniform-discrepancy-deviation} hold. Suppose Assumptions~\ref{ass:consistency} through \ref{ass:held-out-separation} hold for block $j$, and suppose representation overlap and uniform outcome-relevant stability hold for every candidate with the common budget $\overline\Gamma_j$. Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
\boxed{
D_j(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+
\varepsilon_{\mathrm{opt},j}.
}
\label{eq:representation-oracle-discrepancy}
\end{align}
On the same event,
\begin{align}
\boxed{
|\tau_{\widehat Z_j}-\tau|
\le
\overline\Gamma_j
\left[
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+
\varepsilon_{\mathrm{opt},j}
\right].
}
\label{eq:representation-oracle-causal}
\end{align}
\end{theorem}

\begin{remark}[Practical learner outside the finite-library theorem]
\label{rem:practical-learning-gap}
The theorem covers hard, finite-valued representations in a finite prespecified library. A neural learner using random features, soft assignments, and Adam requires additional bounds for neural-class complexity, feature approximation, soft-to-hard transfer, and optimization error. These guarantees are not supplied by Theorem~\ref{thm:representation-oracle}.
\end{remark}

\subsection{AIPW Estimation}
\label{sec:aipw-estimation}

Conditional on $\mathcal I_{\mathrm D}$, the map $\widehat\phi_j$ is fixed. A fresh independent $\epsilon_j\sim\operatorname{Unif}(0,1)$ produces $\widehat Z_j=\widehat\phi_j(V_j,\epsilon_j)$. Define the conditional outcome means and representation propensity
\begin{align}
m_{a,j}(z)
&\triangleq
\mathbb E(Y\mid A=a,\widehat Z_j=z),\\
e_j(z)
&\triangleq
P(A=1\mid\widehat Z_j=z).
\label{eq:aipw-nuisances}
\end{align}
For the observed adjustment functional $\tau_{\widehat Z_j}$, the following is the AIPW score:
\begin{align}
\psi_j(O,\epsilon_j)
\triangleq{}&
m_{1,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)
\nonumber\\
&+
\frac{A\{Y-m_{1,j}(\widehat Z_j)\}}{e_j(\widehat Z_j)}
-
\frac{(1-A)\{Y-m_{0,j}(\widehat Z_j)\}}{1-e_j(\widehat Z_j)}
-
\tau_{\widehat Z_j}.
\label{eq:fixed-representation-eif}
\end{align}
When this score is square-integrable, it is the efficient influence function in the nonparametric model for $(\widehat Z_j,A,Y)$. Bounded outcomes together with uniform representation overlap provide one sufficient condition for square integrability. Use the augmented nuisance-learning sample, with $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i},\epsilon_{j,i}^{\mathrm N})$, to estimate the nuisance functions. Denote the estimates by $\widehat m_{0,j}$, $\widehat m_{1,j}$, and $\widehat e_j$.

\begin{definition}[Honest AIPW estimator]
\label{def:honest-aipw}
For $i\in\mathcal I_{\mathrm E}$, define $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i},\epsilon_{j,i}^{\mathrm E})$ and
\begin{align}
S_j(O_i,\epsilon_{j,i}^{\mathrm E})
\triangleq{}&
\widehat m_{1,j}(\widehat Z_{j,i})
-
\widehat m_{0,j}(\widehat Z_{j,i})
\nonumber\\
&+
\frac{A_i\{Y_i-\widehat m_{1,j}(\widehat Z_{j,i})\}}
{\widehat e_j(\widehat Z_{j,i})}
\nonumber\\
&-
\frac{(1-A_i)\{Y_i-\widehat m_{0,j}(\widehat Z_{j,i})\}}
{1-\widehat e_j(\widehat Z_{j,i})}.
\label{eq:aipw-score}
\end{align}
The honest AIPW estimator is
\begin{align}
\boxed{
\widehat\tau_j^{\mathrm{AIPW}}
\triangleq
\frac{1}{n_{\mathrm E}}
\sum_{i\in\mathcal I_{\mathrm E}}S_j(O_i,\epsilon_{j,i}^{\mathrm E}).
}
\label{eq:honest-aipw}
\end{align}
\end{definition}

\begin{proposition}[Exact signed nuisance remainder]
\label{prop:aipw-remainder}
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$,
\begin{align}
&\mathbb E\{S_j(O,\epsilon_j)\}-\tau_{\widehat Z_j}
\nonumber\\
&=
\mathbb E
\Bigg[
\{\widehat e_j(\widehat Z_j)-e_j(\widehat Z_j)\}
\Bigg\{
\frac{\widehat m_{1,j}(\widehat Z_j)-m_{1,j}(\widehat Z_j)}
{\widehat e_j(\widehat Z_j)}
\nonumber\\
&\hspace{51mm}
+
\frac{\widehat m_{0,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)}
{1-\widehat e_j(\widehat Z_j)}
\Bigg\}
\Bigg].
\label{eq:aipw-remainder}
\end{align}
\end{proposition}

The remainder is zero if the propensity estimator is correct. It is also zero if both conditional outcome estimators are correct.

\begin{theorem}[Conditional finite-sample AIPW error]
\label{thm:conditional-aipw-error}
Suppose $|Y|\le B$ almost surely. Clip the nuisance estimates so that
\begin{align}
-B\le\widehat m_{a,j}(z)\le B,
\qquad
\eta\le\widehat e_j(z)\le1-\eta.
\label{eq:nuisance-clipping}
\end{align}
For a function $g$ of $\widehat Z_j$, define
\begin{align}
\|g\|_{2,\widehat Z_j}
\triangleq
\left[
\mathbb E\{g(\widehat Z_j)^2\mid
\mathcal I_{\mathrm D},\mathcal I_{\mathrm N}\}
\right]^{1/2}.
\end{align}
After the two learning samples are observed, define
\begin{align}
q_{e,j}&\triangleq\|\widehat e_j-e_j\|_{2,\widehat Z_j},\\
q_{a,j}&\triangleq\|\widehat m_{a,j}-m_{a,j}\|_{2,\widehat Z_j},
\qquad a\in\{0,1\}.
\label{eq:realized-nuisance-errors}
\end{align}
For $\delta_{\mathrm E}\in(0,1)$, define
\begin{align}
\varepsilon_{\mathrm{AIPW},j}
\triangleq{}&
2B\left(1+\frac{1}{\eta}\right)
\sqrt{\frac{2\log(2/\delta_{\mathrm E})}{n_{\mathrm E}}}
\nonumber\\
&+
\frac{q_{e,j}(q_{0,j}+q_{1,j})}{\eta}.
\label{eq:aipw-error-radius}
\end{align}
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$, with probability at least $1-\delta_{\mathrm E}$ over $\mathcal I_{\mathrm E}$,
\begin{align}
\boxed{
|\widehat\tau_j^{\mathrm{AIPW}}-\tau_{\widehat Z_j}|
\le
\varepsilon_{\mathrm{AIPW},j}.
}
\label{eq:conditional-aipw-error}
\end{align}
\end{theorem}

\subsection{End-to-End Finite-Sample Bound}
\label{sec:end-to-end-bound}

\begin{theorem}[Finite-strata end-to-end sensitivity bound]
\label{thm:end-to-end-sensitivity}
Suppose the conditions of Proposition~\ref{prop:uniform-discrepancy-deviation}, Theorem~\ref{thm:representation-oracle}, and Theorem~\ref{thm:conditional-aipw-error} hold. Then, jointly over the three independent samples, with probability at least $1-\delta_{\mathrm D}-\delta_{\mathrm E}$,
\begin{align}
\boxed{
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
\le
\varepsilon_{\mathrm{AIPW},j}
+
\overline\Gamma_j
\left[
\widehat D_j(\widehat\phi_j)
+
r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\right].
}
\label{eq:end-to-end-empirical}
\end{align}
It also satisfies the oracle form
\begin{align}
\boxed{
\begin{aligned}
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
&\le
\varepsilon_{\mathrm{AIPW},j}
\\
&\quad+
\overline\Gamma_j
\left[
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+
\varepsilon_{\mathrm{opt},j}
\right].
\end{aligned}
}
\label{eq:end-to-end-oracle}
\end{align}
\end{theorem}

\begin{corollary}[Consistency]
\label{cor:learned-aipw-consistency}
Consider a sequence of candidate libraries, sample splits, nuisance estimators, optimization tolerances, and stability budgets satisfying Theorem~\ref{thm:end-to-end-sensitivity}. If
\begin{align}
\delta_{\mathrm D,n}+\delta_{\mathrm E,n}&\longrightarrow0,\\
\frac{\log(2/\delta_{\mathrm E,n})}{n_{\mathrm E}}&\longrightarrow0,\\
q_{e,j,n}(q_{0,j,n}+q_{1,j,n})&\xrightarrow{p}0,
\end{align}
and
\begin{align}
\overline\Gamma_{j,n}
\left[
\inf_{\phi\in\mathcal F_{j,n}}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D,n})
+
\varepsilon_{\mathrm{opt},j,n}
\right]
\xrightarrow{p}0,
\label{eq:consistency-representation-condition}
\end{align}
then
\begin{align}
\widehat\tau_{j,n}^{\mathrm{AIPW}}
\xrightarrow{p}
\tau.
\end{align}
\end{corollary}

\begin{remark}[Interpretation and observability]
\label{rem:end-to-end-observability}
Theorem~\ref{thm:end-to-end-sensitivity} separates the final error into an AIPW estimation term and a hidden-confounding sensitivity term. The stability budget $\overline\Gamma_j$ and the nuisance errors $q_{e,j}$, $q_{0,j}$, and $q_{1,j}$ are not generally observed. The result becomes an operational numerical certificate after these quantities are bounded by additional theory, external information, or a declared sensitivity analysis.
\end{remark}

\begin{remark}[Scope]
\label{rem:end-to-end-scope}
The theorem uses one prespecified held-out block, hard finite strata, a finite candidate library, and an honest three-way split. It does not establish a neural-class rate, a soft-to-hard approximation bound, an Adam optimization guarantee, or a cross-fitting theorem. Experimental aggregation across several held-out orientations is also outside this fixed-$j$ result.
\end{remark}
