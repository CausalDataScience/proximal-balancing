\section{Representation Learning and AIPW Estimation}
\label{sec:estimation}

The identification results concern one fixed representation. We now select a representation from data and estimate the adjustment functional that it induces. The representation-learning and AIPW analysis below uses one prespecified held-out block $j$ throughout.

\subsection{Representation Learning}
\label{sec:representation-learning}

Recall that one observed record is $O_i=(X_i,W_i,A_i,Y_i)$. Partition the observed sample into three mutually disjoint subsets,
\begin{align}
	\mathcal I_{\mathrm D},
	\qquad
	\mathcal I_{\mathrm N},
	\qquad
	\mathcal I_{\mathrm E},
	\label{eq:honest-three-way-split}
\end{align}
with respective sizes $n_{\mathrm D}$, $n_{\mathrm N}$, and $n_{\mathrm E}$. The discrepancy-learning sample $\mathcal I_{\mathrm D}$ selects a representation. The nuisance-learning sample $\mathcal I_{\mathrm N}$ estimates the outcome regressions and propensity used by AIPW. The evaluation sample $\mathcal I_{\mathrm E}$ calculates the final estimate. Conditional on the preceding samples, the objects evaluated on the next sample are fixed.

 Let $\mathcal F_{j,n_{\mathrm D}}\subseteq\Phi_j$ be a prespecified, possibly infinite class of deterministic encoders. For $\phi\in\mathcal F_{j,n_{\mathrm D}}$, write $Z_j^\phi=\phi(V_j)$ and retain $T_j=(X,W_j)$ from Section~\ref{sec:approximate-proxy-bound}.

\begin{assumption}[Common representation overlap]
\label{ass:library-overlap}
\normalfont
There is a constant $\eta\in(0,1/2]$ such that, for every $\phi\in\mathcal F_{j,n_{\mathrm D}}$,

\begin{align}
\eta\le e_{j,\phi}(Z_j^\phi)\le1-\eta\quad\text{almost surely}.
\label{eq:library-overlap}
\end{align}

\end{assumption}
For any prespecified nonempty proper subset $S\subset\{1,\ldots,J\}$, the results below also apply with $(V_j,W_j,Z_j^\phi)$ replaced by $(V_S,W_S,\phi(V_S))$ and with the same assumptions imposed on this grouped split.

The Brier-risk definition below estimates residual discrepancy by comparing squared-loss prediction of $A$ from $Z_j^\phi$ alone with prediction from $(T_j,Z_j^\phi)$ \citep{brier1950verification}.

\begin{definition}[Brier-risk residual discrepancy]
\label{def:empirical-held-out-proxy-discrepancy}
\normalfont
 Let $\mathcal H_{0,n_{\mathrm D}}$ and $\mathcal H_{1,n_{\mathrm D}}$ be prespecified classes of measurable $[0,1]$-valued predictors with inputs $Z_j^\phi$ and $(T_j,Z_j^\phi)$, respectively, fixed independently of $\mathcal I_{\mathrm D}$. Assume $\mathcal H_{1,n_{\mathrm D}}$ contains each member of $\mathcal H_{0,n_{\mathrm D}}$ lifted to ignore $T_j$. For $b\in\{0,1\}$, define
\begin{align}
	R_{0,\phi}(h)&\triangleq\mathbb E[\{A-h(Z_j^\phi)\}^2],
	&
	\widehat R_{0,\phi}(h)&\triangleq\frac1{n_{\mathrm D}}\sum_{i\in\mathcal I_{\mathrm D}}\{A_i-h(Z_{j,i}^\phi)\}^2,\\
	R_{1,\phi}(h)&\triangleq\mathbb E[\{A-h(T_j,Z_j^\phi)\}^2],
	&
	\widehat R_{1,\phi}(h)&\triangleq\frac1{n_{\mathrm D}}\sum_{i\in\mathcal I_{\mathrm D}}\{A_i-h(T_{j,i},Z_{j,i}^\phi)\}^2,
	\label{eq:brier-risks}
\end{align}
where $Z_{j,i}^\phi=\phi(V_{j,i})$. For each $\phi$, let $\widehat h_{b,\phi}\in\mathcal H_{b,n_{\mathrm D}}$ be an approximate empirical minimizer satisfying, for some $\varepsilon_{b,\phi}\ge0$,
\begin{align}
	\widehat R_{b,\phi}(\widehat h_{b,\phi})
	\le
	\inf_{h\in\mathcal H_{b,n_{\mathrm D}}}\widehat R_{b,\phi}(h)
	+
	\varepsilon_{b,\phi},
	\qquad b\in\{0,1\}.
	\label{eq:approximate-critic-fits}
\end{align}
Define $\varepsilon_{\mathrm{crit},j}\triangleq\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}}}(\varepsilon_{0,\phi}+\varepsilon_{1,\phi})$ and
\begin{align}
	\widetilde D_{j,n_{\mathrm D}}^2(\phi)
	\triangleq
	\max\left\{(
	\widehat R_{0,\phi}(\widehat h_{0,\phi})
	-
	\widehat R_{1,\phi}(\widehat h_{1,\phi})), 0
	\right\},
	\label{eq:empirical-discrepancy}
\end{align}
\end{definition}

Each encoder $\phi$ and critic $h$ give a squared-loss function of one observation: $\{A-h(\phi(V_j))\}^2$ or $\{A-h(T_j,\phi(V_j))\}^2$. Varying both $\phi$ and $h$ over their prespecified classes gives the loss classes $\mathcal L_{0,j,n_{\mathrm D}}$ and $\mathcal L_{1,j,n_{\mathrm D}}$, respectively. Let $\mathfrak R_{n_{\mathrm D}}(\mathcal L)$ denote the expected Rademacher complexity of a loss class $\mathcal L$. For $\delta_{\mathrm D}\in(0,1)$, define

\begin{align}
	r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
	\triangleq
	2\{\mathfrak R_{n_{\mathrm D}}(\mathcal L_{0,j,n_{\mathrm D}})+\mathfrak R_{n_{\mathrm D}}(\mathcal L_{1,j,n_{\mathrm D}})\}
	+
	2\sqrt{\frac{\log(4/\delta_{\mathrm D})}{2n_{\mathrm D}}}.
	\label{eq:discrepancy-radius}
\end{align}

This radius controls sampling fluctuation uniformly over the encoder--critic choices. The complexity terms account for the flexibility of the two loss classes, and the last term sets the failure-probability level $\delta_{\mathrm D}$.

\begin{proposition}[Uniform residual-discrepancy deviation]
\label{prop:uniform-discrepancy-deviation}
\normalfont
For each $\phi\in\mathcal F_{j,n_{\mathrm D}}$, define
\begin{align}
	a_{0,n_{\mathrm D}}(\phi)
	&\triangleq
	\inf_{h\in\mathcal H_{0,n_{\mathrm D}}}R_{0,\phi}(h)
	-
	\mathbb E[\{A-e_{j,\phi}(Z_j^\phi)\}^2],\\
	a_{1,n_{\mathrm D}}(\phi)
	&\triangleq
	\inf_{h\in\mathcal H_{1,n_{\mathrm D}}}R_{1,\phi}(h)
	-
	\mathbb E[\{A-q_{j,\phi}(T_j,Z_j^\phi)\}^2],\\
	a_{j,n_{\mathrm D}}
	&\triangleq
	\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}}}
	\{a_{0,n_{\mathrm D}}(\phi)+a_{1,n_{\mathrm D}}(\phi)\}.
	\label{eq:critic-approximation-error}
\end{align}

Under the measurability conditions stated in   Assumption~\ref{ass:learning-measurability}, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
	\sup_{\phi\in\mathcal F_{j,n_{\mathrm D}}}
	|\widetilde D_{j,n_{\mathrm D}}^2(\phi)-D_{\mathrm{res},j}^2(\phi)|
	\le
	r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}.
	\label{eq:uniform-discrepancy-deviation}
\end{align}
\end{proposition}
At any fixed confidence level $1-\delta_{\mathrm D}$, this error bound is $O(n_{\mathrm D}^{-1/2})$ if both Rademacher complexities, $a_{j,n_{\mathrm D}}$, and $\varepsilon_{\mathrm{crit},j}$ are each $O(n_{\mathrm D}^{-1/2})$.

\begin{definition}[Approximate empirical minimizer]
\label{def:approximate-empirical-minimizer}
\normalfont
 For a prescribed tolerance $\varepsilon_{\mathrm{enc},j}\ge0$, an approximate empirical minimizer is an encoder $\widehat\phi_j\in\mathcal F_{j,n_{\mathrm D}}$ selected measurably from the training observations and algorithmic seed such that
\begin{align}
\widetilde D_{j,n_{\mathrm D}}^2(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}\widetilde D_{j,n_{\mathrm D}}^2(\phi)
+
\varepsilon_{\mathrm{enc},j}.
\label{eq:approximate-erm}
\end{align}
An exact empirical minimizer has $\varepsilon_{\mathrm{enc},j}=0$.

\end{definition}

\begin{proposition}[Residual-discrepancy oracle inequality]
\label{thm:representation-oracle}
\normalfont
 Let $\delta_{\mathrm D}\in(0,1)$. Suppose the prespecified encoder and $[0,1]$-valued critic classes,  approximate critic fits in Proposition~\ref{prop:uniform-discrepancy-deviation} and Assumption~\ref{ass:learning-measurability} hold, and $\widehat\phi_j$ satisfies Equation~\eqref{eq:approximate-erm}. Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
 D_{\mathrm{res},j}^2(\widehat\phi_j)
 &\le
 \left[\widetilde D_{j,n_{\mathrm D}}^2(\widehat\phi_j)\right]_+
 +r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j},
 \label{eq:representation-empirical-discrepancy}\\
 D_{\mathrm{res},j}^2(\widehat\phi_j)
 &\le
 \inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)
 +2\{r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}\}
 +\varepsilon_{\mathrm{enc},j}.
\label{eq:representation-oracle-discrepancy}
\end{align}
\end{proposition}

Condition on the training observations $\{O_i:i\in\mathcal I_{\mathrm D}\}$ and any algorithmic seed used in training, which fixes $\widehat\phi_j$. Here and below, population expectations involving this learned encoder use a fresh observation $O=(X,W,A,Y)$ independent of the training observations and seed, with this conditioning understood.
The learned representation and its population adjustment functional are

\begin{align}
Z_j^{\widehat\phi_j}
&\triangleq\widehat\phi_j(V_j),\qquad\widehat Z_j\triangleq Z_j^{\widehat\phi_j},\nonumber\\
\tau_{\widehat Z_j}
&\triangleq
\mathbb E\!\left[
\mathbb E(Y\mid A=1,\widehat Z_j)
-
\mathbb E(Y\mid A=0,\widehat Z_j)
\right].
\label{eq:learned-adjustment-functional}
\end{align}

Thus $\tau_{\widehat Z_j}=\tau_\phi\big|_{\phi=\widehat\phi_j}$, where $\tau_\phi$ is the population adjustment functional defined in Proposition~\ref{prop:population-sensitivity-region}.
The adjustment functional is used whenever its two conditional means are absolutely integrable under the law of $\widehat Z_j$.

\begin{theorem}[Causal error of the learned representation]
\label{cor:representation-causal}
\normalfont

Let $\delta_{\mathrm D}\in(0,1)$. Suppose the prespecified encoder and $[0,1]$-valued critic classes,  approximate critic fits in Proposition~\ref{prop:uniform-discrepancy-deviation} and Assumption~\ref{ass:learning-measurability} hold, and $\widehat\phi_j$ satisfies Equation~\eqref{eq:approximate-erm}. Suppose Assumptions~\ref{ass:consistency}, \ref{ass:joint-latent-exchangeability}, and \ref{ass:uniform-stability} hold for every $Z_j^\phi$ with $\phi\in\mathcal F_{j,n_{\mathrm D}}$, and there is a constant $\eta\in(0,1/2]$ such that $\eta\le e_{j,\phi}(Z_j^\phi)\le1-\eta$ almost surely for every such $\phi$. Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
	|\tau_{\widehat Z_j}-\tau|
	\le
	\frac{\Gamma_j(\widehat\phi_j)}{\eta(1-\eta)}
	\left[
	\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)
	+2\{r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}\}
	+\varepsilon_{\mathrm{enc},j}
	\right]^{1/2}.
	\label{eq:representation-oracle-causal}
\end{align}
\end{theorem}
At fixed confidence and fixed overlap constant $\eta$, the right-hand side is $O_p(n_{\mathrm D}^{-1/4})$ if $\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)=0$, each of $r_{j,n_{\mathrm D}}(\delta_{\mathrm D})$, $a_{j,n_{\mathrm D}}$, $\varepsilon_{\mathrm{crit},j}$, and $\varepsilon_{\mathrm{enc},j}$ is $O(n_{\mathrm D}^{-1/2})$, and $\Gamma_j(\widehat\phi_j)=O_p(1)$. This is a conditional upper-bound rate; it is not a universal or minimax rate.

\subsection{AIPW Estimation}
\label{sec:aipw-estimation}
 Conditional on the observations $\{O_i:i\in\mathcal I_{\mathrm D}\}$ and any algorithmic seed used in training, the map $\widehat\phi_j$ is fixed and $\widehat Z_j=\widehat\phi_j(V_j)$. Below, conditioning on learning samples means conditioning on all observations indexed by those samples and the seeds used to fit the encoder and nuisance functions; population expectations use a fresh independent observation. Define the conditional outcome means and representation propensity
\begin{align}
m_{a,j}(z)
&\triangleq
\mathbb E(Y\mid A=a,\widehat Z_j=z),\\
e_j(z)
&\triangleq
P(A=1\mid\widehat Z_j=z).
\label{eq:aipw-nuisances}
\end{align}
For the observed adjustment functional $\tau_{\widehat Z_j}$, the following is the AIPW score:
\begin{align}
\psi_j(O)
\triangleq{}&
m_{1,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)
\nonumber\\
&+
\frac{A\{Y-m_{1,j}(\widehat Z_j)\}}{e_j(\widehat Z_j)}
-
\frac{(1-A)\{Y-m_{0,j}(\widehat Z_j)\}}{1-e_j(\widehat Z_j)}
-
\tau_{\widehat Z_j}.
\label{eq:fixed-representation-eif}
\end{align}
When this score is square-integrable, it is the efficient influence function in the nonparametric model for $(\widehat Z_j,A,Y)$. Bounded outcomes together with uniform representation overlap provide one sufficient condition for square integrability. Use the nuisance-learning sample, with $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i})$, to estimate the nuisance functions. Denote the estimates by $\widehat m_{0,j}$, $\widehat m_{1,j}$, and $\widehat e_j$.

\begin{definition}[Honest AIPW estimator]
\label{def:honest-aipw}
For $i\in\mathcal I_{\mathrm E}$, define $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i})$ and
\begin{align}
S_j(O_i)
\triangleq{}&
\widehat m_{1,j}(\widehat Z_{j,i})
-
\widehat m_{0,j}(\widehat Z_{j,i})
\nonumber\\
&+
\frac{A_i\{Y_i-\widehat m_{1,j}(\widehat Z_{j,i})\}}
{\widehat e_j(\widehat Z_{j,i})}
\nonumber\\
&-
\frac{(1-A_i)\{Y_i-\widehat m_{0,j}(\widehat Z_{j,i})\}}
{1-\widehat e_j(\widehat Z_{j,i})}.
\label{eq:aipw-score}
\end{align}
The honest AIPW estimator is
\begin{align}
\boxed{
\widehat\tau_j^{\mathrm{AIPW}}
\triangleq
\frac{1}{n_{\mathrm E}}
\sum_{i\in\mathcal I_{\mathrm E}}S_j(O_i).
}
\label{eq:honest-aipw}
\end{align}
\end{definition}

\begin{proposition}[Exact signed nuisance remainder]
\label{prop:aipw-remainder}
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$,
\begin{align}
&\mathbb E\{S_j(O)\}-\tau_{\widehat Z_j}
\nonumber\\
&=
\mathbb E
\Bigg[
\{\widehat e_j(\widehat Z_j)-e_j(\widehat Z_j)\}
\Bigg\{
\frac{\widehat m_{1,j}(\widehat Z_j)-m_{1,j}(\widehat Z_j)}
{\widehat e_j(\widehat Z_j)}
\nonumber\\
&\hspace{51mm}
+
\frac{\widehat m_{0,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)}
{1-\widehat e_j(\widehat Z_j)}
\Bigg\}
\Bigg].
\label{eq:aipw-remainder}
\end{align}
\end{proposition}

The remainder is zero if the propensity estimator is correct. It is also zero if both conditional outcome estimators are correct.

\begin{theorem}[Conditional finite-sample AIPW error]
\label{thm:conditional-aipw-error}
Suppose $|Y|\le B$ almost surely. Clip the nuisance estimates so that
\begin{align}
-B\le\widehat m_{a,j}(z)\le B,
\qquad
\eta\le\widehat e_j(z)\le1-\eta.
\label{eq:nuisance-clipping}
\end{align}
For a function $g$ of $\widehat Z_j$, define
\begin{align}
\|g\|_{2,\widehat Z_j}
\triangleq
\left[
\mathbb E\{g(\widehat Z_j)^2\mid
\mathcal I_{\mathrm D},\mathcal I_{\mathrm N}\}
\right]^{1/2}.
\end{align}
After the two learning samples are observed, define
\begin{align}
q_{e,j}&\triangleq\|\widehat e_j-e_j\|_{2,\widehat Z_j},\\
q_{a,j}&\triangleq\|\widehat m_{a,j}-m_{a,j}\|_{2,\widehat Z_j},
\qquad a\in\{0,1\}.
\label{eq:realized-nuisance-errors}
\end{align}
For $\delta_{\mathrm E}\in(0,1)$, define
\begin{align}
\varepsilon_{\mathrm{AIPW},j}
\triangleq{}&
2B\left(1+\frac{1}{\eta}\right)
\sqrt{\frac{2\log(2/\delta_{\mathrm E})}{n_{\mathrm E}}}
\nonumber\\
&+
\frac{q_{e,j}(q_{0,j}+q_{1,j})}{\eta}.
\label{eq:aipw-error-radius}
\end{align}
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$, with probability at least $1-\delta_{\mathrm E}$ over $\mathcal I_{\mathrm E}$,
\begin{align}
\boxed{
|\widehat\tau_j^{\mathrm{AIPW}}-\tau_{\widehat Z_j}|
\le
\varepsilon_{\mathrm{AIPW},j}.
}
\label{eq:conditional-aipw-error}
\end{align}
\end{theorem}

\subsection{End-to-End Finite-Sample Bound}
\label{sec:end-to-end-bound}

\begin{theorem}[End-to-end sensitivity bound]
\label{thm:end-to-end-sensitivity}
 Suppose the conditions of Proposition~\ref{prop:uniform-discrepancy-deviation},  Proposition~\ref{thm:representation-oracle},  Theorem~\ref{cor:representation-causal}, and Theorem~\ref{thm:conditional-aipw-error} hold. Then, jointly over the three independent samples, with probability at least $1-\delta_{\mathrm D}-\delta_{\mathrm E}$,
\begin{align}
\boxed{
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
\le
\varepsilon_{\mathrm{AIPW},j}
+
\frac{\Gamma_j(\widehat\phi_j)}{\eta(1-\eta)}
\left[
\widetilde D_{j,n_{\mathrm D}}^2(\widehat\phi_j)
+r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}
\right]^{1/2}.
}
\label{eq:end-to-end-empirical}
\end{align}
It also satisfies the oracle form
\begin{align}
\boxed{
\begin{aligned}
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
&\le
\varepsilon_{\mathrm{AIPW},j}
\\
&\quad+
\frac{\Gamma_j(\widehat\phi_j)}{\eta(1-\eta)}
\left[
\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)
+2\{r_{j,n_{\mathrm D}}(\delta_{\mathrm D})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j}\}
+\varepsilon_{\mathrm{enc},j}
\right]^{1/2}.
\end{aligned}
}
\label{eq:end-to-end-oracle}
\end{align}
\end{theorem}

\begin{corollary}[Consistency]
\label{cor:learned-aipw-consistency}
 Consider a sequence of encoder and critic classes, sample splits, nuisance estimators, optimization tolerances, and candidate-specific stability constants satisfying Theorem~\ref{thm:end-to-end-sensitivity} with common constants $B<\infty$ and $\eta\in(0,1/2]$; write $\widehat\phi_{j,n}$ for the selected encoder at total sample size $n$. If
\begin{align}
\delta_{\mathrm D,n}+\delta_{\mathrm E,n}&\longrightarrow0,\\
\frac{\log(2/\delta_{\mathrm E,n})}{n_{\mathrm E}}&\longrightarrow0,\\
q_{e,j,n}(q_{0,j,n}+q_{1,j,n})&\xrightarrow{p}0,
\end{align}
and
\begin{align}
\frac{\Gamma_j(\widehat\phi_{j,n})}{\eta(1-\eta)}
\left[
\inf_{\phi\in\mathcal F_{j,n_{\mathrm D}}}D_{\mathrm{res},j}^2(\phi)
+2\{r_{j,n_{\mathrm D}}(\delta_{\mathrm D,n})+a_{j,n_{\mathrm D}}+\varepsilon_{\mathrm{crit},j,n}\}
+\varepsilon_{\mathrm{enc},j,n}
\right]^{1/2}
\xrightarrow{p}0,
\label{eq:consistency-representation-condition}
\end{align}
then
\begin{align}
\widehat\tau_{j,n}^{\mathrm{AIPW}}
\xrightarrow{p}
\tau.
\end{align}
\end{corollary}

\begin{remark}[Interpretation and observability]
\label{rem:end-to-end-observability}
Theorem~\ref{thm:end-to-end-sensitivity} separates the final error into an AIPW estimation term and a hidden-confounding sensitivity term. The  selected stability constant $\Gamma_j(\widehat\phi_j)$, critic approximation error $a_{j,n_{\mathrm D}}$, optimization tolerances, and nuisance errors are not generally observed. The result becomes an operational numerical certificate after these quantities are bounded by theory, computation, external information, or a declared sensitivity analysis.
\end{remark}

\begin{remark}[Scope]
\label{rem:end-to-end-scope}
The theorem uses one prespecified held-out block, deterministic encoders, bounded critics, and an honest three-way split. It permits continuous representations and complexity-controlled neural classes, but it does not guarantee that a particular numerical optimizer reaches a global minimum. The random-search result below concerns only computational subsampling of population orientations and does not extend this fixed-$j$ learning and inference result.
\end{remark}

\subsection{Searching Across Held-Out Orientations}
\label{sec:random-orientation-search}

Exhaustive evaluation of all orientations in $\mathfrak S$ may be computationally expensive. We therefore consider randomized subsampling of the population candidates defined in Section~\ref{sec:unknown-valid-held-out-block}. This result concerns computational search over exact population values. It is not a finite-sample clustering or causal-inference guarantee.

Let $T\triangleq|\mathfrak S|=2^J-2$ and $N\triangleq|\mathcal B|$. Let the search budget $M\in\{0,\ldots,T\}$ be chosen independently of the observed data and all representation-learning outcomes. Conditional on $M=m$, sample $m$ orientations uniformly without replacement from $\mathfrak S$. Let $\mathcal R_m$ be the sampled orientations that also belong to $\mathcal B$, and let $R\triangleq|\mathcal R_m|$. Then
\begin{align}
R\mid M=m\sim\operatorname{Hypergeometric}(T,N,m).
\label{eq:retained-orientation-count}
\end{align}
Conditional on $(M=m,R=r)$, the retained set $\mathcal R_m$ is a uniformly sampled $r$-element subset of $\mathcal B$.

For $c\in\mathbb R$, define the population count
\begin{align}
n(c)\triangleq\bigl|\{S\in\mathcal B:\theta_S=c\}\bigr|.
\label{eq:population-orientation-count}
\end{align}
Let
\begin{align}
L\triangleq\bigl|\{c\in\mathbb R:c\ne\tau,\ n(c)>0\}\bigr|
\label{eq:number-nontarget-clusters}
\end{align}
be the number of non-target population values. When $L\ge1$, define
\begin{align}
p_\tau
\triangleq
\frac{n(\tau)}{N},
\qquad
p_\star
\triangleq
\max_{c\ne\tau:\,n(c)>0}\frac{n(c)}{N},
\qquad
\Delta\triangleq p_\tau-p_\star.
\label{eq:population-plurality-gap}
\end{align}

\begin{theorem}[Recovery by random orientation search]
\label{thm:random-orientation-plurality}
\normalfont
Suppose $\mathcal B\ne\varnothing$ and
\begin{align*}
n(\tau)>\max_{c\ne\tau:\,n(c)>0}n(c),
\end{align*}
where the maximum over an empty collection is zero. Fix $m\in\{0,\ldots,T\}$ such that $P(M=m)>0$, and fix an integer $1\le r_0\le\min(m,N)$. Suppose
\begin{align}
P(R\ge r_0\mid M=m)\ge1-\eta_R(m).
\label{eq:retained-count-probability}
\end{align}
The procedure returns failure when $R=0$. If $L\ge1$, select the value having the largest retained count and treat any tie as failure. Conditional on $M=m$, this procedure returns $\tau$ with probability at least
\begin{align}
\left[
1-\eta_R(m)-L\exp\left\{-\frac{r_0\Delta^2}{2}\right\}
\right]_+,
\qquad
[x]_+\triangleq\max(x,0).
\label{eq:random-search-plurality-probability}
\end{align}
If $L=0$, every retained candidate equals $\tau$, so recovery succeeds whenever $R\ge1$; neither $p_\tau$, $p_\star$, nor $\Delta$ is needed in this case.
\end{theorem}

The probability in Theorem~\ref{thm:random-orientation-plurality} concerns only the randomized computational search. If $M$ is random and $P(R\ge r_0)\ge1-\eta_R$, averaging the same argument over $M$ gives the unconditional version of equation~\eqref{eq:random-search-plurality-probability} with $\eta_R(m)$ replaced by $\eta_R$. When $L=0$, the success probability is at least $1-\eta_R$ provided $r_0\ge1$.
