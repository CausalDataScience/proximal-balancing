\section{Representation Learning and AIPW Estimation}
\label{sec:estimation}

The identification results concern one fixed representation. We now select a representation from data and estimate the adjustment functional that it induces. The representation-learning and AIPW analysis below uses one prespecified held-out block $j$ throughout.

\subsection{Representation Learning}
\label{sec:representation-learning}

Recall that one observed record is $O_i=(X_i,W_i,A_i,Y_i)$. Partition the observed sample into three mutually disjoint subsets,
\begin{align}
	\mathcal I_{\mathrm D},
	\qquad
	\mathcal I_{\mathrm N},
	\qquad
	\mathcal I_{\mathrm E},
	\label{eq:honest-three-way-split}
\end{align}
with respective sizes $n_{\mathrm D}$, $n_{\mathrm N}$, and $n_{\mathrm E}$. The discrepancy-learning sample $\mathcal I_{\mathrm D}$ selects a representation. The nuisance-learning sample $\mathcal I_{\mathrm N}$ estimates the outcome regressions and propensity used by AIPW. The evaluation sample $\mathcal I_{\mathrm E}$ calculates the final estimate. Conditional on the preceding samples, the objects evaluated on the next sample are fixed.

We now give a fully specified finite-sample result for deterministic maps $v\mapsto\phi(v)$. Assume every candidate takes values in a common finite set $\mathcal S_j\subset\mathbb R^{d_{Z_j}}$, and define $M_j\triangleq|\mathcal S_j|$.

Fix a finite candidate library
\begin{align}
	\mathcal F_j\subseteq\Phi_j,
	\qquad
	N_j\triangleq|\mathcal F_j|<\infty.
\end{align}

For $\phi\in\mathcal F_j$, define
\begin{align}
	Z_j^\phi\triangleq\phi(X,W_{-j})
\end{align}
and let $\mathcal R_j(\phi)$ be its set of positive-mass strata. For $z\in\mathcal R_j(\phi)$, write $p_{j,z}(\phi)\triangleq P(Z_j^\phi=z)$. Assume there are constants $p_{\min}>0$ and $\eta\in(0,1/2]$ such that every candidate and every $z\in\mathcal R_j(\phi)$ satisfy
\begin{align}
P(Z_j^\phi=z)&\ge p_{\min},
\label{eq:library-minimum-mass}\\
\eta
\le P(A=1\mid Z_j^\phi=z)&\le1-\eta.
\label{eq:library-overlap}
\end{align}
Assume also that the bounded kernel satisfies
\begin{align}
\sup_v\sqrt{\kappa_j(v,v)}\le\rho_j<\infty.
\label{eq:kernel-bound}
\end{align}
For each $\phi\in\mathcal F_j$, let $\Gamma_j(\phi)$ denote a valid stability constant in Assumption~\ref{ass:uniform-stability}. Require a common stability budget across the candidate library:
\begin{align}
\sup_{\phi\in\mathcal F_j}\Gamma_j(\phi)
\le
\overline\Gamma_j<\infty.
\label{eq:uniform-candidate-stability}
\end{align}
The two primitive held-out-block clauses of Assumption~\ref{ass:joint-latent-exchangeability} do not depend on the selected candidate. Representation overlap and outcome-relevant stability must hold for every candidate-specific representation.

\begin{definition}[Empirical held-out proxy discrepancy]
\label{def:empirical-held-out-proxy-discrepancy}
For observation $i\in\mathcal I_{\mathrm D}$ and candidate $\phi\in\mathcal F_j$, write
\begin{align}
Z_{j,i}^{\phi}\triangleq\phi(X_i,W_{-j,i}).
\end{align}
For treatment arm $a$ and stratum $z$, define
\begin{align}
N_{a,z}^{\phi}
&\triangleq
\sum_{i\in\mathcal I_{\mathrm D}}
\mathbf 1\{A_i=a,Z_{j,i}^{\phi}=z\},\\
N_z^\phi
&\triangleq
N_{0,z}^{\phi}+N_{1,z}^{\phi},\\
\widehat p_{j,z}(\phi)
&\triangleq
\frac{N_z^\phi}{n_{\mathrm D}}.
\end{align}
When $N_{a,z}^{\phi}>0$, define the empirical kernel mean
\begin{align}
\widehat m_{a,j,z}(\phi)
\triangleq
\frac{1}{N_{a,z}^{\phi}}
\sum_{i\in\mathcal I_{\mathrm D}}
\mathbf 1\{A_i=a,Z_{j,i}^{\phi}=z\}
\kappa_j\{(X_i,W_{j,i}),\cdot\}.
\end{align}
When both treatment-arm cells are nonempty, define
\begin{align}
\widehat\Delta_{j,z}(\phi)
\triangleq
\|\widehat m_{1,j,z}(\phi)-\widehat m_{0,j,z}(\phi)\|_{\mathcal H_j}
\end{align}
and
\begin{align}
\widehat D_j(\phi)
\triangleq
\sum_{z:N_z^\phi>0}
\widehat p_{j,z}(\phi)\widehat\Delta_{j,z}(\phi).
\label{eq:empirical-discrepancy}
\end{align}
If an occupied empirical stratum contains only one treatment arm, set $\widehat D_j(\phi)=+\infty$.
\end{definition}

\begin{proposition}[Uniform deviation for the finite-strata specialization]
\label{prop:uniform-discrepancy-deviation}
Let $\delta_{\mathrm D}\in(0,1)$ and suppose
\begin{align}
n_{\mathrm D}
\ge
\frac{8}{p_{\min}\eta}
\log\left(\frac{8M_jN_j}{\delta_{\mathrm D}}\right).
\label{eq:learning-sample-threshold}
\end{align}
Define
\begin{align}
e_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\triangleq
\rho_j
\left[
2+
\sqrt{2\log\left(\frac{4M_jN_j}{\delta_{\mathrm D}}\right)}
\right]
\sqrt{\frac{2}{n_{\mathrm D}p_{\min}\eta}}
\end{align}
and
\begin{align}
r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\triangleq{}&
2e_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\nonumber\\
&+
2\rho_j
\sqrt{
\frac{
2\left[
M_j\log2+
\log\left(\frac{4N_j}{\delta_{\mathrm D}}\right)
\right]
}{n_{\mathrm D}}
}.
\label{eq:discrepancy-radius}
\end{align}
Under equations~\eqref{eq:library-minimum-mass} through \eqref{eq:kernel-bound}, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
\boxed{
\sup_{\phi\in\mathcal F_j}
|\widehat D_j(\phi)-D_j(\phi)|
\le
r_{j,n_{\mathrm D}}(\delta_{\mathrm D}).
}
\label{eq:uniform-discrepancy-deviation}
\end{align}
\end{proposition}

\begin{definition}[Approximate empirical minimizer]
\label{def:approximate-empirical-minimizer}
The nonnegative number $\varepsilon_{\mathrm{opt},j}$ is the allowed gap between the empirical discrepancy attained by the learner and the smallest empirical discrepancy in the candidate library. The representation learner returns $\widehat\phi_j\in\mathcal F_j$ satisfying
\begin{align}
\widehat D_j(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_j}\widehat D_j(\phi)
+
\varepsilon_{\mathrm{opt},j}.
\label{eq:approximate-erm}
\end{align}
In this deterministic setting, the learned representation and its population adjustment functional are
\begin{align}
\widehat Z_j
&\triangleq
\widehat\phi_j(V_j),\nonumber\\
\tau_{\widehat Z_j}
&\triangleq
\mathbb E\!\left[
\mathbb E(Y\mid A=1,\widehat Z_j)
-
\mathbb E(Y\mid A=0,\widehat Z_j)
\right].
\label{eq:learned-adjustment-functional}
\end{align}
An exact empirical minimizer has $\varepsilon_{\mathrm{opt},j}=0$. The adjustment functional is used whenever its two conditional means are absolutely integrable under the law of $\widehat Z_j$.
\end{definition}

\begin{theorem}[Finite-sample representation learning]
\label{thm:representation-oracle}
Let $\delta_{\mathrm D}\in(0,1)$, suppose the minimum-mass, overlap, and bounded-kernel conditions in equations~\eqref{eq:library-minimum-mass}--\eqref{eq:kernel-bound} hold, and suppose the learning-sample size satisfies equation~\eqref{eq:learning-sample-threshold}. Let $r_{j,n_{\mathrm D}}(\delta_{\mathrm D})$ be the explicit radius in equation~\eqref{eq:discrepancy-radius}. Suppose $\widehat\phi_j\in\mathcal F_j$ satisfies equation~\eqref{eq:approximate-erm} for some $\varepsilon_{\mathrm{opt},j}\ge0$. Suppose Assumption~\ref{ass:consistency} holds and Assumption~\ref{ass:joint-latent-exchangeability} holds for block $j$. For every $\phi\in\mathcal F_j$, suppose Assumption~\ref{ass:uniform-stability} holds with $\Gamma_j(\phi)\le\overline\Gamma_j$. Then, with probability at least $1-\delta_{\mathrm D}$,
\begin{align}
\boxed{
D_j(\widehat\phi_j)
\le
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+
\varepsilon_{\mathrm{opt},j}.
}
\label{eq:representation-oracle-discrepancy}
\end{align}
On the same event,
\begin{align}
\boxed{
|\tau_{\widehat Z_j}-\tau|
\le
\overline\Gamma_j
\left[
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+
\varepsilon_{\mathrm{opt},j}
\right].
}
\label{eq:representation-oracle-causal}
\end{align}
\end{theorem}

\begin{remark}[Practical learner outside the finite-library theorem]
\label{rem:practical-learning-gap}
The theorem covers hard, finite-valued representations in a finite prespecified library. A neural learner using random features, soft assignments, and Adam requires additional bounds for neural-class complexity, feature approximation, soft-to-hard transfer, and optimization error. These guarantees are not supplied by Theorem~\ref{thm:representation-oracle}.
\end{remark}

\subsection{AIPW Estimation}
\label{sec:aipw-estimation}

Conditional on $\mathcal I_{\mathrm D}$ and any algorithmic seed used in training, the map $\widehat\phi_j$ is fixed and $\widehat Z_j=\widehat\phi_j(V_j)$. Define the conditional outcome means and representation propensity
\begin{align}
m_{a,j}(z)
&\triangleq
\mathbb E(Y\mid A=a,\widehat Z_j=z),\\
e_j(z)
&\triangleq
P(A=1\mid\widehat Z_j=z).
\label{eq:aipw-nuisances}
\end{align}
For the observed adjustment functional $\tau_{\widehat Z_j}$, the following is the AIPW score:
\begin{align}
\psi_j(O)
\triangleq{}&
m_{1,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)
\nonumber\\
&+
\frac{A\{Y-m_{1,j}(\widehat Z_j)\}}{e_j(\widehat Z_j)}
-
\frac{(1-A)\{Y-m_{0,j}(\widehat Z_j)\}}{1-e_j(\widehat Z_j)}
-
\tau_{\widehat Z_j}.
\label{eq:fixed-representation-eif}
\end{align}
When this score is square-integrable, it is the efficient influence function in the nonparametric model for $(\widehat Z_j,A,Y)$. Bounded outcomes together with uniform representation overlap provide one sufficient condition for square integrability. Use the nuisance-learning sample, with $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i})$, to estimate the nuisance functions. Denote the estimates by $\widehat m_{0,j}$, $\widehat m_{1,j}$, and $\widehat e_j$.

\begin{definition}[Honest AIPW estimator]
\label{def:honest-aipw}
For $i\in\mathcal I_{\mathrm E}$, define $\widehat Z_{j,i}=\widehat\phi_j(V_{j,i})$ and
\begin{align}
S_j(O_i)
\triangleq{}&
\widehat m_{1,j}(\widehat Z_{j,i})
-
\widehat m_{0,j}(\widehat Z_{j,i})
\nonumber\\
&+
\frac{A_i\{Y_i-\widehat m_{1,j}(\widehat Z_{j,i})\}}
{\widehat e_j(\widehat Z_{j,i})}
\nonumber\\
&-
\frac{(1-A_i)\{Y_i-\widehat m_{0,j}(\widehat Z_{j,i})\}}
{1-\widehat e_j(\widehat Z_{j,i})}.
\label{eq:aipw-score}
\end{align}
The honest AIPW estimator is
\begin{align}
\boxed{
\widehat\tau_j^{\mathrm{AIPW}}
\triangleq
\frac{1}{n_{\mathrm E}}
\sum_{i\in\mathcal I_{\mathrm E}}S_j(O_i).
}
\label{eq:honest-aipw}
\end{align}
\end{definition}

\begin{proposition}[Exact signed nuisance remainder]
\label{prop:aipw-remainder}
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$,
\begin{align}
&\mathbb E\{S_j(O)\}-\tau_{\widehat Z_j}
\nonumber\\
&=
\mathbb E
\Bigg[
\{\widehat e_j(\widehat Z_j)-e_j(\widehat Z_j)\}
\Bigg\{
\frac{\widehat m_{1,j}(\widehat Z_j)-m_{1,j}(\widehat Z_j)}
{\widehat e_j(\widehat Z_j)}
\nonumber\\
&\hspace{51mm}
+
\frac{\widehat m_{0,j}(\widehat Z_j)-m_{0,j}(\widehat Z_j)}
{1-\widehat e_j(\widehat Z_j)}
\Bigg\}
\Bigg].
\label{eq:aipw-remainder}
\end{align}
\end{proposition}

The remainder is zero if the propensity estimator is correct. It is also zero if both conditional outcome estimators are correct.

\begin{theorem}[Conditional finite-sample AIPW error]
\label{thm:conditional-aipw-error}
Suppose $|Y|\le B$ almost surely. Clip the nuisance estimates so that
\begin{align}
-B\le\widehat m_{a,j}(z)\le B,
\qquad
\eta\le\widehat e_j(z)\le1-\eta.
\label{eq:nuisance-clipping}
\end{align}
For a function $g$ of $\widehat Z_j$, define
\begin{align}
\|g\|_{2,\widehat Z_j}
\triangleq
\left[
\mathbb E\{g(\widehat Z_j)^2\mid
\mathcal I_{\mathrm D},\mathcal I_{\mathrm N}\}
\right]^{1/2}.
\end{align}
After the two learning samples are observed, define
\begin{align}
q_{e,j}&\triangleq\|\widehat e_j-e_j\|_{2,\widehat Z_j},\\
q_{a,j}&\triangleq\|\widehat m_{a,j}-m_{a,j}\|_{2,\widehat Z_j},
\qquad a\in\{0,1\}.
\label{eq:realized-nuisance-errors}
\end{align}
For $\delta_{\mathrm E}\in(0,1)$, define
\begin{align}
\varepsilon_{\mathrm{AIPW},j}
\triangleq{}&
2B\left(1+\frac{1}{\eta}\right)
\sqrt{\frac{2\log(2/\delta_{\mathrm E})}{n_{\mathrm E}}}
\nonumber\\
&+
\frac{q_{e,j}(q_{0,j}+q_{1,j})}{\eta}.
\label{eq:aipw-error-radius}
\end{align}
Conditional on $\mathcal I_{\mathrm D}$ and $\mathcal I_{\mathrm N}$, with probability at least $1-\delta_{\mathrm E}$ over $\mathcal I_{\mathrm E}$,
\begin{align}
\boxed{
|\widehat\tau_j^{\mathrm{AIPW}}-\tau_{\widehat Z_j}|
\le
\varepsilon_{\mathrm{AIPW},j}.
}
\label{eq:conditional-aipw-error}
\end{align}
\end{theorem}

\subsection{End-to-End Finite-Sample Bound}
\label{sec:end-to-end-bound}

\begin{theorem}[Finite-strata end-to-end sensitivity bound]
\label{thm:end-to-end-sensitivity}
Suppose the conditions of Proposition~\ref{prop:uniform-discrepancy-deviation}, Theorem~\ref{thm:representation-oracle}, and Theorem~\ref{thm:conditional-aipw-error} hold. Then, jointly over the three independent samples, with probability at least $1-\delta_{\mathrm D}-\delta_{\mathrm E}$,
\begin{align}
\boxed{
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
\le
\varepsilon_{\mathrm{AIPW},j}
+
\overline\Gamma_j
\left[
\widehat D_j(\widehat\phi_j)
+
r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
\right].
}
\label{eq:end-to-end-empirical}
\end{align}
It also satisfies the oracle form
\begin{align}
\boxed{
\begin{aligned}
|\widehat\tau_j^{\mathrm{AIPW}}-\tau|
&\le
\varepsilon_{\mathrm{AIPW},j}
\\
&\quad+
\overline\Gamma_j
\left[
\inf_{\phi\in\mathcal F_j}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D})
+
\varepsilon_{\mathrm{opt},j}
\right].
\end{aligned}
}
\label{eq:end-to-end-oracle}
\end{align}
\end{theorem}

\begin{corollary}[Consistency]
\label{cor:learned-aipw-consistency}
Consider a sequence of candidate libraries, sample splits, nuisance estimators, optimization tolerances, and stability budgets satisfying Theorem~\ref{thm:end-to-end-sensitivity}. If
\begin{align}
\delta_{\mathrm D,n}+\delta_{\mathrm E,n}&\longrightarrow0,\\
\frac{\log(2/\delta_{\mathrm E,n})}{n_{\mathrm E}}&\longrightarrow0,\\
q_{e,j,n}(q_{0,j,n}+q_{1,j,n})&\xrightarrow{p}0,
\end{align}
and
\begin{align}
\overline\Gamma_{j,n}
\left[
\inf_{\phi\in\mathcal F_{j,n}}D_j(\phi)
+
2r_{j,n_{\mathrm D}}(\delta_{\mathrm D,n})
+
\varepsilon_{\mathrm{opt},j,n}
\right]
\xrightarrow{p}0,
\label{eq:consistency-representation-condition}
\end{align}
then
\begin{align}
\widehat\tau_{j,n}^{\mathrm{AIPW}}
\xrightarrow{p}
\tau.
\end{align}
\end{corollary}

\begin{remark}[Interpretation and observability]
\label{rem:end-to-end-observability}
Theorem~\ref{thm:end-to-end-sensitivity} separates the final error into an AIPW estimation term and a hidden-confounding sensitivity term. The stability budget $\overline\Gamma_j$ and the nuisance errors $q_{e,j}$, $q_{0,j}$, and $q_{1,j}$ are not generally observed. The result becomes an operational numerical certificate after these quantities are bounded by additional theory, external information, or a declared sensitivity analysis.
\end{remark}

\begin{remark}[Scope]
\label{rem:end-to-end-scope}
The theorem uses one prespecified held-out block, hard finite strata, a finite candidate library, and an honest three-way split. It does not establish a neural-class rate, a soft-to-hard approximation bound, an Adam optimization guarantee, or a cross-fitting theorem. The random-search result below concerns only computational subsampling of population orientations and does not extend this fixed-$j$ learning and inference result.
\end{remark}

\subsection{Searching Across Held-Out Orientations}
\label{sec:random-orientation-search}

Exhaustive evaluation of all orientations in $\mathfrak S$ may be computationally expensive. We therefore consider randomized subsampling of the population candidates defined in Section~\ref{sec:unknown-valid-held-out-block}. This result concerns computational search over exact population values. It is not a finite-sample clustering or causal-inference guarantee.

Let $T\triangleq|\mathfrak S|=2^J-2$ and $N\triangleq|\mathcal B|$. Let the search budget $M\in\{0,\ldots,T\}$ be chosen independently of the observed data and all representation-learning outcomes. Conditional on $M=m$, sample $m$ orientations uniformly without replacement from $\mathfrak S$. Let $\mathcal R_m$ be the sampled orientations that also belong to $\mathcal B$, and let $R\triangleq|\mathcal R_m|$. Then
\begin{align}
R\mid M=m\sim\operatorname{Hypergeometric}(T,N,m).
\label{eq:retained-orientation-count}
\end{align}
Conditional on $(M=m,R=r)$, the retained set $\mathcal R_m$ is a uniformly sampled $r$-element subset of $\mathcal B$.

For $c\in\mathbb R$, define the population count
\begin{align}
n(c)\triangleq\bigl|\{S\in\mathcal B:\theta_S=c\}\bigr|.
\label{eq:population-orientation-count}
\end{align}
Let
\begin{align}
L\triangleq\bigl|\{c\in\mathbb R:c\ne\tau,\ n(c)>0\}\bigr|
\label{eq:number-nontarget-clusters}
\end{align}
be the number of non-target population values. When $L\ge1$, define
\begin{align}
p_\tau
\triangleq
\frac{n(\tau)}{N},
\qquad
p_\star
\triangleq
\max_{c\ne\tau:\,n(c)>0}\frac{n(c)}{N},
\qquad
\Delta\triangleq p_\tau-p_\star.
\label{eq:population-plurality-gap}
\end{align}

\begin{theorem}[Recovery by random orientation search]
\label{thm:random-orientation-plurality}
\normalfont
Suppose $\mathcal B\ne\varnothing$ and
\begin{align*}
n(\tau)>\max_{c\ne\tau:\,n(c)>0}n(c),
\end{align*}
where the maximum over an empty collection is zero. Fix $m\in\{0,\ldots,T\}$ such that $P(M=m)>0$, and fix an integer $1\le r_0\le\min(m,N)$. Suppose
\begin{align}
P(R\ge r_0\mid M=m)\ge1-\eta_R(m).
\label{eq:retained-count-probability}
\end{align}
The procedure returns failure when $R=0$. If $L\ge1$, select the value having the largest retained count and treat any tie as failure. Conditional on $M=m$, this procedure returns $\tau$ with probability at least
\begin{align}
\left[
1-\eta_R(m)-L\exp\left\{-\frac{r_0\Delta^2}{2}\right\}
\right]_+,
\qquad
[x]_+\triangleq\max(x,0).
\label{eq:random-search-plurality-probability}
\end{align}
If $L=0$, every retained candidate equals $\tau$, so recovery succeeds whenever $R\ge1$; neither $p_\tau$, $p_\star$, nor $\Delta$ is needed in this case.
\end{theorem}

The probability in Theorem~\ref{thm:random-orientation-plurality} concerns only the randomized computational search. If $M$ is random and $P(R\ge r_0)\ge1-\eta_R$, averaging the same argument over $M$ gives the unconditional version of equation~\eqref{eq:random-search-plurality-probability} with $\eta_R(m)$ replaced by $\eta_R$. When $L=0$, the success probability is at least $1-\eta_R$ provided $r_0\ge1$.
